	public static ManaGroup Parse(string shortString)
	{
		if (!TryParse(shortString, out var result))
		{
			throw new ArgumentException("Cannot convert " + shortString + " to ManaGroup");
		}
		return result;
	}

	public static bool TryParse(string shortString, out ManaGroup result)
	{
		result = default(ManaGroup);
		int i = 0;
		int num = 0;
		for (; i < shortString.Length && char.IsDigit(shortString[i]); i++)
		{
			num = num * 10 + (shortString[i] - 48);
		}
		result.Any += num;
		for (; i < shortString.Length; i++)
		{
			switch (shortString[i])
			{
			case ':':
			{
				if (i != shortString.Length - 3)
				{
					return false;
				}
				ManaColor? manaColor3 = ManaColors.FromShortName(shortString[i + 1]);
				ManaColor? manaColor4 = ManaColors.FromShortName(shortString[i + 2]);
				if (!manaColor3.HasValue || !manaColor4.HasValue)
				{
					return false;
				}
				result.SetHybridColor(manaColor3.Value, manaColor4.Value);
				return true;
			}
			case '{':
			{
				i++;
				num = 0;
				for (; i < shortString.Length; i++)
				{
					char c = shortString[i];
					if (c < '0' || c > '9')
					{
						break;
					}
					num = num * 10 + (c - 48);
				}
				if (i >= shortString.Length)
				{
					return false;
				}
				ManaColor? manaColor2 = ManaColors.FromShortName(shortString[i]);
				if (!manaColor2.HasValue)
				{
					return false;
				}
				i++;
				if (i >= shortString.Length || shortString[i] != '}')
				{
					return false;
				}
				result[manaColor2.Value] += num;
				break;
			}
			default:
			{
				ManaColor? manaColor = ManaColors.FromShortName(shortString[i]);
				if (!manaColor.HasValue)
				{
					return false;
				}
				ManaColor value = manaColor.Value;
				int value2 = result[value] + 1;
				result[value] = value2;
				break;
			}
			}
		}
		return true;
	}

	public bool CanAffordExact(ManaGroup cost)
	{
		if (Any != 0)
		{
			throw new InvalidOperationException(string.Format("{0}.{1} invoked with any = {2}, which mustn't be available mana.", "ManaGroup", "CanAffordExact", Any));
		}
		return (this - cost).IsValid;
	}

	public bool CanAfford(ManaGroup cost)
	{
		if (Any != 0)
		{
			throw new InvalidOperationException(string.Format("{0}.{1} invoked with any = {2}, which mustn't be available mana.", "ManaGroup", "CanAfford", Any));
		}
		if (Hybrid != 0)
		{
			throw new InvalidOperationException(string.Format("{0}.{1} invoked with hybrid = {2}, which mustn't be available mana.", "ManaGroup", "CanAfford", Hybrid));
		}
		ManaGroup manaGroup = this - cost;
		manaGroup.Any = 0;
		if (manaGroup.Amount < cost.Any)
		{
			return false;
		}
		int num = manaGroup.Philosophy;
		foreach (ManaColor singleColor in ManaColors.SingleColors)
		{
			int num2 = manaGroup[singleColor];
			if (num2 < 0)
			{
				num += num2;
			}
		}
		if (cost.Hybrid > 0)
		{
			List<ManaColor> list = HybridColors[cost.HybridColor];
			ManaColor color = list[0];
			ManaColor color2 = list[1];
			int num3 = Math.Max(0, manaGroup[color]);
			int num4 = Math.Max(0, manaGroup[color2]);
			int num5 = num3 + num4 - cost.Hybrid;
			if (num5 < 0)
			{
				num += num5;
			}
		}
		return num >= 0;
	}

	public override string ToString()
	{
		StringBuilder stringBuilder = new StringBuilder();
		if (Any > 0)
		{
			stringBuilder.Append(Any);
		}
		else if (Any < 0)
		{
			stringBuilder.Append("(").Append(Any).Append(")");
		}
		AppendValue(stringBuilder, ManaColor.White, White);
		AppendValue(stringBuilder, ManaColor.Blue, Blue);
		AppendValue(stringBuilder, ManaColor.Black, Black);
		AppendValue(stringBuilder, ManaColor.Red, Red);
		AppendValue(stringBuilder, ManaColor.Green, Green);
		AppendValue(stringBuilder, ManaColor.Colorless, Colorless);
		AppendValue(stringBuilder, ManaColor.Philosophy, Philosophy);
		AppendValue(stringBuilder, ManaColor.Hybrid, Hybrid);
		if (Hybrid > 0)
		{
			stringBuilder.Append(":").Append(HybridColorShortNames[HybridColor]);
		}
		if (stringBuilder.Length != 0)
		{
			return stringBuilder.ToString();
		}
		return "0";
		static void AppendValue(StringBuilder builder, ManaColor color, int value)
		{
			if (value > 3 || value < 0)
			{
				builder.Append('{').Append(value).Append(color.ToShortName())
					.Append('}');
			}
			else
			{
				for (int i = 0; i < value; i++)
				{
					builder.Append(color.ToShortName());
				}
			}
		}
	}

	private static bool Equals(ManaGroup lhs, ManaGroup rhs)
	{
		if ((lhs.Hybrid != 0 || rhs.Hybrid != 0) && lhs.HybridColor != rhs.HybridColor)
		{
			return false;
		}
		return ManaColors.AllColorsWithHybrid.All((ManaColor color) => lhs[color] == rhs[color]);
	}

		if (obj is ManaGroup rhs)
		{
			return Equals(this, rhs);
		}
		return false;
	}

	public override int GetHashCode()
	{
		HashCode hashCode = default(HashCode);
		hashCode.Add(Any);
		hashCode.Add(White);
		hashCode.Add(Blue);
		hashCode.Add(Black);
		hashCode.Add(Red);
		hashCode.Add(Green);
		hashCode.Add(Colorless);
		hashCode.Add(Philosophy);
		hashCode.Add(Hybrid);
		hashCode.Add(HybridColor);
		return hashCode.ToHashCode();
	}

	public static bool operator ==(ManaGroup lhs, ManaGroup rhs)
	{
		return Equals(lhs, rhs);
	}

	public static bool operator !=(ManaGroup lhs, ManaGroup rhs)
	{
		return !Equals(lhs, rhs);
	}

	public void SetHybridColor(int hybridColorCode)
	{
		if (hybridColorCode >= 0 && hybridColorCode <= 9)
		{
			HybridColor = hybridColorCode;
		}
		else
		{
			Debug.LogError("Setting Hybrid color: hybridColorCode is out of range.");
		}
	}

	public void SetHybridColor(ManaColor color1, ManaColor color2)
	{
		if (color1 != color2 && ManaColors.TrivialColors.Contains(color1) && ManaColors.TrivialColors.Contains(color2))
		{
			for (int i = 0; i < 10; i++)
			{
				if (HybridColors[i].Contains(color1) && HybridColors[i].Contains(color2))
				{
					HybridColor = i;
					break;
				}
			}
		}
		else
		{
			Debug.LogError($"Setting Hybrid color with wrong colors: {color1} and {color2}");
		}
	}

	public void Visit(Action<int> visitor)
	{
		visitor(Any);
		visitor(White);
		visitor(Blue);
		visitor(Black);
		visitor(Red);
		visitor(Green);
		visitor(Colorless);
		visitor(Philosophy);
	}

	public void Visit(Func<int, int> visitor)
	{
		Any = visitor(Any);
		White = visitor(White);
		Blue = visitor(Blue);
		Black = visitor(Black);
		Red = visitor(Red);
		Green = visitor(Green);
		Colorless = visitor(Colorless);
		Philosophy = visitor(Philosophy);
	}

	public bool AllColorsAre(Predicate<int> predicator)
	{
		if (predicator(Any) && predicator(White) && predicator(Blue) && predicator(Black) && predicator(Red) && predicator(Green) && predicator(Colorless))
		{
			return predicator(Philosophy);
		}
		return false;
	}

	public bool AnyColorIs(Predicate<int> predicator)
	{
		if (!predicator(Any) && !predicator(White) && !predicator(Blue) && !predicator(Black) && !predicator(Red) && !predicator(Green) && !predicator(Colorless))
		{
			return predicator(Philosophy);
		}
		return true;
	}
