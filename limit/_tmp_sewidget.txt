using DG.Tweening;
using LBoL.Core;
using LBoL.Core.StatusEffects;
using LBoL.Presentation.I10N;
using LBoL.Presentation.Units;
using TMPro;
using UnityEngine;
using UnityEngine.UI;

namespace LBoL.Presentation.UI.Widgets;

public class StatusEffectWidget : MonoBehaviour
{
	[SerializeField]
	private Image image;

	[SerializeField]
	private Image effectImage;

	[SerializeField]
	private TextMeshProUGUI upText;

	[SerializeField]
	private TextMeshProUGUI downText;

	[SerializeField]
	private ParticleSystem hintParticle;

	private StatusEffect _statusEffect;

	private bool _changed;

	private bool _isHighlight;

	public Vector3 CenterWorldPosition
	{
		get
		{
			RectTransform obj = (RectTransform)base.transform;
			return obj.TransformPoint(obj.rect.center);
		}
	}

	public StatusEffect StatusEffect
	{
		get
		{
			return _statusEffect;
		}
		set
		{
			if (_statusEffect != null)
			{
				RemoveHandlers(_statusEffect);
			}
			_statusEffect = value;
			if (value != null)
			{
				AddHandlers(value);
			}
			Refresh();
		}
	}

	private void Awake()
	{
		effectImage.gameObject.SetActive(value: false);
	}

	private void LateUpdate()
	{
		if (_changed)
		{
			_changed = false;
			TextRefresh();
			string overrideIconName = _statusEffect.OverrideIconName;
			if (overrideIconName != null)
			{
				Sprite sprite = ResourcesHelper.TryGetSprite<StatusEffect>(overrideIconName);
				if ((object)sprite != null)
				{
					image.sprite = sprite;
				}
			}
		}
		if (!_isHighlight && _statusEffect.Highlight)
		{
			PlayHighlight();
		}
		if (_isHighlight && !_statusEffect.Highlight)
		{
			CleanHighlight();
		}
	}

	private void PlayHighlight()
	{
		_isHighlight = true;
		hintParticle.gameObject.SetActive(value: true);
		hintParticle.Play();
	}

	private void CleanHighlight()
	{
		_isHighlight = false;
		hintParticle.Stop();
		hintParticle.gameObject.SetActive(value: true);
	}

	private void OnEnable()
	{
		L10nManager.LocaleChanged += OnLocaleChanged;
	}

	private void OnDisable()
	{
		L10nManager.LocaleChanged -= OnLocaleChanged;
	}

	private void OnLocaleChanged()
	{
		if (_statusEffect != null)
		{
			_changed = true;
		}
	}

	private void OnDestroy()
	{
		if (_statusEffect != null)
		{
			_statusEffect.PropertyChanged -= OnPropertyChanged;
			_statusEffect.Activating -= OnActivating;
		}
		_statusEffect = null;
		effectImage.DOKill();
	}

	private void AddHandlers(StatusEffect effect)
	{
		effect.Activating += OnActivating;
		effect.PropertyChanged += OnPropertyChanged;
	}

	private void RemoveHandlers(StatusEffect effect)
	{
		effect.PropertyChanged -= OnPropertyChanged;
		effect.Activating -= OnActivating;
	}

	private void OnActivating()
	{
		ShowActivating();
	}

	private void OnPropertyChanged()
	{
		_changed = true;
		string unitEffectName = _statusEffect.UnitEffectName;
		if (unitEffectName != null)
		{
			GameDirector.GetUnit(_statusEffect.Owner).SendEffectMessage(unitEffectName, "OnPropertyChanged", _statusEffect);
		}
	}

	public void Refresh()
	{
		if (_statusEffect != null)
		{
			Sprite sprite = ResourcesHelper.TryGetSprite<StatusEffect>(_statusEffect.OverrideIconName ?? _statusEffect.IconName);
			if ((object)sprite != null)
			{
				image.sprite = sprite;
			}
			TextRefresh();
		}
		else
		{
			image.sprite = null;
			upText.text = string.Empty;
			downText.text = string.Empty;
		}
	}

	private void TextRefresh()
	{
		upText.text = ((_statusEffect.HasCount && _statusEffect.ShowCount) ? _statusEffect.Count.ToString() : string.Empty);
		if (_statusEffect.ForceNotShowDownText)
		{
			downText.text = string.Empty;
		}
		else if (_statusEffect.HasLevel)
		{
			downText.text = _statusEffect.Level.ToString();
			downText.color = GlobalConfig.UiRed;
			if (_statusEffect.HasDuration)
			{
				Debug.LogWarning(base.name + " has both Level and Duration, which cause UI display problem.");
			}
		}
		else if (_statusEffect.HasDuration)
		{
			downText.text = _statusEffect.Duration.ToString();
			downText.color = GlobalConfig.UiBlue;
		}
		else
		{
			downText.text = string.Empty;
		}
	}

	public YieldInstruction ShowAdded()
	{
		effectImage.gameObject.SetActive(value: true);
		effectImage.sprite = image.sprite;
		effectImage.DOKill();
		return DOTween.Sequence().Append(effectImage.transform.DOScale(Vector3.one * 3f, 0.8f).From(Vector3.one)).Join(effectImage.DOFade(0f, 0.8f).From(1f))
			.SetEase(Ease.OutQuad)
			.OnComplete(delegate
			{
				effectImage.gameObject.SetActive(value: false);
			})
			.SetTarget(effectImage)
			.WaitForCompletion();
	}

	public YieldInstruction ShowActivating()
	{
		effectImage.gameObject.SetActive(value: true);
		effectImage.sprite = image.sprite;
		effectImage.DOKill();
		return DOTween.Sequence().Append(effectImage.transform.DOScale(Vector3.one, 0.8f).From(Vector3.one * 3f)).Join(effectImage.DOFade(1f, 0.4f).From(0f).SetLoops(1, LoopType.Yoyo))
			.SetEase(Ease.OutQuad)
			.OnComplete(delegate
			{
				effectImage.gameObject.SetActive(value: false);
			})
			.SetTarget(effectImage)
			.WaitForCompletion();
	}
}
