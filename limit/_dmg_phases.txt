GetPhases()
	{
		if (DealingArgs.Targets.All((Unit t) => t.IsDead))
		{
			Debug.LogWarning("[DamageAction] Targets [" + string.Join(", ", DealingArgs.Targets.Select((Unit t) => t.Name)) + "] all dead, action is canceled");
			yield break;
		}
		Unit source = DealingArgs.Source;
		if (source != null)
		{
			yield return CreateEventPhase("DamageDealing", DealingArgs, source.DamageDealing);
		}
		_dealingPassed = true;
		DamageInfo damageInfo = DealingArgs.DamageInfo;
		foreach (var item4 in DealingArgs.Targets.Zip(DamageArgs))
		{
			Unit item = item4.Item1;
			DamageEventArgs item2 = item4.Item2;
			item2.DamageInfo = damageInfo;
			if (item2.Target.IsInvalidTarget)
			{
				item2.ForceCancelBecause(CancelCause.InvalidTarget);
			}
			else
			{
				yield return CreateEventPhase("DamageReceiving", item2, item.DamageReceiving);
			}
		}
		yield return CreatePhase("MeasureDamage", delegate
		{
			DamageEventArgs[] damageArgs4 = DamageArgs;
			foreach (DamageEventArgs damageEventArgs7 in damageArgs4)
			{
				if (damageEventArgs7.IsCanceled)
				{
					break;
				}
				damageEventArgs7.DamageInfo = damageEventArgs7.Target.MeasureDamage(damageEventArgs7.DamageInfo);
				damageEventArgs7.IsModified = true;
			}
		});
		DamageEventArgs[] damageArgs = DamageArgs;
		foreach (DamageEventArgs damageEventArgs in damageArgs)
		{
			Unit source2 = damageEventArgs.Source;
			if (source2 != null && !damageEventArgs.IsCanceled)
			{
				yield return CreateEventPhase("DamageGiving", damageEventArgs, source2.DamageGiving);
			}
		}
		damageArgs = DamageArgs;
		foreach (DamageEventArgs damageEventArgs2 in damageArgs)
		{
			if (!damageEventArgs2.IsCanceled)
			{
				yield return CreateEventPhase("DamageTaking", damageEventArgs2, damageEventArgs2.Target.DamageTaking);
			}
		}
		yield return CreatePhase("Main", delegate
		{
			DamageEventArgs[] damageArgs3 = DamageArgs;
			foreach (DamageEventArgs damageEventArgs6 in damageArgs3)
			{
				if (damageEventArgs6.IsCanceled)
				{
					break;
				}
				damageEventArgs6.DamageInfo = base.Battle.Damage(damageEventArgs6.Source, damageEventArgs6.Target, damageEventArgs6.DamageInfo, base.Source);
				damageEventArgs6.IsModified = true;
			}
		}, hasViewer: true);
		List<(Unit, DieCause)> deadUnits = new List<(Unit, DieCause)>();
		DamageEventArgs[] damageArgs2 = DamageArgs;
		foreach (DamageEventArgs damageEventArgs3 in damageArgs2)
		{
			Unit target = damageEventArgs3.Target;
			if (target.Hp == 0 && target.Status == UnitStatus.Alive)
			{
				target.Status = UnitStatus.Dying;
				DieCause item3 = damageEventArgs3.DamageInfo.DamageType switch
				{
					DamageType.HpLose => DieCause.LoseHp, 
					DamageType.Reaction => DieCause.Reaction, 
					DamageType.Attack => DieCause.Attack, 
					_ => throw new ArgumentOutOfRangeException(), 
				};
				deadUnits.Add((target, item3));
				damageEventArgs3.Kill = true;
			}
		}
		if (deadUnits.Count > 0)
		{
			yield return CreatePhase("Dying", delegate
			{
				React(new DieAction(deadUnits, DealingArgs.Source, DealingArgs.ActionSource), base.Source);
			});
		}
		damageArgs = DamageArgs;
		foreach (DamageEventArgs damageEventArgs4 in damageArgs)
		{
			if (!damageEventArgs4.IsCanceled)
			{
				Unit source3 = damageEventArgs4.Source;
				if (source3 != null)
				{
					yield return CreateEventPhase("DamageDealt", damageEventArgs4, source3.DamageDealt);
				}
			}
		}
