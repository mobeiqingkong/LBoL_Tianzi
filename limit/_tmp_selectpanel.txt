	public void Awake()
	{
		confirmButton.onClick.AddListener(Confirm);
		cancelButton.onClick.AddListener(Cancel);
		miniSkipButton.onClick.AddListener(Skip);
		miniCancelButton.onClick.AddListener(Cancel);
		cardParent.transform.DestroyChildren();
	}

	protected override void OnEnterGameRun()
	{
		base.GameRun.InteractionViewer.Register<SelectHandInteraction>(ViewSelectHand);
		base.GameRun.InteractionViewer.Register<SelectCardInteraction>(ViewSelectCard);
		base.GameRun.InteractionViewer.Register<MiniSelectCardInteraction>(ViewMiniSelect);
	}

	protected override void OnLeaveGameRun()
	{
		base.GameRun.InteractionViewer.Unregister<SelectHandInteraction>(ViewSelectHand);
		base.GameRun.InteractionViewer.Unregister<SelectCardInteraction>(ViewSelectCard);
		base.GameRun.InteractionViewer.Unregister<MiniSelectCardInteraction>(ViewMiniSelect);
	}

	protected override void OnShowing(SelectCardPayload payload)
	{
		UiManager.GetPanel<PlayBoard>().SetUnitGamepadSelectionBlocked(blocked: true);
		_payload = payload;
		if (!miniSelectCardRoot.gameObject.activeSelf || !_gamepadDefaultSelection || !_gamepadDefaultSelection.activeInHierarchy)
		{
			_gamepadDefaultSelection = null;
		}
		minimizedButton.Init();
		foreach (SelectCardWidget selectCardWidget3 in _selectCardWidgets)
		{
			UnityEngine.Object.Destroy(selectCardWidget3.gameObject);
		}
		_selectCardWidgets.Clear();
		_selectIndexOrder.Clear();
		cancelButton.gameObject.SetActive(payload.CanCancel);
		miniCancelButton.gameObject.SetActive(payload.CanCancel);
		Min = payload.Min;
		Max = payload.Max;
		Sortable = payload.Sortable;
		confirmButton.interactable = 0 >= Min;
		SelectedCards = null;
		IsCanceled = false;
		foreach (Card card in payload.Cards)
		{
			CardWidget cardWidget = UnityEngine.Object.Instantiate(cardTemplate, cardParent);
			cardWidget.Card = card;
			if (base.GameRun.Player.HasExhibit<ZhinengYinxiang>())
			{
				int? cardInstanceId = base.GameRun.Player.GetExhibit<ZhinengYinxiang>().CardInstanceId;
				if (cardInstanceId.HasValue && cardInstanceId.Value == card.InstanceId)
				{
					cardWidget.ShowStickerYinxiang = true;
				}
			}
			cardWidget.gameObject.AddComponent<GamepadScrollRectItem>();
			ShowingCard showingCard = cardWidget.gameObject.AddComponent<ShowingCard>();
			GamepadCardController component = cardWidget.GetComponent<GamepadCardController>();
			component.AddGamepadButton(GamepadButtonKey.X, showingCard.OnGamepadClickCardDetail);
			component.CardCursor.OnSelectChanged.AddListener(showingCard.OnGamepadSelectedChanged);
			if (_selectCardWidgets.Count == 0)
			{
				component.SetNavigationOrigin(value: true);
				if (normalSelectCardRoot.gameObject.activeSelf)
				{
					SetGamepadDefaultSelection(component.CardCursor.gameObject);
				}
			}
			showingCard.SetScale(1f, 1.1f);
			SelectCardWidget selectCardWidget = cardWidget.gameObject.AddComponent<SelectCardWidget>();
			selectCardWidget.SelectParticle = UnityEngine.Object.Instantiate(selectParticle, selectCardWidget.transform);
			selectCardWidget.SelectParticle.SetActive(value: false);
			selectCardWidget.SelectedChanged += delegate(object sender, EventArgs args)
			{
				SelectCardWidget widget = sender as SelectCardWidget;
				if ((object)widget != null)
				{
					if (widget.IsSelected && !_selectIndexOrder.Exists((int t) => t == _selectCardWidgets.IndexOf(widget)))
					{
						_selectIndexOrder.Add(_selectCardWidgets.IndexOf(widget));
					}
					if (!widget.IsSelected)
					{
						_selectIndexOrder.Remove(_selectCardWidgets.IndexOf(widget));
					}
				}
				int num = _selectCardWidgets.Count((SelectCardWidget w) => w.IsSelected);
				if (num > Max)
				{
					_selectCardWidgets[_selectIndexOrder[0]].SetSelected(select: false, notify: false);
					_selectIndexOrder.RemoveAt(0);
				}
				num = _selectCardWidgets.Count((SelectCardWidget w) => w.IsSelected);
				confirmButton.interactable = num >= Min && num <= Max;
				if (Sortable)
				{
					foreach (var item in _selectCardWidgets.WithIndices())
					{
						var (i, selectCardWidget2) = item;
						if (_selectIndexOrder.Exists((int t) => t == i))
						{
							selectCardWidget2.CardWidget.ShowDeckIndex(_selectIndexOrder.IndexOf(i) + 1);
						}
						else
						{
							selectCardWidget2.CardWidget.HideDeckIndex();
						}
					}
				}
			};
			_selectCardWidgets.Add(selectCardWidget);
		}
		cardParent.anchoredPosition = Vector2.zero;
		Cursor.visible = true;
		UiManager.PushActionHandler(this);
	}

	protected override void OnHiding()
	{
		UiManager.GetPanel<PlayBoard>().SetUnitGamepadSelectionBlocked(blocked: false);
		UiManager.GetPanel<PlayBoard>().SetCursorVisible();
		_payload = null;
		_gamepadDefaultSelection = null;
		UiManager.PopActionHandler(this);
	}

	protected override void OnHided()
	{
		foreach (SelectCardWidget selectCardWidget in _selectCardWidgets)
		{
			UnityEngine.Object.Destroy(selectCardWidget.gameObject);
		}
		_selectCardWidgets.Clear();
	}

	private void LateUpdate()
	{
		if (!base.IsVisible || Singleton<InputDeviceManager>.Instance.CurrentInputDevice != InputDeviceType.Gamepad || Singleton<GamepadNavigationManager>.Instance.GetTopPanel() != GetPanelName())
		{
			return;
		}
		GameObject gameObject = (EventSystem.current ? EventSystem.current.currentSelectedGameObject : null);
		if (!(gameObject != null) || !gameObject.transform.IsChildOf(base.transform))
		{
			if (!_gamepadDefaultSelection || !_gamepadDefaultSelection.activeInHierarchy)
			{
				EventSystem.current?.SetSelectedGameObject(null);
			}
			else
			{
				SelectDefaultForGamepad();
			}
		}
	}

	private void SetGamepadDefaultSelection(GameObject target)
	{
		_gamepadDefaultSelection = target;
		if ((bool)target)
		{
			GamepadNavigationManager.SetOverrideOrigin(target, GetPanelName());
		}
	}

	private void SelectDefaultForGamepad()
	{
		if ((bool)_gamepadDefaultSelection && _gamepadDefaultSelection.activeInHierarchy)
		{
			GamepadNavigationManager.SetOverrideOrigin(_gamepadDefaultSelection, GetPanelName());
			GamepadNavigationManager.AsnycSelectGameObject(_gamepadDefaultSelection, forceAsnyc: true);
		}
	}

	public bool SwitchMinimized()
	{
		if (!base.IsVisible)
		{
			return false;
		}
		CanvasGroup component = GetComponent<CanvasGroup>();
		if (!component || !component.interactable)
		{
			return false;
		}
		minimizedButton.SwitchMinimized();
		return true;
	}

	private void Confirm()
	{
		SelectedCards = new List<Card>();
		if (Sortable)
		{
			foreach (SelectCardWidget item in from index in _selectIndexOrder
				select _selectCardWidgets.TryGetValue(index) into selectWidget
				where (bool)selectWidget && selectWidget.IsSelected
				select selectWidget)
			{
				SelectedCards.Add(item.Card);
			}
		}
		else
		{
			foreach (SelectCardWidget selectCardWidget in _selectCardWidgets)
			{
				if (selectCardWidget.IsSelected)
				{
					SelectedCards.Add(selectCardWidget.Card);
				}
			}
		}
		IsCanceled = false;
		Hide();
	}

	private void Cancel()
	{
		IsCanceled = true;
		Hide();
	}

	private void Skip()
	{
		Hide();
	}

	public void OnCancel()
	{
		if (_payload != null)
		{
			if (_payload.CanCancel)
			{
				Cancel();
			}
		}
		else
		{
			Hide();
		}
	}

	public IEnumerator ShowAsync(SelectCardPayload payload)
	{
		Show(payload);
		yield return new WaitWhile(() => base.IsVisible);
	}

	public IEnumerator ShowMiniSelect(SelectCardPayload payload)
	{
		selectCardLayout.transform.DestroyChildren();
		miniSelectCardRoot.gameObject.SetActive(value: true);
		normalSelectCardRoot.gameObject.SetActive(value: false);
		titleRoot.GetComponent<RectTransform>().anchoredPosition = new Vector2(0f, -300f);
		miniSkipButton.gameObject.SetActive(payload.CanSkip);
		SetTitle(payload.Name);
		List<Card> cards = payload.Cards.ToList();
		int total = cards.Count;
		if (total > 5)
		{
			total = 5;
			cards = cards.Take(total).ToList();
			Debug.LogError("Too many cards in reward.");
		}
		bool slidePlayed = false;
		DOTween.Sequence().Insert(0f, miniCg.DOFade(1f, 0.2f).From(0f)).SetLink(miniSelectCardRoot.gameObject)
			.OnComplete(delegate
			{
				foreach (var item2 in cards.WithIndices())
				{
					int i = item2.index;
					Card item = item2.elem;
					CardWidget widget = UnityEngine.Object.Instantiate(cardTemplate, selectCardLayout.transform);
					widget.Card = item;
					widget.name = "Card:" + item.Name;
					if (i == 0)
					{
						GamepadCardController component = widget.GetComponent<GamepadCardController>();
						component.SetNavigationOrigin(value: true);
						SetGamepadDefaultSelection(component.CardCursor.gameObject);
					}
					Transform transform = widget.transform;
					transform.localPosition = new Vector3(-2300f, 0f, 0f);
					transform.localRotation = Quaternion.Euler(0f, 180f, 0f);
					transform.localScale = new Vector3(1.1f, 1.1f, 1f);
					DOTween.Sequence().AppendInterval(0.1f + (float)i * 0.05f).AppendCallback(delegate
					{
						if (!slidePlayed)
						{
							AudioManager.PlayUi("CardSlide");
							slidePlayed = true;
						}
					})
						.Append(transform.DOLocalMoveX(_xDistance[total - 1] * ((float)i - (float)(total - 1) / 2f), 0.45f).SetEase(Ease.OutCubic))
						.AppendInterval(0.1f + (float)i * 0.05f)
						.AppendCallback(delegate
						{
							AudioManager.PlayUi("CardFlip_" + (5 - total + i + 1));
						})
						.Append(transform.DOLocalRotate(Vector3.zero, 0.35f).SetEase(Ease.OutQuad))
						.SetAutoKill(autoKillOnCompletion: true)
						.SetUpdate(isIndependentUpdate: true)
						.OnComplete(delegate
						{
							ShowingCard showingCard = widget.gameObject.AddComponent<ShowingCard>();
							GamepadCardController component2 = widget.GetComponent<GamepadCardController>();
							component2.AddGamepadButton(GamepadButtonKey.X, showingCard.OnGamepadClickCardDetail);
							component2.CardCursor.OnSelectChanged.AddListener(showingCard.OnGamepadSelectedChanged);
							showingCard.SetScale(1.1f);
							GetMiniSelectCardButton(widget).onClick.AddListener(delegate
							{
								SelectedCards = new List<Card> { widget.Card };
								if (payload.IsAddCardToDeck)
								{
									_bufferedCardWidget = widget;
								}
								Hide();
							});
							if (i == 0)
							{
								SelectDefaultForGamepad();
							}
						});
				}
			});
		yield return ShowAsync(payload);
		miniCg.DOFade(0f, 0.2f).From(1f).OnComplete(delegate
		{
			miniSelectCardRoot.gameObject.SetActive(value: false);
		});
	}

	private IEnumerator ViewSelectHand(SelectHandInteraction interaction)
	{
		miniSelectCardRoot.gameObject.SetActive(value: false);
		normalSelectCardRoot.gameObject.SetActive(value: true);
		normalSelectCardRoot.sizeDelta = normalSelectCardRoot.sizeDelta.WithY((interaction.PendingCards.Count > 5) ? 1600f : 1200f);
		titleRoot.anchoredPosition = new Vector2(0f, (interaction.PendingCards.Count > 5) ? (-200f) : (-400f));
		confirmButton.gameObject.SetActive(value: true);
		SetTitle(interaction);
		yield return ShowAsync(new SelectCardPayload
		{
			Cards = interaction.PendingCards,
			Min = interaction.Min,
			Max = interaction.Max,
			Sortable = interaction.Sortable,
			CanCancel = interaction.CanCancel
		});
		if (IsCanceled)
		{
			interaction.Cancel();
			yield break;
		}
		interaction.SelectedCards = SelectedCards;
		SelectedCards = null;
	}

	private void SetTitle(Interaction interaction)
	{
		string description = interaction.Description;
		GameEntity source = interaction.Source;
		if (description != null)
		{
			titleTmp.text = description;
			titleRoot.gameObject.SetActive(value: true);
		}
		else if (source != null)
		{
			titleTmp.text = source.Name;
			if (source is Card { IsUpgraded: not false })
			{
				titleTmp.text += "+";
			}
			titleRoot.gameObject.SetActive(value: true);
		}
		else
		{
			titleRoot.gameObject.SetActive(value: false);
		}
	}

	private void SetTitle(string titleText)
	{
		titleTmp.text = titleText;
		titleRoot.gameObject.SetActive(value: true);
	}

	public CardWidget ExtractBufferedCard()
	{
		if (!_bufferedCardWidget)
		{
			Debug.LogWarning("[RewardPanel] Cannot extract buffered card: no buffered card");
			return null;
		}
		CardWidget bufferedCardWidget = _bufferedCardWidget;
		_bufferedCardWidget = null;
		bufferedCardWidget.HideTooltip();
		return bufferedCardWidget;
	}
