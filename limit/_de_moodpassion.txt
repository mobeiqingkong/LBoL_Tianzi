using JetBrains.Annotations;
using LBoL.Base;
using LBoL.Core;
using LBoL.Core.Battle;
using LBoL.Core.Battle.BattleActions;
using LBoL.Core.Cards;
using LBoL.Core.StatusEffects;
using LBoL.Core.Units;

namespace LBoL.EntityLib.StatusEffects.Koishi;

public sealed class MoodPassion : Mood
{
	private bool HasLunatic
	{
		get
		{
			if (base.Owner != null)
			{
				return base.Owner.HasStatusEffect<LunaticPassionSe>();
			}
			return false;
		}
	}

	[UsedImplicitly]
	public int AttackPercentage
	{
		get
		{
			if (!HasLunatic)
			{
				return 100;
			}
			return 150;
		}
	}

	[UsedImplicitly]
	public int Percentage => 100;

	private float AttackRatio => ((float)AttackPercentage + 100f) / 100f;

	private float Ratio => ((float)Percentage + 100f) / 100f;

	public override string UnitEffectName => "BenwoLoop";

	protected override string GetBaseDescription()
	{
		if (!HasLunatic)
		{
			return base.GetBaseDescription();
		}
		return base.ExtraDescription;
	}

	protected override void OnAdded(Unit unit)
	{
		base.OnAdded(unit);
		HandleOwnerEvent(base.Owner.DamageDealing, OnDamageDealing);
		HandleOwnerEvent(base.Owner.DamageReceiving, OnDamageReceiving);
		if (base.Owner.HasStatusEffect<PassionDrawSe>())
		{
			int level = base.Owner.GetStatusEffect<PassionDrawSe>().Level;
			React(new DrawManyCardAction(level));
		}
	}

	private void OnDamageDealing(DamageDealingEventArgs args)
	{
		if (args.DamageInfo.DamageType == DamageType.Attack)
		{
			args.DamageInfo = args.DamageInfo.MultiplyBy(AttackRatio);
			if (HasLunatic && args.ActionSource is Card card && card.Config.Type == CardType.Attack)
			{
				DamageInfo damageInfo = args.DamageInfo;
				damageInfo.IsAccuracy = true;
				args.DamageInfo = damageInfo;
			}
			args.AddModifier(this);
			if (args.Cause != ActionCause.OnlyCalculate)
			{
				NotifyActivating();
			}
		}
	}

	private void OnDamageReceiving(DamageEventArgs args)
	{
		if (args.DamageInfo.DamageType == DamageType.Attack)
		{
			args.DamageInfo = args.DamageInfo.MultiplyBy(Ratio);
			args.AddModifier(this);
			if (args.Cause != ActionCause.OnlyCalculate)
			{
				NotifyActivating();
			}
		}
	}
}
