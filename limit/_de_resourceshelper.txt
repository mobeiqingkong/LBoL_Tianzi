using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Diagnostics.CodeAnalysis;
using System.Linq;
using Cysharp.Threading.Tasks;
using LBoL.Base.Extensions;
using LBoL.Core;
using LBoL.Core.StatusEffects;
using LBoL.Core.Units;
using LBoL.Presentation.Effect;
using Spine.Unity;
using TMPro;
using UnityEngine;
using UnityEngine.AddressableAssets;
using UnityEngine.ResourceManagement.AsyncOperations;

namespace LBoL.Presentation;

public static class ResourcesHelper
{
	private static readonly Dictionary<Type, Dictionary<string, Sprite>> Sprites = new Dictionary<Type, Dictionary<string, Sprite>>();

	private static readonly Dictionary<string, Texture> CardImages = new Dictionary<string, Texture>();

	private static readonly Dictionary<string, Sprite> BossIcons = new Dictionary<string, Sprite>();

	private static readonly Dictionary<string, Sprite> IntentionSprites = new Dictionary<string, Sprite>();

	private static readonly Dictionary<string, Sprite> AchievementSprites = new Dictionary<string, Sprite>();

	private static readonly Dictionary<string, Sprite> StickerSprites = new Dictionary<string, Sprite>();

	private static bool _preLoaded;

	private static bool _loaded;

	private static bool _postLoaded;

	public static async UniTask PreInitializeAsync()
	{
		if (!_preLoaded)
		{
			await Addressables.InitializeAsync();
			await LoadUiSprites<UltimateSkill>();
			_preLoaded = true;
		}
	}

	public static async UniTask InitializeAsync()
	{
		await PreInitializeAsync();
		if (!_loaded)
		{
			await LoadAllByLabel("UI/BossIcons", BossIcons);
			await LoadAllByLabel("UI/Intention", IntentionSprites);
			await LoadAllByLabel("UI/Achievement", AchievementSprites);
			await LoadAllByLabel("UI/Stickers", StickerSprites);
			_loaded = true;
		}
	}

	public static async UniTask PostInitializeAsync()
	{
		await InitializeAsync();
		if (!_postLoaded)
		{
			await LoadUiSprites<Exhibit>();
			await LoadUiSprites<StatusEffect>();
			await LoadAllByLabel("CardImages", CardImages);
			_postLoaded = true;
		}
	}

	internal static async UniTask<IList<T>> LoadAllByLabel<T>(string label, IDictionary<string, T> dict) where T : UnityEngine.Object
	{
		try
		{
			IList<T> list = await Addressables.LoadAssetsAsync<T>(label, delegate
			{
			}).ToUniTask();
			foreach (T item in list.Distinct())
			{
				if (dict.ContainsKey(item.name))
				{
					UnityEngine.Debug.LogWarning("Loading folder " + label + ": duplicated asset name " + item.name);
				}
				else
				{
					dict.Add(item.name, item);
				}
			}
			return list;
		}
		catch (Exception arg)
		{
			UnityEngine.Debug.LogError($"Error loading folder {label}: {arg}");
			return null;
		}
	}

	private static async UniTask LoadUiSprites<T>()
	{
		Dictionary<string, Sprite> dict = new Dictionary<string, Sprite>();
		try
		{
			await LoadAllByLabel("UI/" + typeof(T).Name + "s", dict);
		}
		finally
		{
			Sprites.Add(typeof(T), dict);
		}
	}

	public static Texture TryGetCardImage(string id)
	{
		if (!CardImages.TryGetValue(id, out var value))
		{
			return null;
		}
		return value;
	}

	public static Sprite TryGetSprite<T>(string id)
	{
		if (!Sprites.TryGetValue(typeof(T), out var value))
		{
			return null;
		}
		if (!value.TryGetValue(id, out var value2))
		{
			return null;
		}
		return value2;
	}

	public static Sprite TryGetBossIcon(string id)
	{
		if (!BossIcons.TryGetValue(id, out var value))
		{
			return null;
		}
		return value;
	}

	public static Sprite TryGetAchievementSprite(string id)
	{
		if (!AchievementSprites.TryGetValue(id, out var value))
		{
			return null;
		}
		return value;
	}

	public static Sprite TryGetSticker(string id, bool on = true)
	{
		if (!on)
		{
			id += "Off";
		}
		if (!StickerSprites.TryGetValue(id, out var value))
		{
			return null;
		}
		return value;
	}

	public static Sprite TryGetIntention(string id, string suffix = null)
	{
		if (id.EndsWith("Intention"))
		{
			id = id.RemoveEnd("Intention");
		}
		if (!IntentionSprites.TryGetValue(id + suffix, out var value))
		{
			return null;
		}
		return value;
	}

	public static void Release(AsyncOperationHandle handle)
	{
		Addressables.Release(handle);
	}

	private static T SyncLoadAddressable<T>(object key) where T : UnityEngine.Object
	{
		Stopwatch stopwatch = Stopwatch.StartNew();
		AsyncOperationHandle<T> asyncOperationHandle = Addressables.LoadAssetAsync<T>(key);
		asyncOperationHandle.WaitForCompletion();
		stopwatch.Stop();
		if (asyncOperationHandle.Status == AsyncOperationStatus.Succeeded)
		{
			return asyncOperationHandle.Result;
		}
		throw new ArgumentException($"Cannot load addressable {key} of type {typeof(T)}");
	}

	private static async UniTask<T> LoadAddressableAsyncWithTiming<T>(object key) where T : UnityEngine.Object
	{
		AsyncOperationHandle<T> asyncOperationHandle = Addressables.LoadAssetAsync<T>(key);
		try
		{
			return await asyncOperationHandle.Task;
		}
		catch (Exception)
		{
			throw;
		}
	}

	public static Sprite LoadCharacterAvatarSprite(string characterName)
	{
		return SyncLoadAddressable<Sprite>("Assets/AA/StandPicture/" + characterName + ".png[" + characterName + "_Avatar]");
	}

	public static async UniTask<Sprite> LoadSpellPortraitAsync(string characterName)
	{
		return await LoadAddressableAsyncWithTiming<Sprite>("Assets/AA/Units/SpellPortraits/" + characterName + ".png");
	}

	public static async UniTask<Sprite> LoadSimpleUnitSpriteAsync(string characterName)
	{
		return await LoadAddressableAsyncWithTiming<Sprite>("Assets/AA/Units/Sprite/" + characterName + ".png");
	}

	public static async UniTask<SkeletonDataAsset> LoadSpineUnitAsync(string characterName)
	{
		string text = characterName.ToLowerInvariant();
		return await LoadAddressableAsyncWithTiming<SkeletonDataAsset>("Assets/AA/Units/Spine/" + text + "/" + text + "_SkeletonData.asset");
	}

	public static async UniTask<Sprite> LoadSimpleDollSpriteAsync(string dollName)
	{
		return await LoadAddressableAsyncWithTiming<Sprite>("Assets/AA/Doll/Sprite/" + dollName + ".png");
	}

	public static async UniTask<AudioClip> LoadBgmAsync(string path)
	{
		return await LoadAddressableAsyncWithTiming<AudioClip>("Assets/AA/Audio/Bgm/" + path);
	}

	public static async UniTask<AudioClip> LoadSfxAsync(string path)
	{
		return await LoadAddressableAsyncWithTiming<AudioClip>("Assets/AA/Audio/Sfx/" + path);
	}

	public static AudioClip LoadSfx(string path)
	{
		return SyncLoadAddressable<AudioClip>("Assets/AA/Audio/Sfx/" + path);
	}

	public static async UniTask<AudioClip> LoadUiSoundAsync(string path)
	{
		return await LoadAddressableAsyncWithTiming<AudioClip>("Assets/AA/Audio/UiSound/" + path);
	}

	public static AudioClip LoadUiSound(string path)
	{
		return SyncLoadAddressable<AudioClip>("Assets/AA/Audio/UiSound/" + path);
	}

	public static Texture2D LoadAdventureImage(string name)
	{
		return SyncLoadAddressable<Texture2D>("Assets/AA/AdventureImage/" + name + ".png");
	}

	public static Sprite LoadUiBackground(string bgName)
	{
		return SyncLoadAddressable<Sprite>("Assets/AA/UI/Background/" + bgName + ".png");
	}

	public static async UniTask<GameObject> LoadUiPanelAsync([MaybeNull] string folder, string name)
	{
		string path = ((folder == null) ? name : (folder + "/" + name));
		GameObject obj = (GameObject)(await Resources.LoadAsync<GameObject>("UI/Panels/" + path));
		if (!obj)
		{
			throw new ArgumentException("Cannot load UiPanel at '" + path + "'");
		}
		return obj;
	}

	public static async UniTask<GameObject> LoadUiDialogAsync([MaybeNull] string folder, string name)
	{
		string path = ((folder == null) ? name : (folder + "/" + name));
		GameObject obj = (GameObject)(await Resources.LoadAsync<GameObject>("UI/Dialogs/" + path));
		if (!obj)
		{
			throw new ArgumentException("Cannot load UiDialog at '" + path + "'");
		}
		return obj;
	}

	public static async UniTask<EffectWidget> LoadEffectAsync(string path)
	{
		return (await LoadAddressableAsyncWithTiming<GameObject>("Assets/AA/Effect/" + path + ".prefab")).GetComponent<EffectWidget>();
	}

	public static EffectWidget LoadEffect(string path)
	{
		return SyncLoadAddressable<GameObject>("Assets/AA/Effect/" + path + ".prefab").GetComponent<EffectWidget>();
	}

	public static async UniTask<TMP_FontAsset> LoadFontAsync(string path)
	{
		TMP_FontAsset tMP_FontAsset = Resources.Load<TMP_FontAsset>("Fonts/" + path);
		if (!tMP_FontAsset)
		{
			throw new ArgumentException("Cannot load font at " + path);
		}
		return await UniTask.FromResult(tMP_FontAsset);
	}

	public static void Release(UnityEngine.Object @object)
	{
		Addressables.Release(@object);
	}
}
