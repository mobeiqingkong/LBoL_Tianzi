using System;
using System.Collections.Generic;
using JetBrains.Annotations;
using LBoL.Core.Battle;
using UnityEngine;
using UnityEngine.Scripting;

namespace LBoL.Core;

[RequireDerived]
public abstract class GameEntity : IInitializable, INotifyChanged
{
	public string Id { get; private set; }

	public virtual string DebugName => "<" + Id + ">(" + Name + ")";

	protected string BaseName => LocalizeProperty("Name");

	protected string BaseDescription => LocalizeProperty("Description", decorated: true);

	public virtual string Name => BaseName;

	[UsedImplicitly]
	public virtual EntityName SelfName => new EntityName("Entity." + Id, Name).WithColor();

	[UsedImplicitly]
	public virtual UnitName PlayerName => GameRun?.Player.GetName() ?? UnitNameTable.GetDefaultPlayerName();

	public virtual string Description
	{
		get
		{
			try
			{
				string baseDescription = GetBaseDescription();
				if (baseDescription == null)
				{
					Debug.LogError("<" + DebugName + ">.GetBaseDescription() returns null");
					return "";
				}
				return baseDescription.RuntimeFormat(FormatWrapper);
			}
			catch (Exception exception)
			{
				Debug.LogException(exception);
				return "<Error>";
			}
		}
	}

	internal abstract GameEventPriority DefaultEventPriority { get; }

	internal GameEntityFormatWrapper FormatWrapper { get; private set; }

	public GameRunController GameRun { get; internal set; }

	public event Action PropertyChanged;

	protected abstract string LocalizeProperty(string key, bool decorated = false, bool required = true);

	protected virtual string GetBaseDescription()
	{
		return BaseDescription;
	}

	protected virtual void React(Reactor reactor)
	{
		throw new NotSupportedException("Cannot react from " + GetType().Name);
	}

	protected void React(IEnumerable<BattleAction> sequencedReactor)
	{
		React(new Reactor(sequencedReactor));
	}

	protected void React(LazyActionReactor lazyActionReactor)
	{
		React(new Reactor(lazyActionReactor));
	}

	protected void React(LazySequencedReactor lazySequencedReactor)
	{
		React(new Reactor(lazySequencedReactor));
	}

	public virtual void Initialize()
	{
		Id = GetType().Name;
		FormatWrapper = CreateFormatWrapper();
	}

	internal virtual GameEntityFormatWrapper CreateFormatWrapper()
	{
		return new GameEntityFormatWrapper(this);
	}

	public virtual void NotifyChanged()
	{
		this.PropertyChanged?.Invoke();
	}
}
