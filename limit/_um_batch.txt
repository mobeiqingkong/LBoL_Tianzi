using System;
using System.Collections.Generic;
using System.Linq;
using System.Reflection;
using HarmonyLib;
using LBoL.Base.Extensions;
using LBoLEntitySideloader.Attributes;
using LBoLEntitySideloader.Entities;

namespace LBoLEntitySideloader.ReflectionHelpers;

internal class TemplatesReflection
{
	private static HashSet<Type> templateTypes = new HashSet<Type>();

	private static List<Type> templatesExpectingEntityLogic = new List<Type>();

	public static HashSet<Type> AllTemplateTypes(bool refresh = false)
	{
		if (CollectionsExtensions.Empty<Type>((IReadOnlyCollection<Type>)templateTypes) || refresh)
		{
			CollectionExtensions.Do<Type>(from t in typeof(EntityDefinition).Assembly.GetExportedTypes()
				where t.BaseType == typeof(EntityDefinition)
				select t, (Action<Type>)delegate(Type t)
			{
				templateTypes.Add(t);
			});
		}
		return templateTypes;
	}

	public static bool ExpectsEntityLogic(Type template, bool refresh = false)
	{
		if (CollectionsExtensions.Empty<Type>((IReadOnlyCollection<Type>)templatesExpectingEntityLogic) || refresh)
		{
			Type interfaceType = typeof(ITypeProvider<>).GetGenericTypeDefinition();
			CollectionExtensions.Do<Type>(from t in AllTemplateTypes()
				where t.GetInterfaces().Any((Type i) => TypeExtensions.IsSubclassOfGeneric(i, interfaceType))
				select t, (Action<Type>)delegate(Type t)
			{
				templatesExpectingEntityLogic.Add(t);
			});
		}
		return templatesExpectingEntityLogic.Any((Type t) => template.IsSubclassOf(t) || t == template);
	}

	public static Type Template2FacType(Type template, bool refresh = false)
	{
		if (!ExpectsEntityLogic(template))
		{
			return null;
		}
		Type interfaceType = typeof(ITypeProvider<>).GetGenericTypeDefinition();
		return template.GetInterfaces().First((Type i) => TypeExtensions.IsSubclassOfGeneric(i, interfaceType)).GetGenericArguments()
			.First();
	}

	public static bool IsTemplateType(Type type)
	{
		return AllTemplateTypes().Any((Type t) => type.IsSubclassOf(t));
	}

	public static bool DoOverwrite(Type definitionType, string methodName)
	{
		MethodInfo element = AccessTools.Method(definitionType, methodName, (Type[])null, (Type[])null);
		return element.GetCustomAttribute<DontOverwriteAttribute>(inherit: true) == null;
	}
}
