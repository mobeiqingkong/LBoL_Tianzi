using System.Collections.Generic;
using System.IO;

namespace LBoLEntitySideloader.Entities;

public abstract class UnitModelTemplate : EntityDefinition, IConfigProvider<UnitModelConfig>, IResourceConsumer<LocalizationOption>
{
	[HarmonyPatch(typeof(ResourcesHelper), "LoadSpineUnitAsync")]
	private class LoadSpineUnitAsync_Patch
	{
		private static bool Prefix(string characterName, ref UniTask<SkeletonDataAsset> __result)
		{
			if (UniqueTracker.Instance.IsLoadedOnDemand(typeof(UnitModelTemplate), characterName, out var entityDefinition))
			{
				if (entityDefinition is UnitModelTemplate unitModelTemplate && EntityManager.HandleOverwriteWrap(delegate
				{
				}, unitModelTemplate, "LoadModelOptions", unitModelTemplate.user) && unitModelTemplate.CheckModelOptions())
				{
					__result = unitModelTemplate.LoadModelOptions().loadSpine;
					return false;
				}
				return true;
			}
			return true;
		}
	}

	[HarmonyPatch(typeof(ResourcesHelper), "LoadSimpleUnitSpriteAsync")]
	private class LoadSimpleUnitSpriteAsync_Patch
	{
		private static bool Prefix(string characterName, ref UniTask<Sprite> __result)
		{
			if (UniqueTracker.Instance.IsLoadedOnDemand(typeof(UnitModelTemplate), characterName, out var entityDefinition))
			{
				if (entityDefinition is UnitModelTemplate unitModelTemplate && EntityManager.HandleOverwriteWrap(delegate
				{
				}, unitModelTemplate, "LoadModelOptions", unitModelTemplate.user) && unitModelTemplate.CheckModelOptions())
				{
					__result = unitModelTemplate.LoadModelOptions().loadSprite;
					return false;
				}
				return true;
			}
			return true;
		}
	}

	[HarmonyPatch(typeof(ResourcesHelper), "LoadSpellPortraitAsync")]
	private class LoadSpellPortraitAsync_Patch
	{
		private static bool Prefix(string characterName, ref UniTask<Sprite> __result)
		{
			if (UniqueTracker.Instance.IsLoadedOnDemand(typeof(UnitModelTemplate), characterName, out var entityDefinition))
			{
				if (entityDefinition is UnitModelTemplate unitModelTemplate && EntityManager.HandleOverwriteWrap(delegate
				{
				}, unitModelTemplate, "LoadSpellSprite", unitModelTemplate.user) && unitModelTemplate.CheckSpellSpriteEnabled())
				{
					__result = unitModelTemplate.LoadSpellSprite();
					return false;
				}
				return true;
			}
			return true;
		}
	}

	public override Type ConfigType()
	{
		return typeof(UnitModelConfig);
	}

	public override Type EntityType()
	{
		throw new InvalidDataException();
	}

	public override Type TemplateType()
	{
		return typeof(UnitModelTemplate);
	}

	/// <summary>
	/// Name : Id should be the same as EnemyUnit's;
	/// Type : model type: 0 - Sprite; 1 - Spine model; 3 - effect (like spirits or Kokoro masks);
	/// EffectName : effect Id if Type=2 is used;
	/// Offset : ;
	/// Flip : player models seem to get auto flipped?;
	/// Dielevel : 0, 1 or 2 0 - "UnitDeathSmall", 1 - "UnitDeath", 2 - "UnitDeathLarge";
	/// Box : collision box size. Player sized units it's (0.80f, 1.80f);
	/// Shield : barrier? effect radius;
	/// Block : block? effect radius;
	/// Hp : hp bar position;
	/// HpLength : hp bar length;
	/// Info : ;
	/// Select : ;
	/// ShootStartTime : ;
	/// ShootPoint : ;
	/// ShooterPoint : actual shoot position is the sum of these to point?;
	/// Hit : ;
	/// HitRep : ;
	/// GuardRep : ;
	/// Chat : ;
	/// ChatPortraitXY : ;
	/// ChatPortraitWH : ;
	/// HasSpellPortrait : should load extra sprite for bomb cast;
	/// SpellPosition : scale of this vector is in 100s;
	/// SpellScale : ;
	/// SpellColor : 4 color values or empty. If the list is empty UnitView.GetDefaultSpellColors will be used;
	///
	/// this config is for player sized models 2do
	/// </summary>
	/// <returns></returns>
	public UnitModelConfig DefaultConfig()
	{
		return new UnitModelConfig("", 0, "", new Vector2(0f, 0f), Flip: false, 0, new Vector2(0.8f, 1.8f), 1.2f, 1.3f, new Vector2(0f, -1.3f), 480f, new Vector2(0f, 1.2f), new Vector2(1.6f, 2f), (IReadOnlyList<float>)new List<float> { 0.1f }, (IReadOnlyList<Vector2>)new List<Vector2>
		{
			new Vector2(0.6f, 0.3f)
		}, (IReadOnlyList<Vector2>)new List<Vector2>
		{
			new Vector2(0.6f, 0.3f)
		}, new Vector2(0.3f, 0f), 0.1f, 0.1f, new Vector2(0.4f, 0.8f), new Vector2(-0.8f, -0.58f), new Vector2(0.6f, 0.5f), HasSpellPortrait: false, new Vector2(0f, 0f), 1f, (IReadOnlyList<Color32>)new List<Color32>
		{
			new Color32(101, 229, 71, byte.MaxValue),
			new Color32(131, 221, 117, byte.MaxValue),
			new Color32(137, 221, 117, 150),
			new Color32(153, 220, 127, byte.MaxValue)
		});
	}

	public abstract UnitModelConfig MakeConfig();

	public abstract ModelOption LoadModelOptions();

	public abstract UniTask<Sprite> LoadSpellSprite();

	/// <summary>
	/// Default:, Short: and Long: . Default is mandatory. 
	/// For name display above model and naming playable character 
	/// </summary>
	/// <returns></returns>
	public abstract LocalizationOption LoadLocalization();

	public void Consume(LocalizationOption locOption)
	{
		if (locOption == null)
		{
			return;
		}
		if (locOption is GlobalLocalization globalLocalization)
		{
			if (globalLocalization.LocalizationFiles.locTable.NotEmpty() && !UniqueTracker.Instance.unitNamesGlobalLocalizationFiles.TryAdd(userAssembly, globalLocalization.LocalizationFiles))
			{
				ManualLogSource obj = Log.LogDev();
				if (obj != null)
				{
					obj.LogWarning((object)$"{userAssembly.GetName().Name}: {GetType()} tries to set global unit name localization files but they've already been set by another {TemplateType().Name}.");
				}
			}
			UniqueTracker.Instance.unitIdsToLocalize.TryAdd(userAssembly, new HashSet<IdContainer>());
			UniqueTracker.Instance.unitIdsToLocalize[userAssembly].Add(GetId());
		}
		else if (locOption is LocalizationFiles localizationFiles)
		{
			YamlMappingNode yamlMappingNode = localizationFiles.Load(Localization.CurrentLocale);
			YamlScalarNode key = new YamlScalarNode(GetId());
			if (yamlMappingNode.Children.TryGetValue(key, out YamlNode value) && value is YamlMappingNode value2)
			{
				LocalizationOption.FillUnitNameTable(new YamlMappingNode(new KeyValuePair<YamlNode, YamlNode>(key, value2)), localizationFiles.mergeTerms);
			}
			else
			{
				localizationFiles.fileNames.TryGetValue(localizationFiles.GetAvailableLocale(), out var value3);
				Log.log.LogWarning((object)$"{GetId()} not found in {value3}");
			}
		}
		else if (locOption is DirectLocalization directLocalization)
		{
			Dictionary<string, Dictionary<string, object>> termDic = directLocalization.WrapTermDic(base.UniqueId);
			LocalizationOption.FillUnitNameTable(LocalizationOption.TermDic2YamlMapping(termDic), directLocalization.mergeTerms);
		}
		else if (locOption is BatchLocalization batchLocalization)
		{
			batchLocalization.RegisterSelf(userAssembly);
		}
	}

	private bool CheckModelOptions()
	{
		ModelOption modelOption = LoadModelOptions();
		if (modelOption == null)
		{
			Log.LogDev().LogWarning((object)("ModelOption not provided by " + GetType().Name + "."));
			return false;
		}
		UnitView.ModelType modelType = UnitView.ModelType.SingleSprite;
		try
		{
			modelType = (UnitView.ModelType)UnitModelConfig.FromName(base.UniqueId).Type;
		}
		catch (Exception arg)
		{
			Log.log.LogError((object)$"UnitModelConfig.Type should be number from 0 to 2 : {arg}");
			return false;
		}
		if (modelType != modelOption.modelType)
		{
			Log.log.LogError((object)(GetType().Name + ": UnitModelConfig.Type mismatches options supplied by " + typeof(ModelOption).Name));
			return false;
		}
		return true;
	}

	private bool CheckSpellSpriteEnabled()
	{
		if (!UnitModelConfig.FromName(base.UniqueId).HasSpellPortrait)
		{
			Log.log.LogWarning((object)(GetType().Name + ": UnitModelConfig.HasSpellPortrait is false but HasSpellPortrait is being loaded anyway"));
			return false;
		}
		return true;
	}
}
