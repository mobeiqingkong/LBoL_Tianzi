using System;
using JetBrains.Annotations;

namespace LBoL.Core.Battle;

public abstract class Interaction
{
	private readonly WeakReference<GameEntity> _source = new WeakReference<GameEntity>(null);

	public GameEntity Source
	{
		get
		{
			if (!_source.TryGetTarget(out var target))
			{
				return null;
			}
			return target;
		}
		set
		{
			_source.SetTarget(value);
		}
	}

	[CanBeNull]
	public string Description { get; set; }

	public bool CanCancel { get; set; }

	public bool IsCanceled { get; private set; }

	public void Cancel()
	{
		if (!CanCancel)
		{
			throw new InvalidOperationException($"Cannot cancel {this}");
		}
		IsCanceled = true;
	}
}
