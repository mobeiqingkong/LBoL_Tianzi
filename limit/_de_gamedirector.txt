				Unit unit2 = e.Unit;

				Unit unit = e.Unit;

			if (!unitView)
			{

			}
			return unitView;
		}
	}

	private static void SetTimeStep()
	{
		TimeStep = 1f / 60f;
		if ((double)TimeStep <= 0.001)
		{
			Debug.LogError("TimeSteptoo fast!");
		}
	}

	private void Update()
	{
		float time = Time.time;
		float num = time - CurrentTime;
		if (num > 0.25f)
		{
			num = 0.25f;
		}
		CurrentTime = time;
		Accumulator += num;
		while (Accumulator >= TimeStep)
		{
			MasterTick();
			Accumulator -= TimeStep;
		}
	}

	private void MasterTick()
	{
		if ((bool)PlayerUnitView)
		{
			PlayerUnitView.Tick();
			foreach (DollView doll in PlayerUnitView.Dolls)
			{
				doll.Tick();
			}
		}
		foreach (UnitView enemyUnitView in EnemyUnitViews)
		{
			enemyUnitView.Tick();
		}
		Singleton<GunManager>.Instance.Tick();
		Singleton<EffectManager>.Instance.Tick();
	}

	private void Awake()
	{
		Formations = new Dictionary<string, EnemyFormation>();
		foreach (Transform item in unitRoot)
		{
			EnemyFormation component = item.GetComponent<EnemyFormation>();
			if ((bool)component)
			{
				Formations.Add(component.name, component);
			}
		}
		SetTimeStep();
		_cameraOffset = CameraController.MainCamera.transform.localPosition;
	}

	private async UniTask<UnitView> InternalLoadPlayerAsync(PlayerUnit player, bool hidden)
	{
		GameObject gameObject = UnityEngine.Object.Instantiate(unitPrefab, playerRoot);
		UnitView playerView = gameObject.GetComponent<UnitView>();
		playerView.Unit = player;
		playerView.SetStatusWidget(UiManager.GetPanel<UnitStatusHud>().CreateStatusWidget(player), 1f);
		playerView.SetInfoWidget(UiManager.GetPanel<UnitStatusHud>().CreateInfoWidget(player), 1f);
		playerView.IsHidden = hidden;
		await playerView.LoadUnitModelAsync(player.ModelName, player: true);
		player.SetView(playerView);
		playerView.SetPlayerHpBarLength(player.MaxHp);
		PlayerUnitView = playerView;
		return playerView;
	}

	public static UniTask<UnitView> LoadEnemyAsync(EnemyUnit enemy, string formationName = null, bool hidden = false, int? forceRootIndex = null)
	{
		if (formationName == null)
		{
			formationName = "Triangle";
		}
		if (Singleton<GameDirector>.Instance.Formations.TryGetValue(formationName, out var value))
		{
			ActiveFormation = formationName;
			return Singleton<GameDirector>.Instance.InternalLoadEnemyAsync(enemy, value, hidden, forceRootIndex);
		}
		ActiveFormation = "Triangle";
		Debug.LogWarning("Formation name '" + formationName + "' not found, using default.");
		return Singleton<GameDirector>.Instance.InternalLoadEnemyAsync(enemy, Singleton<GameDirector>.Instance.Formations["Triangle"], hidden, forceRootIndex);
	}

	private async UniTask<UnitView> InternalLoadEnemyAsync(EnemyUnit enemy, EnemyFormation formation, bool hidden, int? forceRootIndex)
	{
		int rootIndex = forceRootIndex ?? enemy.RootIndex;
		if (rootIndex < 0 || rootIndex >= formation.enemyLocations.Count)
		{
			throw new ArgumentException($"Invalid root-index: {rootIndex} (formation size: {formation.enemyLocations.Count})", "rootIndex");
		}
		GameObject gameObject = UnityEngine.Object.Instantiate(unitPrefab, formation.enemyLocations.Values.ToList()[rootIndex]);
		UnitView enemyView = gameObject.GetComponent<UnitView>();
		enemyView.Unit = enemy;
		enemyView.SetStatusWidget(UiManager.GetPanel<UnitStatusHud>().CreateStatusWidget(enemy), 0f);
		enemyView.SetInfoWidget(UiManager.GetPanel<UnitStatusHud>().CreateInfoWidget(enemy), (!hidden) ? 1 : 0);
		enemyView.IsHidden = hidden;
		float? hpLength = EnemyUnitConfig.FromId(enemy.Id).HpLength;
		await enemyView.LoadUnitModelAsync(enemy.ModelName, player: false, hpLength);
		enemy.SetView(enemyView);
		EnemyUnitViews.Add(enemyView);
		enemyView.RootIndex = rootIndex;
		enemyView.AngleIndex = formation.enemyLocations.Keys.ToList()[rootIndex];
		return enemyView;
	}

	private static void UpdateEnemyMoveOrder()
	{
		if (!_enemyMoveOrderVisible)
		{
			return;
		}
		foreach (var item in (from enemyView in Enemies
			let enemy = (EnemyUnit)enemyView.Unit
			where enemy.IsAlive
			orderby enemy.MovePriority, enemy.RootIndex
			select enemyView).ToList().WithIndices())
		{
			var (num, _) = item;
			item.elem.SetMoveOrder(num + 1);
		}
	}

	public static void SetEnemyMoveOrderVisible(bool visible)
	{
		_enemyMoveOrderVisible = visible;
		if (visible)
		{
			UpdateEnemyMoveOrder();
			{
				foreach (UnitView enemy in Enemies)
				{
					if (enemy.Unit.IsAlive)
					{
						enemy.SetMoveOrderVisible(visible: true);
					}
				}
				return;
			}
		}
		foreach (UnitView enemy2 in Enemies)
		{
			enemy2.SetMoveOrderVisible(visible: false);
		}
	}

	public static IEnumerable<UnitView> EnumeratePotentialTargets(TargetType targetType)
	{
		if (targetType == TargetType.All || targetType == TargetType.Self)
		{
			yield return Player;
		}
		if (targetType != TargetType.All && targetType != TargetType.SingleEnemy && targetType != TargetType.RandomEnemy && targetType != TargetType.AllEnemies)
		{
			yield break;
		}
		foreach (UnitView enemy in Enemies)
		{
			if (enemy.Unit.IsAlive)
			{
				yield return enemy;
			}
		}
	}

	private static IEnumerator DebutFlow(bool hasKaguya, string playerName)
	{
		UnitView eirin = Enemies[0];
		yield return new WaitForSeconds(2f);
		if (hasKaguya)
		{
			UnitView kaguya = Enemies[1];
			List<string> kaguyaChats = "Debut.KaguyaChats".LocalizeStrings().ToList();
			Transform erinTrans = Singleton<GameDirector>.Instance.lore0;
			Transform kaguyaTrans = Singleton<GameDirector>.Instance.lore1;
			erinTrans.DOKill(complete: true);
			erinTrans.DOScaleX(-1f, 0.3f);
			kaguya.Chat(kaguyaChats[0], 3f, ChatWidget.CloudType.RightTalk);
			yield return new WaitForSecondsRealtime(3.5f);
			kaguya.Chat(kaguyaChats[1], 3f, ChatWidget.CloudType.RightTalk);
			yield return new WaitForSecondsRealtime(3.1f);
			kaguyaTrans.DOKill(complete: true);
			DOTween.Sequence().Append(kaguyaTrans.DOScaleX(-1f, 0.2f)).Append(kaguyaTrans.DOLocalMoveX(5f, 2f).SetRelative(isRelative: true).SetEase(Ease.InSine))
				.SetUpdate(isIndependentUpdate: true)
				.SetAutoKill(autoKillOnCompletion: true);
			yield return new WaitForSecondsRealtime(0.4f);
			eirin.Chat(kaguyaChats[2], 3f, ChatWidget.CloudType.LeftTalk);
			yield return new WaitForSecondsRealtime(4f);
			erinTrans.DOScaleX(1f, 0.3f);
			yield return new WaitForSecondsRealtime(0.5f);
		}
		List<string> eirinChats = "Debut.EirinChats".LocalizeStrings().ToList();
		eirinChats[0] = string.Format(eirinChats[0], playerName);
		while (true)
		{
			foreach (string item in eirinChats)
			{
				eirin.Chat(item, 3f, ChatWidget.CloudType.RightTalk);
				yield return new WaitForSecondsRealtime(7f);
			}
		}
	}

	private static IEnumerator DamageActionViewer(DamageAction action)
	{
		DamageEventArgs[] damageArgs = action.DamageArgs;
		Unit source = damageArgs.First().Source;
		UnitView unit = GetUnit(source);
		if (source != null && unit == null)
		{
			Debug.LogWarning("Damaging source " + source.DebugName + " not found in GameDirector");
			return null;
		}
		List<(UnitView, DamageInfo)> list = new List<(UnitView, DamageInfo)>();
		DamageEventArgs[] array = damageArgs;
		foreach (DamageEventArgs damageEventArgs in array)
		{
			UnitView unit2 = GetUnit(damageEventArgs.Target);
			if (damageEventArgs.Target == null)
			{
				Debug.LogWarning("Damaging target is null");
			}
			else if (unit2 == null)
			{
				Debug.LogWarning("Damaging target " + damageEventArgs.Target.DebugName + " not found in GameDirector");
			}
			else
			{
				list.Add((unit2, damageEventArgs.DamageInfo));
			}
		}
		ActionCause cause = action.Cause;
		if (cause == ActionCause.Card || cause == ActionCause.Us || cause == ActionCause.EnemyAction || cause == ActionCause.StatusEffect || cause == ActionCause.Exhibit || cause == ActionCause.JadeBox || cause == ActionCause.Unit)
		{
			string text = action.GunName;
			if (action.GunName == null)
			{
				Debug.LogWarning("Action [" + (action.Source?.DebugName ?? "<null>") + "]: GunName is null");
				text = "Empty";
			}
			else if (text.IsNullOrEmpty() || text == "Empty")
			{
				Debug.LogWarning("Action [" + (action.Source?.DebugName ?? "<null>") + "]: GunName is empty");
			}
			return GunShootAction(unit, list, text, action.GunType);
		}
		if (action.Source is Doll doll)
		{
			DollView doll2 = unit.GetDoll(doll);
			if ((bool)doll2)
			{
				string text2 = action.GunName;
				if (action.GunName == null)
				{
					Debug.LogWarning("Action [" + (action.Source?.DebugName ?? "<null>") + "]: GunName is null");
					text2 = "Empty";
				}
				else if (text2.IsNullOrEmpty() || text2 == "Empty")
				{
					Debug.LogWarning("Action [" + (action.Source?.DebugName ?? "<null>") + "]: GunName is empty");
				}
				return GunShootAction(doll2, list, text2, action.GunType);
			}
			return null;
		}
		foreach (var (unitView, damage) in list)
		{
			if (unitView.Unit is PlayerUnit)
			{
				UiManager.GetPanel<SystemBoard>().OnHpChanged();
			}
			unitView.OnDamageReceived(damage);
		}
		return null;
	}

	private static IEnumerator ExplodeActionViewer(ExplodeAction action)
	{
		DamageEventArgs[] damageArgs = action.DamageArgs;
		Unit source2 = damageArgs.First().Source;

	private static IEnumerator GunShootAction(UnitView source, IList<(UnitView target, DamageInfo damageInfo)> pairs, string gunName, GunType type)
	{
		source.Targets = pairs.Select(((UnitView target, DamageInfo damageInfo) p) => p.target).ToList();
		int count = source.Targets.Count;
		if (count <= 2)
		{
			source.Target = source.Targets.First();
		}
		else
		{
			int num = (count - 1) / 2;
			List<UnitView> list = source.Targets.ToList();
			List<UnitView> list2 = new List<UnitView>();
			for (int i = 0; i <= num; i++)
			{
				UnitView smallOne = list.First();
				foreach (UnitView item3 in list.Where((UnitView u) => u.AngleIndex < smallOne.AngleIndex))
				{
					smallOne = item3;
				}
				list2.Add(smallOne);
				list.Remove(smallOne);
			}
			source.Target = list2[num];
		}
		foreach (var pair in pairs)
		{
			UnitView item = pair.target;
			DamageInfo item2 = pair.damageInfo;
			item.ComingDamage = item2;
		}
		_gunHitArgs = new GunHitArgs(source == Player, pairs, gunName);
		yield return source.Shoot(gunName, type);
		yield return new WaitWhile(() => _someoneCrashing);
	}

	private static void GunHitPresentation(GunHitArgs gunHitArgs)
	{
		float num = 0f;
		float shakePower = 1f;
		GunConfig gunConfig = GunConfig.FromName(gunHitArgs.GunName);
		if (gunConfig != null)
		{
			shakePower = gunConfig.ShakePower;
		}
		foreach (var (unitView, damage) in gunHitArgs.Pairs)
		{
			if (num < damage.Amount)
			{
				num = damage.Amount;
			}
			unitView.HitEnd();
			if (unitView.Unit is PlayerUnit)
			{
				UiManager.GetPanel<SystemBoard>().OnHpChanged();
			}
			unitView.OnDamageReceived(damage);
			PopupHud.Instance.DamagePopupFromScene(damage, unitView.transform.position, gunHitArgs.SourceIsPlayer);
		}
		if (GameMaster.Shake)
		{
			ShakeWithDamage(num, shakePower);
		}
		if (gunHitArgs.Pairs.Any(((UnitView target, DamageInfo damageInfo) pair) => pair.target.WillCrash))
		{
			Singleton<GameDirector>.Instance.StartCoroutine(CrashingTimer());
		}
	}

	private static void ShakeWithDamage(float damage, float shakePower)
	{
		ShakeWithDamage((damage * shakePower).ToInt());
	}

	private static void ShakeWithDamage(int damage)
	{
		ShakingDamage += damage;
		if (ShakingDamage >= ShakeFloors[4])
		{
			Shake(4);
		}
		else if (ShakingDamage >= ShakeFloors[3])
		{
			Shake(3);
		}
		else if (ShakingDamage >= ShakeFloors[2])
		{
			Shake(2);
		}
		else if (ShakingDamage >= ShakeFloors[1])
		{
			Shake(1);
		}
		else if (ShakingDamage >= ShakeFloors[0])
		{
			Shake();
		}
	}

	public static void Shake(int level = 0, bool withDamage = true)
	{
		if (level < 4)
		{
			switch (level)
			{
			case 0:
				ShakeHandler(0.3f, 1f, withDamage);
				break;
			case 1:
				ShakeHandler(0.3f, 1.5f, withDamage);
				break;
			case 2:
				ShakeHandler(0.4f, 2f, withDamage);
				break;
			case 3:
				ShakeHandler(0.4f, 3f, withDamage);
				break;
			}
		}
		else
		{
			ShakeHandler(0.5f, 4f, withDamage);
		}
	}

	private static void ShakeHandler(float duration, float strength, bool withDamage)
	{
		Transform trans = CameraController.MainCamera.transform;
		trans.DOKill();
		trans.localPosition = _cameraOffset;
		trans.DOShakePosition(duration, new Vector3(strength, strength) * 0.08f).SetUpdate(isIndependentUpdate: true).OnComplete(delegate
		{
			trans.localPosition = _cameraOffset;
			if (withDamage)
			{
				ShakingDamage = 0;
			}
		});
	}

	public static void ShakeUi(int level = 0)
	{
		if (level < 4)

			switch (level)

	}

												{

													{

														{
															if (!(args is PerformAction.DeathAnimationArgs deathAnimationArgs))

																if (!(args is PerformAction.WaitArgs waitArgs))

																{

																}
																else
																{
																	yield return new WaitForSeconds(waitArgs.Time);
																}
															}
															else
															{
																UnitView unit = GetUnit(deathAnimationArgs.Source);
																if (!unit)
																{
																	Debug.LogWarning("[GameDirector] Perform: cannot get source view for " + deathAnimationArgs.Source.DebugName);
																}
																else
																{
																	unit.DeathAnimation();
																}
															}
														}
														else
														{
															UnitView unit2 = GetUnit(transformModelArgs.Source);
															if (!unit2)
															{
																Debug.LogWarning("[GameDirector] Perform: cannot get source view for " + transformModelArgs.Source.DebugName);
															}
															else
															{
																yield return unit2.LoadUnitModelAsync(transformModelArgs.ModelName);

														}

													}
												}
											}
											else
											{
												UnitView unit4 = GetUnit(effectMessageArgs.Source);
												if (!unit4)
												{
													Debug.LogWarning("[GameDirector] Perform: cannot get source view for " + effectMessageArgs.Source.DebugName);
												}
												else
												{
													unit4.SendEffectMessage(effectMessageArgs.EffectName, effectMessageArgs.Message, effectMessageArgs.Args);

											}

										}
										UnitView unit5 = GetUnit(effectArgs.Source);

											break;
										case PerformAction.EffectBehavior.Add:
											unit5.PlayEffectLoop(effectArgs.EffectName);
											break;
										case PerformAction.EffectBehavior.Remove:
											unit5.EndEffectLoop(effectArgs.EffectName, instant: true);
											break;
										case PerformAction.EffectBehavior.DieOut:
											unit5.EndEffectLoop(effectArgs.EffectName, instant: false);
											break;
										default:
											throw new ArgumentOutOfRangeException();
										}
										if (!effectArgs.SfxId.IsNullOrEmpty())
										{
											AudioManager.PlaySfxDelay(effectArgs.SfxId, effectArgs.SfxDelay);
										}

				unit10.ComingDamage = DamageInfo.Attack(0f);
				unit9.PerformShoot(gunArgs.GunId);
				yield return new WaitForSeconds(gunArgs.WaitTime);
			}
		}
		else
		{
			yield return UiManager.GetPanel<PlayBoard>().CardUi.ViewCardFromZone(viewCardArgs.Card, viewCardArgs.Zone);
		}

			int bluePoint2 = bluePoint;
			int money2 = money;
			yield return EnemyDeathPoints(unitView.transform, red, bluePoint2, money2);
		}
		GameRunController gameRun = unit.GameRun;
		if (!gameRun.IsAutoSeed || !gameRun.JadeBoxes.Empty())
		{
			yield break;
		}
		if (unit is Aya && unit.TurnCounter == 0)
		{
			GameMaster.UnlockAchievement(AchievementKey.AyaSpecial);
			yield break;
		}
		if (unit is Youmu)
		{
			LouguanJianSe statusEffect = unit.GetStatusEffect<LouguanJianSe>();
			if (statusEffect != null && statusEffect.TriggerCount == 0)
			{
				GameMaster.UnlockAchievement(AchievementKey.YoumuSpecial);
				yield break;
			}
		}
		if (unit is Doremy { IsWokeUp: not false })
		{
			GameMaster.UnlockAchievement(AchievementKey.DoremySpecial);
		}
		else if (unit is Sakuya && gameRun.CurrentStage is BambooForest && gameRun.Player.IsExtraTurn)
		{
			GameMaster.UnlockAchievement(AchievementKey.SakuyaSpecial);
		}
	}

	private static IEnumerator MultipleDieViewer(DieAction action)
	{
		List<Coroutine> list = new List<Coroutine>();
		DieEventArgs[] argsList = action.ArgsList;
		foreach (DieEventArgs dieEventArgs in argsList)
		{
			if (GetUnit(dieEventArgs.Unit) != null)
			{
				list.Add(Singleton<GameDirector>.Instance.StartCoroutine(InternalDieViewer(dieEventArgs)));
			}
		}
		foreach (Coroutine item in list)
		{
			yield return item;
		}
		UpdateEnemyMoveOrder();
	}

	private static IEnumerator EnemyDeathPoints(Transform enemyTrans, int red, int bluePoint, int money)
	{
		int num = red;
		int num2 = 0;
		int num3 = bluePoint;
		int num4 = 0;
		while (num > 8)
		{
			num -= 5;
			num2++;
		}
		while (num3 > 8)
		{
			num3 -= 5;
			num4++;
		}
		for (int i = 0; i < num; i++)
		{
			EffectManager.CreateEffectBullet(new Point
			{
				Type = Point.PointType.Power,
				TargetPosition = CameraController.ScenePositionToWorldPositionInUI(Player.transform.position)
			}, CameraController.ScenePositionToWorldPositionInUI(enemyTrans.position), UiManager.GetPanel<GameRunVisualPanel>().transform);
		}
		for (int j = 0; j < num2; j++)
		{
			EffectManager.CreateEffectBullet(new Point
			{
				Type = Point.PointType.BigPower,
				TargetPosition = CameraController.ScenePositionToWorldPositionInUI(Player.transform.position)
			}, CameraController.ScenePositionToWorldPositionInUI(enemyTrans.position), UiManager.GetPanel<GameRunVisualPanel>().transform);
		}
		for (int k = 0; k < num3; k++)
		{
			EffectManager.CreateEffectBullet(new Point
			{
				Type = Point.PointType.Blue,
				TargetPosition = Player.transform.position
			}, enemyTrans.position);
		}
		for (int l = 0; l < num4; l++)
		{
			EffectManager.CreateEffectBullet(new Point
			{
				Type = Point.PointType.BigBlue,
				TargetPosition = Player.transform.position
			}, enemyTrans.position);
		}
		if (money > 0)
		{
			GameRunVisualPanel panel = UiManager.GetPanel<GameRunVisualPanel>();
			UiManager.GetPanel<SystemBoard>().CreateMoneyGainVisual(CameraController.ScenePositionToWorldPositionInUI(enemyTrans.position), money, panel.transform);
		}
		yield return new WaitForSeconds(1f);
		UiManager.GetPanel<UltimateSkillPanel>().GainPower(red);
		if (money > 0)
		{
			UiManager.GetPanel<SystemBoard>().OnMoneyChanged();
		}
	}

	private static IEnumerator CastBlockShieldViewer(CastBlockShieldAction action)
	{
		UnitView unit = GetUnit(action.Args.Source);
		UnitView unit2 = GetUnit(action.Args.Target);
		if (!(unit2 == null))
		{
			if (unit2 == unit)
			{
				yield return unit2.GainBlockShieldToSelfViewer(action);
				yield break;
			}
			unit.CastShieldToOther(action.Cast);
			yield return unit2.ReceiveShieldViewer(action);
		}
	}

	private static IEnumerator LoseBlockShieldViewer(LoseBlockShieldAction action)
	{
		UnitView unit = GetUnit(action.Args.Target);
		if (!(unit == null))
		{
			return unit.LoseBlockShieldViewer(action);
		}
		return null;
	}

	private IEnumerator ApplyStatusEffectViewer(ApplyStatusEffectAction action)
	{
		StatusEffectApplyEventArgs args = action.Args;
		StatusEffect effect = args.Effect;
		UnitView unit = GetUnit(action.Args.Unit);
		if (unit != null && unit.Unit.IsAlive)
		{
			int number = 0;
			if (effect.HasLevel && args.Level.HasValue)
			{
				number = args.Level.Value;
			}
			else if (effect.HasDuration && args.Duration.HasValue)
			{
				number = args.Duration.Value;
			}
			unit.ShowSePopup(effect.Name, effect.Type, number);
			unit.OnAddStatusEffect(args.Effect, args.AddResult.Value);
			if (effect.Config.SFX.IsNullOrEmpty() || effect.Config.SFX == "Default")
			{
				switch (effect.Config.Type)
				{
				case StatusEffectType.Positive:
					AudioManager.PlaySfx("Buff");
					break;
				case StatusEffectType.Negative:
					AudioManager.PlaySfx("Debuff");
					break;
				default:
					throw new ArgumentOutOfRangeException();
				case StatusEffectType.Special:
					break;
				}
			}
			else
			{
				AudioManager.PlaySfx(effect.Config.SFX);
			}
			if (!effect.Config.VFX.IsNullOrEmpty() && !(effect.Config.VFX == "Default"))
			{
				EffectManager.CreateEffect(effect.Config.VFX, unit.EffectRoot);
			}
			string unitEffectName = effect.UnitEffectName;
			if (unitEffectName != null && args.AddResult == StatusEffectAddResult.Added)
			{
				unit.TryPlayEffectLoop(unitEffectName);
				unit.SendEffectMessage(unitEffectName, "OnPropertyChanged", effect);
				yield return new WaitForSeconds(0.2f);
			}
		}
		yield return new WaitForSeconds(args.WaitTime);
	}

	private IEnumerator RemoveStatusEffectViewer(RemoveStatusEffectAction action)
	{
		StatusEffectEventArgs args = action.Args;
		StatusEffect effect = args.Effect;
		UnitView unit = GetUnit(action.Args.Unit);
		if (unit != null)
		{
			if (unit.Unit.IsAlive)
			{
				unit.ShowSePopup(effect.Name, effect.Type, 0, UnitInfoWidget.SePopType.Remove);
			}
			unit.OnRemoveStatusEffect(effect);
			string unitEffectName = effect.UnitEffectName;
			if (unitEffectName != null)
			{
				unit.EndEffectLoop(unitEffectName, instant: true);
			}
		}
		yield return new WaitForSeconds(args.WaitTime);
	}
