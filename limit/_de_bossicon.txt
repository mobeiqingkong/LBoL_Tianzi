	/// <summary>
	/// Numbering should start from 0. First enemy from enemy list will go to slot 0, second to slot 1 and so on. Make sure enemy act order display is consistent with the game's (left to right first, then top to bottom).
	/// </summary>
	/// <param name="name"></param>
	/// <param name="enemyLocations"></param>
	public static void AddFormation(string name, Dictionary<int, Vector2> enemyLocations, Assembly callingAssembly = null)
	{
		if (callingAssembly == null)
		{
			callingAssembly = Assembly.GetCallingAssembly();
		}
		Action item = delegate
		{
			if (!Singleton<GameDirector>.Instance.Formations.ContainsKey(name))
			{
				GameObject gameObject = new GameObject(name);
				EnemyFormation enemyFormation = gameObject.AddComponent<EnemyFormation>();
				customFormations.Add(name);
				gameObject.transform.SetParent(Singleton<GameDirector>.Instance.unitRoot);
				gameObject.transform.position = Singleton<GameDirector>.Instance.unitRoot.position;
				Singleton<GameDirector>.Instance.Formations.Add(name, enemyFormation);
				enemyFormation.enemyLocations = new AssociationList<int, Transform>();
				{
					foreach (KeyValuePair<int, Vector2> enemyLocation in enemyLocations)
					{
						GameObject gameObject2 = new GameObject(enemyLocation.Key.ToString());
						gameObject2.transform.SetParent(enemyFormation.transform);
						gameObject2.transform.localPosition = enemyLocation.Value;
						enemyFormation.enemyLocations.Add(enemyLocation.Key, gameObject2.transform);
					}
					return;
				}
			}
			Log.log.LogError((object)("Formation with name " + name + " is already present"));
		};
		if (callingAssembly.IsLoadedFromDisk())
		{
			loadedFromDiskCustomFormations.Add(item);
		}
		UniqueTracker.Instance.formationAddActions.Add(item);
	}

	internal static void ReloadFormations()
	{
		List<GameObject> formationsToDestroy = FindFormationsToUnload();
		UnloadCustomFormations(formationsToDestroy);
		LoadCustomFormations();
	}

	internal static void LoadCustomFormations()
	{
		CollectionExtensions.Do<Action>((IEnumerable<Action>)UniqueTracker.Instance.formationAddActions, (Action<Action>)delegate(Action a)
		{
			a();
		});
	}

	internal static List<GameObject> FindFormationsToUnload()
	{
		List<GameObject> list = new List<GameObject>();
		foreach (object item in Singleton<GameDirector>.Instance.unitRoot)
		{
			GameObject gameObject = (item as Transform).gameObject;
			if (gameObject.TryGetComponent<EnemyFormation>(out var _) && customFormations.Contains(gameObject.name))
			{
				list.Add(gameObject);
			}
		}
		customFormations.Clear();
		return list;
	}

	internal static void UnloadCustomFormations(List<GameObject> formationsToDestroy)
	{
		foreach (GameObject item in formationsToDestroy)
		{
			Singleton<GameDirector>.Instance.Formations.Remove(item.name);
			UnityEngine.Object.Destroy(item);
		}
	}

	/// <summary>
	/// Id : should be a string. In vanilla strings which look like numbers are used but any string can be used
	/// Name : 
	/// FormationName : preset positions for all enemy units. Must be one of available formations. Custom formations can be added with EnemyGroupTemplate.AddFormation
	/// Enemies : List of enemy Ids. The list order determines enemy acting order with the first entry acting first, second acting second and so on. An entry "Empty" can be added to this list indicating that the slot is empty and can be filled with summoned unit.
	/// EnemyType : 
	/// DebutTime : 
	/// RollBossExhibit : 
	/// PlayerRoot : always Vector2(-4f, 0.5f)
	/// PreBattleDialogName : no way to add dialog yet
	/// PostBattleDialogName : no way to add dialog yet 
	/// </summary>
	/// <returns></returns>
	public EnemyGroupConfig DefaultConfig()
	{
		return new EnemyGroupConfig("", IsSub: false, (IReadOnlyList<string>)new List<string>(), "", "Single", (IReadOnlyList<string>)new List<string>(), EnemyType.Normal, Hidden: false, 1f, RollBossExhibit: false, new Vector2(-4f, 0.5f), "", "", "");
	}

	public abstract EnemyGroupConfig MakeConfig();
}
