	private GameRunController(GameRunStartupParameters startupParameters)
		: this()
	{
		Stats = new GameRunStats();
		HasClearBonus = startupParameters.UserProfile.HasClearBonus;
		UnlockLevel = startupParameters.UnlockLevel;
		PlayerUnit player = startupParameters.Player;
		Exhibit initExhibit = startupParameters.InitExhibit;
		Mode = startupParameters.Mode;
		ShowRandomResult = startupParameters.ShowRandomResult;
		Difficulty = startupParameters.Difficulty;
		Puzzles = startupParameters.Puzzles;
		Packs = startupParameters.UserProfile.Settings.Packs.ToList();
		_stages = new List<Stage>(startupParameters.Stages);
		if (_stages.Empty())
		{
			throw new ArgumentException("Cannot create game-run with empty stages");
		}
		foreach (var item4 in _stages.WithIndices())
		{
			int item = item4.index;
			Stage item2 = item4.elem;
			item2.GameRun = this;
			item2.Index = item;
			item2.InitExtraFlags(startupParameters.UserProfile);
		}
		ulong? seed = startupParameters.Seed;
		if (seed.HasValue)
		{
			ulong valueOrDefault = seed.GetValueOrDefault();
			RootSeed = valueOrDefault;
			IsAutoSeed = false;
			HasClearBonus = true;
		}
		else
		{
			RootSeed = RandomGen.GetRandomSeed();
			IsAutoSeed = true;
		}
		RootRng = new RandomGen(RootSeed);
		foreach (Stage stage in _stages)
		{
			stage.MapSeed = RootRng.NextULong();
		}
		_stages[0].DebutAdventureType = startupParameters.DebutAdventureType;
		StationRng = new RandomGen(RootRng.NextULong());
		InitBossSeed = RootRng.NextULong();
		ShopRng = new RandomGen(RootRng.NextULong());
		AdventureRng = new RandomGen(RootRng.NextULong());
		ExhibitRng = new RandomGen(RootRng.NextULong());
		ShiningExhibitRng = new RandomGen(RootRng.NextULong());
		CardRng = new RandomGen(RootRng.NextULong());
		GameRunEventRng = new RandomGen(RootRng.NextULong());
		BattleRng = new RandomGen(RootRng.NextULong());
		BattleCardRng = new RandomGen(RootRng.NextULong());
		BattleTransformCardRng = new RandomGen(RootRng.NextULong());
		ShuffleRng = new RandomGen(RootRng.NextULong());
		EnemyMoveRng = new RandomGen(RootRng.NextULong());
		EnemyBattleRng = new RandomGen(RootRng.NextULong());
		DebutRng = new RandomGen(RootRng.NextULong());
		FinalBossSeed = RootRng.NextULong();
		UISeed = RootRng.NextULong();
		RandomGen randomGen = new RandomGen(InitBossSeed);
		foreach (Stage stage2 in _stages)
		{
			stage2.InitBoss(randomGen);
		}
		foreach (Stage stage3 in _stages)
		{
			stage3.InitFirstAdventure(randomGen);
		}
		Player = player;
		PlayerType = startupParameters.PlayerType;
		player.Us.GameRun = this;
		_baseDeck = new List<Card>(startupParameters.Deck);
		foreach (Card item5 in _baseDeck)
		{
			item5.IsGamerunInitial = true;
		}
		if (Puzzles.HasFlag(PuzzleFlag.StartMisfortune))
		{
			_baseDeck.Add(Library.CreateCard<Zhukeling>());
		}
		foreach (Card item6 in _baseDeck)
		{
			item6.InstanceId = ++_deckCardInstanceId;
		}
		BaseMana = player.Config.InitialMana;
		Money = startupParameters.InitMoneyOverride ?? player.Config.InitialMoney;
		TotalMoney = Money;
		ExhibitPool = new List<Type>();
		ShiningExhibitPool = new List<Type>();
		foreach (var (item3, exhibitConfig) in Library.EnumerateRollableExhibitTypes(startupParameters.UnlockLevel))
		{
			if (exhibitConfig.Rarity == Rarity.Shining && exhibitConfig.Id != initExhibit.Id)
			{
				ShiningExhibitPool.Add(item3);
			}
			else if (exhibitConfig.IsPooled)
			{
				ExhibitPool.Add(item3);
			}
		}
		_cardRewardWeightFactors = new Dictionary<string, float>();
		_cardRareWeightFactor = 0.85f;
		foreach (Card item7 in _baseDeck)
		{
			item7.GameRun = this;
		}
		player.EnterGameRun(this);
		if (Puzzles.HasFlag(PuzzleFlag.LowMaxHp))
		{
			int num = Math.Max(1, player.MaxHp - 10);
			player.SetMaxHp(num, num);
		}
		GainExhibitInstantly(initExhibit);
		HashSet<string> hashSet = new HashSet<string>();
		foreach (JadeBox jadeBox in startupParameters.JadeBoxes)
		{
			foreach (string item8 in jadeBox.Config.Group)
			{
				if (hashSet.Contains(item8))
				{
					throw new InvalidOperationException("Cannot gain jade-box " + jadeBox.DebugName + ": already has group " + item8);
				}
				hashSet.Add(item8);
			}
			GainJadeBox(jadeBox);
		}
		if (HasJadeBox<TwoColorStart>())
		{
			Exhibit exhibit = Library.CreateExhibit((PlayerType == PlayerType.TypeA) ? player.Config.ExhibitB : player.Config.ExhibitA);
			GainExhibitInstantly(exhibit);
			ShiningExhibitPool.Remove(exhibit.GetType());
			GainBaseMana(ManaGroup.Single(ManaColor.Philosophy));
			LoseBaseMana(initExhibit.BaseMana);
			LoseBaseMana(exhibit.BaseMana);
		}
		if (Difficulty == GameDifficulty.Easy || _jadeBoxes.NotEmpty())
		{
			CharacterStatsSaveData characterStatsSaveData = startupParameters.UserProfile.CharacterStats.FirstOrDefault((CharacterStatsSaveData s) => s.CharacterId == player.Id);
			if (characterStatsSaveData != null && characterStatsSaveData.HighestPerfectSuccessDifficulty.HasValue)
			{
				ExtraFlags.Add("DebutYuzhi");
			}
		}
	}

	public bool HasJadeBox(string id)
	{
		Type type = TypeFactory<Exhibit>.GetType(id);
		if (type != null)
		{
			return HasJadeBox(type);
		}
		Debug.LogError("Cannot find jadeBox type '" + id + "'");
		return false;
	}

	public bool HasJadeBox(Type type)
	{
		return _jadeBoxes.Any((JadeBox jadeBox) => jadeBox.GetType() == type);
	}

	public bool HasJadeBox<T>() where T : JadeBox
	{
		return HasJadeBox(typeof(T));
	}

	public bool HasJadeBox(JadeBox jadeBox)
	{
		return _jadeBoxes.Contains(jadeBox);
	}

	private void CheckMapMode()
	{
		_activeMapModeOverrider = null;
		_mapMode = GameRunMapMode.Normal;
		foreach (IMapModeOverrider mapModeOverrider in _mapModeOverriders)
		{
			GameRunMapMode? mapMode = mapModeOverrider.MapMode;
			if (mapMode.HasValue)
			{
				GameRunMapMode valueOrDefault = mapMode.GetValueOrDefault();
				if (valueOrDefault.CompareTo(_mapMode) > 0)
				{
					_activeMapModeOverrider = mapModeOverrider;
					_mapMode = valueOrDefault;
				}
			}
		}
		Station currentStation = CurrentStation;
		if (currentStation != null && currentStation.Status == StationStatus.Finished)
		{
			CurrentMap.SetAdjacentNodesStatus(_mapMode);
		}
	}

	public void AddMapModeOverrider(IMapModeOverrider overrider)
	{
		if (overrider.MapMode == GameRunMapMode.TeleportBoss)
		{
			Stage currentStage = CurrentStage;
			if (currentStage.IsSelectingBoss && currentStage.SelectedBoss == null)
			{
				throw new InvalidOperationException("Current stage " + currentStage.Id + " boss is not selected, cannot teleport");
			}
		}
		if (_mapModeOverriders.Contains(overrider))
		{
			throw new InvalidOperationException($"MapModeOverrider {overrider} already in overrider list");
		}
		_mapModeOverriders.Add(overrider);
		CheckMapMode();
	}

	public void RemoveMapModeOverrider(IMapModeOverrider overrider)
	{
		if (!_mapModeOverriders.Remove(overrider))
		{
			throw new InvalidOperationException($"Removing MapModeOverrider {overrider} not in overrider list");
		}
		if (overrider == _activeMapModeOverrider)
		{
			CheckMapMode();
		}
	}

	public void UpgradeNewDeckCardOnFlags(IEnumerable<Card> cards)

	}

	public void UpgradeNewDeckCardOnFlags(Card card)
	{
		if (card.IsUpgraded || !card.CanUpgrade)
		{
			return;
		}
		switch (card.CardType)
		{
		case CardType.Attack:
			if (UpgradeNewDeckAttackCardFlag > 0)
			{
				card.Upgrade();
			}
			break;
		case CardType.Defense:
			if (UpgradeNewDeckDefenseCardFlag > 0)
			{
				card.Upgrade();
			}
			break;
		case CardType.Skill:
			if (UpgradeNewDeckSkillCardFlag > 0)
			{
				card.Upgrade();
			}
			break;
		case CardType.Ability:
			if (UpgradeNewDeckAbilityCardFlag > 0)
			{
				card.Upgrade();
			}
			break;
		default:
			throw new ArgumentOutOfRangeException();
		case CardType.Unknown:
		case CardType.Friend:
		case CardType.Tool:
		case CardType.Status:
		case CardType.Misfortune:
			break;
		}
	}

	private void EnterStage(int index)
	{
		if (index < 0 || index >= _stages.Count)
		{
			throw new IndexOutOfRangeException($"Enter stage index {index} out of range");
		}
		_stageIndex = index;
		Stage stage2 = (CurrentStage = _stages[index]);
		CurrentMap = stage2.CreateMap();
		stage2.Enter();
		StageRecords.Add(new StageRecord
		{
			Id = stage2.Id
		});
		StageEntered.Execute(new GameEventArgs
		{
			CanCancel = false
		});
	}

	public void EnterNextStage()
	{
		if (Status != 0)
		{
			throw new InvalidOperationException("Cannot enter next stage while not running");
		}
		LeaveStation();
		if (_stageIndex >= 0)
		{
			if (Puzzles.HasFlag(PuzzleFlag.LowStageRegen))
			{
				Heal(((float)Player.MaxHp * 0.5f).RoundToInt());
			}
			else
			{
				HealToMaxHp();
			}
		}
		EnterStage(_stageIndex + 1);
	}

	private void LeaveStation()
	{
		Station currentStation = CurrentStation;
		if (currentStation != null)
		{
			StationEventArgs args = new StationEventArgs
			{
				CanCancel = false,
				Station = currentStation
			};
			StationLeaving.Execute(args);
			currentStation.OnLeave();
			StageRecord stageRecord = StageRecords.LastOrDefault();
			if (stageRecord == null)
			{
				Debug.LogError("Leave station while stage records is empty");
			}
			else
			{
				stageRecord.Stations.Add(currentStation.GenerateRecord());
			}
			CurrentStation = null;
			StationLeft.Execute(args);
		}
	}

	private void EnterStation(Station station)
	{
		StationEventArgs args = new StationEventArgs
		{
			CanCancel = false,
			Station = station
		};
		StationEntering.Execute(args);
		CurrentStation = station;
		station.OnEnter();
		StationEntered.Execute(args);
	}

	public GameRunSaveData EnterMapNode(MapNode node, bool forced = false)
	{
		if (Status != 0)
		{
			throw new InvalidOperationException("Cannot enter map node while not running");
		}
		if (CurrentStation != null && CurrentStation.Status != StationStatus.Finished)
		{
			throw new InvalidOperationException("Cannot enter other map-node when current station is not finished");
		}
		if (!forced)
		{
			switch (MapMode)
			{
			case GameRunMapMode.Normal:
				if (node.Status != MapNodeStatus.Active)
				{
					throw new InvalidOperationException(string.Format("Entering MapNode status {0} != {1}", node.Status, "Active"));
				}
				break;
			case GameRunMapMode.Crossing:
				if (node.Status != MapNodeStatus.Active && node.Status != MapNodeStatus.CrossActive)
				{
					throw new InvalidOperationException(string.Format("Entering MapNode status {0} != {1} or {2}", node.Status, "Active", "CrossActive"));
				}
				break;
			case GameRunMapMode.TeleportBoss:
				if (node.StationType != StationType.Boss)
				{
					throw new InvalidOperationException("Teleporting to non-boss map-node");
				}
				break;
			}
		}
		LeaveStation();
		GameRunSaveData gameRunSaveData = Save();
		gameRunSaveData.Timing = SaveTiming.EnterMapNode;
		gameRunSaveData.EnteringNode = new MapNodeSaveData
		{
			X = node.X,
			Y = node.Y
		};
		if (MapMode == GameRunMapMode.Normal)
		{
			CurrentMap.EnterNode(node, freeMove: false, forced);
		}
		else if (MapMode == GameRunMapMode.Crossing)
		{
			if (node.Status == MapNodeStatus.CrossActive)
			{
				_activeMapModeOverrider.OnEnteredWithMode();
				CurrentMap.EnterNode(node, freeMove: true, forced);
			}
			else
			{
				CurrentMap.EnterNode(node, freeMove: false, forced);
			}
		}
		else if (MapMode == GameRunMapMode.TeleportBoss)
		{
			_activeMapModeOverrider.OnEnteredWithMode();
			CurrentMap.EnterNode(node, freeMove: true, forced);
		}
		else if (MapMode == GameRunMapMode.Free)
		{
			_activeMapModeOverrider.OnEnteredWithMode();
			CurrentMap.EnterNode(node, freeMove: false, forced);
		}
		CheckMapMode();
		Station station = CurrentStage.CreateStation(node);
		EnterStation(station);
		return gameRunSaveData;
	}

	public void EnterBattle(EnemyGroup enemyGroup)
	{
		RevealEnemyGroup(enemyGroup);
		Battle = new BattleController(this, enemyGroup, _baseDeck);
		foreach (Exhibit exhibit in Player.Exhibits)
		{
			exhibit.EnterBattle();
		}
		foreach (JadeBox jadeBox in _jadeBoxes)
		{
			jadeBox.EnterBattle();
		}
	}

	public BattleStats LeaveBattle(EnemyGroup enemyGroup)
	{
		foreach (Exhibit exhibit in Player.Exhibits)
		{
			exhibit.LeaveBattle();
		}
		foreach (JadeBox jadeBox in _jadeBoxes)
		{
			jadeBox.LeaveBattle();
		}
		Battle.Leave();
		BattleStats stats = Battle.Stats;
		Stats.ContinuousTurnCount = Math.Max(Stats.ContinuousTurnCount, stats.ContinuousTurnCount);
		Stats.MaxSingleAttackDamage = Math.Max(Stats.MaxSingleAttackDamage, stats.MaxSingleAttackDamage);
		if (Player.IsDead)
		{
			Status = GameRunStatus.Failure;
			Stats.PlayerSuicide = stats.PlayerSuicide;
			GenerateRecords(isAbandon: false, enemyGroup);
		}
		else
		{
			while (Stats.Stages.Count <= _stageIndex)
			{
				Stats.Stages.Add(new StageStats());
			}
			StageStats stageStats = Stats.Stages[_stageIndex];
			switch (enemyGroup.EnemyType)
			{
			case EnemyType.Normal:
				stageStats.NormalEnemyDefeated++;
				stageStats.NormalEnemyBluePoint += stats.BluePoint;
				break;
			case EnemyType.Elite:
				stageStats.EliteEnemyDefeated++;
				stageStats.EliteEnemyBluePoint += stats.BluePoint;
				if (!stats.PlayerDamaged)
				{
					Stats.PerfectElite++;
				}
				break;
			case EnemyType.Boss:
				stageStats.BossDefeated++;
				stageStats.BossBluePoint += stats.BluePoint;
				if (!CurrentStage.IsTrueEndFinalStage && !stats.PlayerDamaged)
				{
					Stats.PerfectBoss++;
				}
				break;
			default:
				throw new ArgumentOutOfRangeException();
			}
		}
		if (enemyGroup.EnemyType == EnemyType.Boss)
		{
			Stats.Bosses.Add(enemyGroup.Id);
		}
		Battle = null;
		foreach (EnemyUnit item in enemyGroup)
		{
			item.LeaveGameRun();
		}
		return stats;
	}

	internal Exhibit RollNormalExhibit(RandomGen rng, ExhibitWeightTable weightTable, Func<Exhibit> fallback, [MaybeNull] Predicate<ExhibitConfig> filter = null)
	{
		RepeatableRandomPool<Type> repeatableRandomPool = new RepeatableRandomPool<Type>();
		foreach (Type item in ExhibitPool)
		{
			ExhibitConfig exhibitConfig = ExhibitConfig.FromId(item.Name);
			ManaColor? baseManaRequirement = exhibitConfig.BaseManaRequirement;
			if (baseManaRequirement.HasValue)
			{
				ManaColor valueOrDefault = baseManaRequirement.GetValueOrDefault();
				if (BaseMana[valueOrDefault] == 0)
				{
					continue;
				}
			}
			if (filter == null || filter(exhibitConfig))
			{
				float num = weightTable.WeightFor(exhibitConfig);
				if (num > 0f)
				{
					repeatableRandomPool.Add(item, Library.WeightForExhibit(item, this) * num);
				}
			}
		}
		Type type = repeatableRandomPool.SampleOrDefault(rng);
		if (type != null)
		{
			ExhibitPool.Remove(type);
			return Library.CreateExhibit(type);
		}
		return fallback();
	}

	internal Exhibit RollShiningExhibit(RandomGen rng, [MaybeNull] Func<Exhibit> fallback = null, [MaybeNull] Predicate<ExhibitConfig> filter = null)
	{
		RepeatableRandomPool<Type> repeatableRandomPool = new RepeatableRandomPool<Type>();
		foreach (Type item in ShiningExhibitPool)
		{
			ExhibitConfig obj = ExhibitConfig.FromId(item.Name);
			if ((filter == null || filter(obj)) && !Player.HasExhibit(item))
			{
				repeatableRandomPool.Add(item, Library.WeightForExhibit(item, this));
			}
		}
		if (repeatableRandomPool.IsEmpty)
		{
			return fallback?.Invoke();
		}
		Type type = repeatableRandomPool.Sample(rng);
		ShiningExhibitPool.Remove(type);
		return Library.CreateExhibit(type);
	}

	internal Exhibit[] RollBossExhibits(RandomGen rng, string bossId, bool rollBossExhibit, Func<Exhibit> fallback)
	{
		EnemyUnitConfig enemyUnitConfig = EnemyUnitConfig.FromId(bossId);
		if (enemyUnitConfig == null)
		{
			throw new ArgumentException("Cannot find boss config for '" + bossId + "'", "bossId");
		}
		Exhibit exhibit = null;
		if (rollBossExhibit)
		{
			exhibit = RollShiningExhibit(rng, null, (ExhibitConfig config) => bossId.Equals(config.Owner, StringComparison.Ordinal));
		}
		if (exhibit == null)
		{
			IReadOnlyList<ManaColor> bossColors = enemyUnitConfig.BaseManaColor;
			if (bossColors.Empty())
			{
				Debug.LogError("Cannot roll fallback exhibit for " + bossId + ": no BaseManaColor defined, using fallback.");
				exhibit = fallback();
			}
			else
			{
				exhibit = RollShiningExhibit(rng, null, delegate(ExhibitConfig config)
				{
					if (!string.IsNullOrWhiteSpace(config.Owner))
					{
						return false;
					}
					ManaColor? baseManaColor2 = config.BaseManaColor;
					if (baseManaColor2.HasValue)
					{
						ManaColor valueOrDefault2 = baseManaColor2.GetValueOrDefault();
						return bossColors.Contains(valueOrDefault2);
					}
					return false;
				});
				if (exhibit == null)
				{
					string text = string.Join(", ", bossColors.Select((ManaColor c) => c.ToShortName()));
					Debug.LogWarning("Cannot roll boss fallback shining exhibit with color [" + text + "], using fallback.");
					exhibit = fallback();
				}
			}
		}
		ManaColor? bossExhibitColor = exhibit?.Config.BaseManaColor;
		HashSet<ManaColor> playerColorSet = (from c in BaseMana.EnumerateColors()
			where c != bossExhibitColor
			select c).ToHashSet();
		Exhibit exhibit2 = RollShiningExhibit(rng, null, delegate(ExhibitConfig config)
		{
			if (!string.IsNullOrWhiteSpace(config.Owner))
			{
				return false;
			}
			ManaColor? baseManaColor = config.BaseManaColor;
			if (baseManaColor.HasValue)
			{
				ManaColor valueOrDefault = baseManaColor.GetValueOrDefault();
				return playerColorSet.Contains(valueOrDefault);
			}
			return false;
		});
		if (exhibit2 == null)
		{
			string text2 = string.Join(", ", playerColorSet);
			Debug.LogWarning("Cannot roll exhibit for " + Player.DebugName + " with color [" + text2 + "], using fallback.");
			exhibit2 = fallback();
		}
		ManaColor? playerExhibitColor = exhibit2?.Config.BaseManaColor;
		Exhibit exhibit3 = RollShiningExhibit(rng, null, delegate(ExhibitConfig config)
		{
			if (!string.IsNullOrWhiteSpace(config.Owner))
			{
				return false;
			}
			return config.BaseManaColor != playerExhibitColor && config.BaseManaColor != bossExhibitColor;
		});
		if (exhibit3 == null)
		{
			string text3 = playerExhibitColor?.ToShortName() + ", " + bossExhibitColor?.ToShortName();
			Debug.Log("Cannot roll exhibit without color [" + text3 + "], using fallback");
			exhibit3 = fallback();
		}
		return new Exhibit[3] { exhibit2, exhibit, exhibit3 };
	}

	private float BaseCardWeight(CardConfig config, bool applyFactors)
	{
		ManaGroup cost = config.Cost;
		float num = Math.Max(config.Colors.Count, config.Cost.TrivialColorCount) switch
		{
			0 => 0.9f, 
			1 => cost.GetValue(cost.MaxTrivialColor) switch
			{
				0 => 0.9f, 
				1 => 0.9f, 
				2 => 1f, 
				3 => 1.1f, 
				4 => 1.2f, 
				_ => throw new InvalidDataException($"Invalid cost pattern {cost} of card '{config.Id}'"), 
			}, 
			2 => cost.GetValue(cost.MaxTrivialColor) switch
			{
				0 => 0.9f, 
				1 => 1f, 
				2 => 1.1f, 
				_ => throw new InvalidDataException($"Invalid cost pattern {cost} of card '{config.Id}'"), 
			}, 
			3 => 1.3f, 
			_ => throw new InvalidDataException($"Invalid cost pattern {cost} of card '{config.Id}'"), 
		};
		float num2 = 1f;
		int count = config.Colors.Count;
		if (count <= 0)
		{
			if (count == 0)
			{
				num2 = 0.8f;
			}
		}
		else
		{
			foreach (ManaColor color in config.Colors)
			{
				float num3 = (float)BaseMana.GetValue(color) / (float)BaseMana.Amount;
				num3 -= 0.5f;
				num3 *= 0.8f;
				num2 += num3;
			}
			num2 = Math.Max(num2, 0.8f);
		}
		num *= num2;
		if (applyFactors)
		{
			if (config.Rarity == Rarity.Rare)
			{
				num *= _cardRareWeightFactor;
			}
			if (_cardRewardWeightFactors.TryGetValue(config.Id, out var value))
			{
				num *= value;
			}
		}
		return num;
	}

	public UniqueRandomPool<Type> CreateValidCardsPool(CardWeightTable weightTable, ManaGroup? manaLimit, bool colorLimit, bool applyFactors, bool battleRolling, [MaybeNull] Predicate<CardConfig> filter = null)
	{
		HashSet<string> hashSet = new HashSet<string> { Player.Id };
		if (AllCharacterCardsFlag > 0)
		{
			foreach (PlayerUnitConfig item2 in PlayerUnitConfig.AllConfig())
			{
				hashSet.Add(item2.Id);
			}
		}
		else
		{
			foreach (Exhibit exhibit in Player.Exhibits)
			{
				if (exhibit.OwnerId != null)
				{
					hashSet.Add(exhibit.OwnerId);
				}
			}
		}
		UniqueRandomPool<Type> uniqueRandomPool = new UniqueRandomPool<Type>();
		foreach (var item3 in Library.EnumerateRollableCardTypes(UnlockLevel))
		{
			Type item = item3.Item1;
			CardConfig config = item3.Item2;
			if (!config.IsPooled || config.DebugLevel > CardValidDebugLevel || (battleRolling && !config.FindInBattle) || (!config.Pack.IsNullOrEmpty() && !Packs.Any((string ps) => ps == config.Pack)) || (filter != null && !filter(config)))
			{
				continue;
			}
			if (manaLimit.HasValue)
			{
				ManaGroup mana = manaLimit.GetValueOrDefault();
				ManaGroup cost = config.Cost;
				if (cost.Amount > 5)
				{
					if (!mana.CanAfford(cost.WithAny(0)))
					{
						continue;
					}
				}
				else if (!mana.CanAfford(cost))
				{
					continue;
				}
				if (colorLimit && config.Colors.Any((ManaColor c) => mana.GetValue(c) == 0))
				{
					continue;
				}
			}
			float num = weightTable.WeightFor(config, Player.Id, hashSet);
			if (num > 0f)
			{
				float num2 = BaseCardWeight(config, applyFactors);
				uniqueRandomPool.Add(item, num * num2);
			}
		}
		return uniqueRandomPool;
	}

	internal Card[] GetRewardCards(CardWeightTable playerWeightTable, CardWeightTable friendWeightTable, CardWeightTable neutralWeightTable, CardWeightTable randomWeightTable, int count, bool isBossReward = false)
	{
		if (_cardRewardDecreaseRepeatRare)
		{
			playerWeightTable = ModifyRare(playerWeightTable, 0.01f);
			friendWeightTable = ModifyRare(friendWeightTable, 0.01f);
			neutralWeightTable = ModifyRare(neutralWeightTable, 0.01f);
			randomWeightTable = ModifyRare(randomWeightTable, 0.01f);
			_cardRewardDecreaseRepeatRare = false;
		}
		List<Card> cards = new List<Card>();
		if (count > 0)
		{
			Card[] collection = RollCards(CardRng, playerWeightTable, 1, BaseMana, RewardAndShopCardColorLimitFlag == 0, applyFactors: true);
			cards.AddRange(collection);
		}
		if (count > 1)
		{
			if (cards.Any((Card c) => c.Config.Rarity == Rarity.Rare && !isBossReward))
			{
				friendWeightTable = ModifyRare(friendWeightTable, 0f);
			}
			Card[] collection2 = RollCards(CardRng, friendWeightTable, 1, BaseMana, RewardAndShopCardColorLimitFlag == 0, applyFactors: true, battleRolling: false, ensureCount: false, (CardConfig config) => config.Id != cards[0].Id);
			cards.AddRange(collection2);
		}
		if (count > 2)
		{
			if (cards.Any((Card c) => c.Config.Rarity == Rarity.Rare && !isBossReward))
			{
				neutralWeightTable = ModifyRare(neutralWeightTable, 0f);
			}
			Card[] collection3 = RollCards(CardRng, neutralWeightTable, 1, BaseMana, RewardAndShopCardColorLimitFlag == 0, applyFactors: true);
			cards.AddRange(collection3);
		}
		if (count > 3)
		{
			if (cards.Any((Card c) => c.Config.Rarity == Rarity.Rare && !isBossReward))
			{
				randomWeightTable = ModifyRare(randomWeightTable, 0f);
			}
			Card[] collection4 = RollCards(CardRng, randomWeightTable, count - 3, BaseMana, RewardAndShopCardColorLimitFlag == 0, applyFactors: true, battleRolling: false, ensureCount: false, (CardConfig config) => cards.All((Card c) => c.Id != config.Id));
			cards.AddRange(collection4);
		}
		if (cards.Any((Card c) => c.Config.Rarity == Rarity.Rare))
		{
			_cardRewardDecreaseRepeatRare = true;
			_cardRareWeightFactor = 0.85f;
		}
		else
		{
			_cardRareWeightFactor += 0.01f;
		}
		foreach (Card item in cards)
		{
			Rarity rarity = item.Config.Rarity;
			if ((rarity == Rarity.Common || rarity == Rarity.Uncommon) && item.CanUpgrade && CardRng.NextFloat(0f, 1f) < CardUpgradedChance)
			{
				item.Upgrade();
			}
		}
		foreach (Card item2 in cards)
		{
			UpgradeNewDeckCardOnFlags(item2);
		}
		return cards.ToArray();
		static CardWeightTable ModifyRare(CardWeightTable table, float rareMultiplier)
		{
			RarityWeightTable rarityTable = table.RarityTable;
			float rare = rarityTable.Rare;
			return table.WithRarity(rarityTable.WithRare(rare * rareMultiplier));
		}
	}

	public void Damage(int damage, DamageType damageType, bool isSelf, bool triggerVisual = false, Adventure fromAdventure = null)
	{
		DamageEventArgs damageEventArgs = new DamageEventArgs
		{
			Target = Player,
			DamageInfo = new DamageInfo(damage, damageType)
		};
		Player.DamageReceiving.Execute(damageEventArgs);
		Player.DamageTaking.Execute(damageEventArgs);
		if (damageEventArgs.IsCanceled)
		{
			return;
		}
		Player.TakeDamage(damageEventArgs.DamageInfo);
		if (Player.Hp == 0)
		{
			Player.Status = UnitStatus.Dying;
			DieEventArgs dieEventArgs = new DieEventArgs
			{
				Unit = Player,
				DieCause = DieCause.GameRun
			};
			Player.Dying.Execute(dieEventArgs);
			if (!dieEventArgs.IsCanceled)
			{
				Player.Status = UnitStatus.Dead;
				if (isSelf)
				{
					Stats.PlayerSuicide = true;
				}
				Status = GameRunStatus.Failure;
				GenerateRecords(isAbandon: false, null, fromAdventure);
			}
			else
			{
				Player.Status = UnitStatus.Alive;
			}
		}
		Player.DamageReceived.Execute(damageEventArgs);
		VisualTrigger?.OnDamage(damageEventArgs.DamageInfo, triggerVisual);
	}

	public void Heal(int amount, bool triggerVisual = true, string audioName = null)
	{
		HealEventArgs healEventArgs = new HealEventArgs
		{
			Amount = amount,
			Target = Player,
			Cause = ActionCause.Gap,
			CanCancel = true
		};
		Player.HealingReceiving.Execute(healEventArgs);
		if (!healEventArgs.IsCanceled)
		{
			healEventArgs.Amount = healEventArgs.Amount.Round();
			int num = Player.Heal(healEventArgs.Amount.ToInt());

	public void UpgradeRandomCards(int amount = 1, CardType? type = null)
	{
		if (amount <= 0)
		{
			throw new InvalidOperationException("随机升级牌数量为0或负数。");
		}
		List<Card> list = new List<Card>();
		foreach (Card item in _baseDeck)
		{
			if (!item.CanUpgradeAndPositive)
			{
				continue;
			}
			if (type.HasValue)
			{
				if (item.CardType == type.Value)
				{
					list.Add(item);
				}
			}
			else
			{
				list.Add(item);
			}
		}
		Card[] cards = list.SampleManyOrAll(amount, GameRunEventRng);
		UpgradeDeckCards(cards);
	}

	public IEnumerator GainExhibitRunner(Exhibit exhibit, bool triggerVisual = false, [MaybeNull] VisualSourceData exhibitSource = null)
	{
		if (exhibit.IsSentinel)
		{
			Debug.LogError("Cannot gain sentinel exhibit " + exhibit.DebugName);
			yield break;
		}
		if (Player.HasExhibit(exhibit.GetType()))
		{
			throw new ArgumentException("Cannot add duplicated Exhibit.", "exhibit");
		}
		exhibit.GameRun = this;
		Player.AddExhibit(exhibit);
		ExhibitRecord.Add(exhibit.Id);
		yield return VisualTrigger?.OnGainExhibit(exhibit, triggerVisual, exhibitSource);
		yield return exhibit.TriggerGain(Player);
		this.ExhibitRevealed?.Invoke(exhibit);
		Rarity rarity = exhibit.Config.Rarity;
		if (rarity != Rarity.Mythic && rarity != Rarity.Shining)
		{
			Stats.NoExhibitFlag = false;
		}
	}

	}

	internal void AcquireCardReward(StationReward reward, Card card, int index)
	{
		if (LootCardCommonDupeCount > 0 && card.Config.Rarity == Rarity.Common)
		{
			List<Card> list = new List<Card> { card };
			list.AddRange(card.Clone(LootCardCommonDupeCount));
			AddDeckCards(list, triggerVisual: true, new VisualSourceData
			{
				SourceType = VisualSourceType.Reward,
				Index = index
			});
		}
		else if (LootCardUncommonDupeCount > 0 && card.Config.Rarity == Rarity.Uncommon)
		{
			List<Card> list2 = new List<Card> { card };
			list2.AddRange(card.Clone(LootCardUncommonDupeCount));
			AddDeckCards(list2, triggerVisual: true, new VisualSourceData
			{
				SourceType = VisualSourceType.Reward,
				Index = index
			});
		}
		else
		{
			AddDeckCards(new Card[1] { card }, triggerVisual: true, new VisualSourceData
			{
				SourceType = VisualSourceType.Reward,
				Index = index
			});
		}
		foreach (Card item in reward.Cards.Where((Card c) => c != card))
		{
			if (!_cardRewardWeightFactors.TryGetValue(item.Id, out var value))
			{
				value = 1f;
			}
			_cardRewardWeightFactors[item.Id] = value * 0.9f;
		}
	}

	public void AbandonReward(StationReward reward)
	{
		GainMoney(RewardCardAbandonMoney, triggerVisual: true, new VisualSourceData
		{
			SourceType = VisualSourceType.AbandonReward
		});
		if (reward.Type == StationRewardType.Card)
		{
			foreach (Card card in reward.Cards)
			{
				if (!_cardRewardWeightFactors.TryGetValue(card.Id, out var value))
				{
					value = 1f;
				}
				_cardRewardWeightFactors[card.Id] = value * 0.85f;
			}
		}
		RewardAbandoned.Execute(new GameEventArgs
		{
			CanCancel = false
		});
	}

	private void GenerateRecords(bool isAbandon, EnemyGroup failingEnemyGroup = null, Adventure failingAdventure = null, bool addCurrentStationRecord = true)
	{
		if (GameRunRecord != null)
		{
			Debug.LogError("GenerateRecords multiple times");
			return;
		}
		if (!isAbandon && addCurrentStationRecord)
		{
			if (CurrentStation == null)
			{
				Debug.LogError("GenerateRecords without current station entered.");
			}
			else
			{
				StageRecord stageRecord = StageRecords.LastOrDefault();
				if (stageRecord == null)
				{
					Debug.LogError("GenerateRecords while stage records is empty.");
				}
				else
				{
					stageRecord.Stations.Add(CurrentStation.GenerateRecord());
				}
			}
		}
		GameResultType resultType = Status switch
		{
			GameRunStatus.Running => throw new InvalidOperationException("Still in game-run"), 
			GameRunStatus.NormalEnd => GameResultType.NormalEnd, 
			GameRunStatus.TrueEnd => GameResultType.TrueEnd, 
			GameRunStatus.Failure => IsNormalEndFinished ? GameResultType.TrueEndFail : GameResultType.Failure, 
			_ => throw new ArgumentOutOfRangeException(), 
		};
		GameRunRecord = new GameRunRecordSaveData
		{
			Player = Player.Id,
			PlayerType = PlayerType,
			Us = Player.Us.Id,
			Mode = Mode,
			Difficulty = Difficulty,
			Puzzles = Puzzles,
			Seed = RootSeed,
			ResultType = resultType,
			FailingEnemyGroup = failingEnemyGroup?.Id,
			FailingAdventure = failingAdventure?.Id,
			Stages = StageRecords.Select((StageRecord r) => r.Clone()).ToArray(),
			MaxHp = Player.MaxHp,
			TotalMoney = TotalMoney,
			BaseMana = BaseMana.ToString(),
			Cards = _baseDeck.Select((Card c) => new CardRecordSaveData
			{
				Id = c.Id,
				Upgraded = c.IsUpgraded,
				UpgradeCounter = c.UpgradeCounter
			}).ToArray(),
			Exhibits = ExhibitRecord.ToArray(),
			JadeBoxes = _jadeBoxes.Select((JadeBox j) => j.Id).ToArray(),
			IsAutoSeed = IsAutoSeed,
			ShowRandomResult = ShowRandomResult,
			ReloadTimes = ReloadTimes,
			GameVersion = VersionInfo.Current.Version,
			Packs = Packs.ToArray()
		};
	}

	public GameRunSaveData Save()
	{
		GameRunSaveData gameRunSaveData = new GameRunSaveData
		{
			Status = Status,
			IsNormalEndFinished = IsNormalEndFinished,
			Mode = Mode,
			Difficulty = Difficulty,
			Puzzles = Puzzles,
			RootSeed = RootSeed,
			IsAutoSeed = IsAutoSeed,
			RootRng = RootRng.State,
			StationRng = StationRng.State,
			InitBossSeed = InitBossSeed,
			ShopRng = ShopRng.State,
			AdventureRng = AdventureRng.State,
			ExhibitRng = ExhibitRng.State,
			ShiningExhibitRng = ShiningExhibitRng.State,
			CardRng = CardRng.State,
			GamerunEventRng = GameRunEventRng.State,
			BattleRng = BattleRng.State,
			BattleCardRng = BattleCardRng.State,
			BattleTransformCardRng = BattleTransformCardRng.State,
			ShuffleRng = ShuffleRng.State,
			EnemyMoveRng = EnemyMoveRng.State,
			EnemyBattleRng = EnemyBattleRng.State,
			DebutRng = DebutRng.State,
			FinalBossSeed = FinalBossSeed,
			UISeed = UISeed,
			HasClearBonus = HasClearBonus,
			UnlockLevel = UnlockLevel,
			FinalBossInitialDamage = FinalBossInitialDamage,
			ExtraExhibitReward = ExtraExhibitReward?.Id,
			ReloadTimes = ReloadTimes,
			ShowRandomResult = ShowRandomResult,
			Packs = Packs
		};
		foreach (Stage stage in _stages)
		{
			gameRunSaveData.Stages.Add(new StageSaveData
			{
				Name = stage.GetType().Name,
				Index = stage.Index,
				MapSeed = stage.MapSeed,
				Level = stage.Level,
				SelectedBoss = stage.SelectedBoss,
				DebutAdventure = stage.DebutAdventureType?.Name,
				IsNormalFinalStage = stage.IsNormalFinalStage,
				IsTrueEndFinalStage = stage.IsTrueEndFinalStage,
				AdventurePool = stage.AdventurePool.Save((Type t) => t.Name),
				EnemyPoolAct1 = stage.EnemyPoolAct1.Save((string s) => s),
				EnemyPoolAct2 = stage.EnemyPoolAct2.Save((string s) => s),
				EnemyPoolAct3 = stage.EnemyPoolAct3.Save((string s) => s),
				EliteEnemyPool = stage.EliteEnemyPool.Save((string s) => s),
				AdventureHistory = stage.AdventureHistory.Select((Type a) => a.Name).ToList(),
				ExtraFlags = stage.ExtraFlags.ToList()
			});
		}
		gameRunSaveData.StageIndex = CurrentStage?.Index;
		_ = CurrentStation;
		foreach (MapNode item in CurrentMap.Path)
		{
			gameRunSaveData.Path.Add(new MapNodeSaveData
			{
				X = item.X,
				Y = item.Y
			});
		}
		gameRunSaveData.Player = new PlayerSaveData
		{
			Name = Player.Id,
			Us = Player.Us?.Id,
			Hp = Player.Hp,
			MaxHp = Player.MaxHp,
			Power = Player.Power
		};
		gameRunSaveData.PlayerType = PlayerType;
		gameRunSaveData.Mana = BaseMana.ToString();
		foreach (Card item2 in _baseDeck)
		{
			gameRunSaveData.Deck.Add(new CardSaveData
			{
				Name = item2.Id,
				InstanceId = item2.InstanceId,
				IsUpgraded = item2.IsUpgraded,
				DeckCounter = item2.DeckCounter,
				UpgradeCounter = item2.UpgradeCounter
			});
		}
		gameRunSaveData.DeckCardInstanceId = _deckCardInstanceId;
		gameRunSaveData.Money = Money;
		gameRunSaveData.TotalMoney = TotalMoney;
		gameRunSaveData.ShopRemoveCardCounter = ShopRemoveCardCounter;
		gameRunSaveData.UltimateUseCount = UltimateUseCount;
		foreach (Exhibit exhibit in Player.Exhibits)
		{
			gameRunSaveData.Exhibits.Add(new ExhibitSaveData
			{
				Name = exhibit.Id,
				Counter = (exhibit.HasCounter ? new int?(exhibit.Counter) : null),
				CardInstanceId = exhibit.CardInstanceId
			});
		}
		foreach (JadeBox jadeBox in _jadeBoxes)
		{
			gameRunSaveData.JadeBoxes.Add(new JadeBoxSaveData
			{
				Name = jadeBox.Id
			});
		}
		gameRunSaveData.CardRareWeightFactor = _cardRareWeightFactor;
		gameRunSaveData.CardRewardWeightFactors = _cardRewardWeightFactors.Select((KeyValuePair<string, float> kv) => new CardWeightFactorSaveData
		{
			Id = kv.Key,
			Value = kv.Value
		}).ToList();
		gameRunSaveData.CardRewardDecreaseRepeatRare = _cardRewardDecreaseRepeatRare;
		gameRunSaveData.ShiningExhibitPool = ShiningExhibitPool.Select((Type e) => e.Name).ToList();
		gameRunSaveData.ExhibitPool = ExhibitPool.Select((Type e) => e.Name).ToList();
		gameRunSaveData.ExhibitRecord = ExhibitRecord.ToList();
		gameRunSaveData.AdventureHistory = AdventureHistory.Select((Type t) => t.Name).ToList();
		gameRunSaveData.PlayedSeconds = PlayedSeconds;
		gameRunSaveData.Stats = Stats.Clone();
		gameRunSaveData.ExtraFlags = ExtraFlags;
		gameRunSaveData.StageRecords = StageRecords.Select((StageRecord r) => r.Clone()).ToList();
		return gameRunSaveData;
	}

			StationRng = RandomGen.FromState(data.StationRng),
			InitBossSeed = data.InitBossSeed,
			ShopRng = RandomGen.FromState(data.ShopRng),
			AdventureRng = RandomGen.FromState(data.AdventureRng),
			ExhibitRng = RandomGen.FromState(data.ExhibitRng),
			ShiningExhibitRng = RandomGen.FromState(data.ShiningExhibitRng),
			CardRng = RandomGen.FromState(data.CardRng),
			GameRunEventRng = RandomGen.FromState(data.GamerunEventRng),
			BattleRng = RandomGen.FromState(data.BattleRng),
			BattleCardRng = RandomGen.FromState(data.BattleCardRng),
			BattleTransformCardRng = RandomGen.FromState(data.BattleTransformCardRng),
			ShuffleRng = RandomGen.FromState(data.ShuffleRng),
			EnemyMoveRng = RandomGen.FromState(data.EnemyMoveRng),
			EnemyBattleRng = RandomGen.FromState(data.EnemyBattleRng),
			DebutRng = RandomGen.FromState(data.DebutRng),
			FinalBossSeed = data.FinalBossSeed,
			UISeed = data.UISeed,
			HasClearBonus = data.HasClearBonus,
			UnlockLevel = data.UnlockLevel,
			FinalBossInitialDamage = data.FinalBossInitialDamage,
			ExtraExhibitReward = ((data.ExtraExhibitReward != null) ? Library.TryCreateExhibit(data.ExtraExhibitReward) : null),
			ReloadTimes = data.ReloadTimes,
			ShowRandomResult = data.ShowRandomResult,
			Packs = data.Packs
		};
		gameRunController._stages = new List<Stage>();
		foreach (StageSaveData stage5 in data.Stages)
		{
			Stage stage = Library.CreateStage(stage5.Name);
			stage.Index = stage5.Index;
			stage.MapSeed = stage5.MapSeed;
			stage.Level = stage5.Level;
			if (stage5.SelectedBoss != null)
			{
				stage.SetBoss(stage5.SelectedBoss);
			}
			if (stage5.DebutAdventure != null)
			{
				stage.DebutAdventureType = TypeFactory<Adventure>.GetType(stage5.DebutAdventure);
			}
			stage.IsNormalFinalStage = stage5.IsNormalFinalStage;
			stage.IsTrueEndFinalStage = stage5.IsTrueEndFinalStage;
			stage.AdventurePool = UniqueRandomPool<Type>.Restore(stage5.AdventurePool, GameEntityTypeConverter<Adventure>);
			stage.EnemyPoolAct1 = UniqueRandomPool<string>.Restore(stage5.EnemyPoolAct1, EnemyGroupConverter);
			stage.EnemyPoolAct2 = UniqueRandomPool<string>.Restore(stage5.EnemyPoolAct2, EnemyGroupConverter);
			stage.EnemyPoolAct3 = UniqueRandomPool<string>.Restore(stage5.EnemyPoolAct3, EnemyGroupConverter);
			stage.EliteEnemyPool = UniqueRandomPool<string>.Restore(stage5.EliteEnemyPool, EnemyGroupConverter);
			foreach (string item in stage5.AdventureHistory)
			{
				Type type = TypeFactory<Adventure>.TryGetType(item);
				if ((object)type != null)
				{
					stage.AdventureHistory.Add(type);
				}
			}
			stage.ExtraFlags = stage5.ExtraFlags.ToHashSet();
			stage.GameRun = gameRunController;
			gameRunController._stages.Add(stage);
		}
		RandomGen randomGen = new RandomGen(gameRunController.InitBossSeed);
		foreach (Stage stage6 in gameRunController._stages)
		{
			stage6.InitBoss(randomGen);
		}
		foreach (Stage stage7 in gameRunController._stages)
		{
			stage7.InitFirstAdventure(randomGen);
		}
		int? stageIndex = data.StageIndex;
		if (stageIndex.HasValue)
		{
			int valueOrDefault = stageIndex.GetValueOrDefault();
			gameRunController._stageIndex = valueOrDefault;
			Stage stage3 = (gameRunController.CurrentStage = gameRunController._stages[valueOrDefault]);
			Stage stage4 = stage3;
			GameMap gameMap2 = (gameRunController.CurrentMap = stage4.CreateMap());
			GameMap map = gameMap2;
			if (data.Path != null)
			{
				map.RestorePath(data.Path.Select((MapNodeSaveData xy) => map.Nodes[xy.X, xy.Y]));
			}
		}
		PlayerUnit playerUnit = Library.CreatePlayerUnit(data.Player.Name);
		if (data.Player.Us != null)
		{
			playerUnit.SetUs(Library.CreateUs(data.Player.Us));
		}
		playerUnit.SetMaxHp(data.Player.Hp, data.Player.MaxHp);
		playerUnit.Power = data.Player.Power;
		gameRunController.Player = playerUnit;
		gameRunController.Player.Us.GameRun = gameRunController;
		gameRunController.PlayerType = data.PlayerType;
		gameRunController.BaseMana = ManaGroup.Parse(data.Mana);
		gameRunController._baseDeck = new List<Card>();
		foreach (CardSaveData item2 in data.Deck)
		{
			Card card = Library.CreateCard(item2.Name);
			card.GameRun = gameRunController;
			card.InstanceId = item2.InstanceId;
			if (item2.IsUpgraded)
			{
				card.Upgrade();
			}
			card.DeckCounter = item2.DeckCounter;
			card.UpgradeCounter = item2.UpgradeCounter;
			gameRunController._baseDeck.Add(card);
		}
		gameRunController._deckCardInstanceId = data.DeckCardInstanceId;
		gameRunController.Money = data.Money;
		gameRunController.TotalMoney = data.TotalMoney;
		gameRunController.ShopRemoveCardCounter = data.ShopRemoveCardCounter;
		gameRunController.UltimateUseCount = data.UltimateUseCount;
		gameRunController._cardRareWeightFactor = data.CardRareWeightFactor;
		gameRunController._cardRewardWeightFactors = new Dictionary<string, float>();
		foreach (CardWeightFactorSaveData cardRewardWeightFactor in data.CardRewardWeightFactors)
		{
			gameRunController._cardRewardWeightFactors.TryAdd(cardRewardWeightFactor.Id, cardRewardWeightFactor.Value);
		}
		gameRunController._cardRewardDecreaseRepeatRare = data.CardRewardDecreaseRepeatRare;
		gameRunController.ShiningExhibitPool = data.ShiningExhibitPool.Select(TypeFactory<Exhibit>.GetType).ToList();
		gameRunController.ExhibitPool = data.ExhibitPool.Select(TypeFactory<Exhibit>.GetType).ToList();
		gameRunController.ExhibitRecord.AddRange(data.ExhibitRecord);
		foreach (string item3 in data.AdventureHistory)
		{
			Type type2 = TypeFactory<Adventure>.TryGetType(item3);
			if ((object)type2 != null)
			{
				gameRunController.AdventureHistory.Add(type2);
			}
		}
		gameRunController.Player.EnterGameRun(gameRunController);
		foreach (ExhibitSaveData exhibit2 in data.Exhibits)
		{
			Exhibit exhibit = Library.CreateExhibit(exhibit2.Name);
			if (exhibit.HasCounter)
			{
				exhibit.Counter = exhibit2.Counter.Value;
			}
			exhibit.CardInstanceId = exhibit2.CardInstanceId;
			exhibit.GameRun = gameRunController;
			gameRunController.Player.AddExhibit(exhibit);
		}
		foreach (JadeBoxSaveData jadeBox2 in data.JadeBoxes)
		{
			JadeBox jadeBox = Library.CreateJadeBox(jadeBox2.Name);
			jadeBox.GameRun = gameRunController;
			gameRunController._jadeBoxes.Add(jadeBox);
			jadeBox.TriggerAdded();
		}
		if (data.Timing == SaveTiming.EnterMapNode)
		{
			if (data.EnteringNode == null)
			{
				throw new InvalidDataException($"Entering node is null while SaveTiming = {data.Timing}");
			}
			gameRunController.CurrentMap.SetAdjacentNodesStatus(gameRunController.MapMode);
		}
		else
		{
			SaveTiming timing = data.Timing;
			if (timing == SaveTiming.BattleFinish || timing == SaveTiming.AfterBossReward)
			{
				StationType type3 = data.EnteredStationType ?? throw new InvalidDataException($"Entered station type is null while SaveTiming = {data.Timing}");
				if (data.Path == null || data.Path.Count == 0)
				{
					throw new InvalidDataException($"Entered station with no path while SaveTiming = {data.Timing}");
				}
				MapNodeSaveData mapNodeSaveData = data.Path.Last();
				MapNode node = gameRunController.CurrentMap.Nodes[mapNodeSaveData.X, mapNodeSaveData.Y];
				if (!(gameRunController.CurrentStage.CreateStationFromType(node, type3) is BattleStation battleStation))
				{
					throw new InvalidDataException($"Entered non-battle station while SaveTiming = {data.Timing}");
				}
				string battleStationEnemyGroup = data.BattleStationEnemyGroup;
				if (battleStationEnemyGroup == null)
				{
					throw new InvalidDataException($"Entered battle station with out enemy group while SaveTiming = {data.Timing}");
				}
				battleStation.EnemyGroupEntry = Library.TryGetEnemyGroupEntry(battleStationEnemyGroup);
				if (battleStation.EnemyGroupEntry == null)
				{
					throw new InvalidDataException($"Entered battle station enemy group {battleStationEnemyGroup} is not found while SaveTiming = {data.Timing}");
				}
				battleStation.RewardsReplaceWithMoney = data.RewardsReplaceWithMoney;
				gameRunController.CurrentStation = battleStation;
				battleStation.ForceFinish();
				gameRunController.FinishStation(battleStation);
			}
			else if (data.Timing == SaveTiming.Adventure)
			{
				StationType type4 = data.EnteredStationType ?? throw new InvalidDataException($"Entered station type is null while SaveTiming = {data.Timing}");
				MapNodeSaveData mapNodeSaveData2 = data.Path.Last();
				MapNode node2 = gameRunController.CurrentMap.Nodes[mapNodeSaveData2.X, mapNodeSaveData2.Y];
				Station station = gameRunController.CurrentStage.CreateStationFromType(node2, type4);
				Adventure adventure = Library.CreateAdventure(data.AdventureState.AdventureId);
				if (!(station is IAdventureStation adventureStation))
				{
					throw new InvalidDataException($"Entered non-adventure station while SaveTiming = {data.Timing}");
				}
				adventureStation.Restore(adventure);
				gameRunController.CurrentStation = station;
			}
		}
		gameRunController.PlayedSeconds = data.PlayedSeconds;
		gameRunController.Stats = data.Stats.Clone();
		gameRunController.ExtraFlags = data.ExtraFlags.ToHashSet();
		gameRunController.StageRecords = data.StageRecords.Select((StageRecord r) => r.Clone()).ToList();
		return gameRunController;
		static string EnemyGroupConverter(string s)
		{
			if (Library.TryGetEnemyGroupEntry(s) == null)
			{
				Debug.LogWarning("Cannot find EnemyGroup '" + s + "'");
			}
			return s;
		}
		static Type GameEntityTypeConverter<T>(string s) where T : class
		{
			Type type5 = TypeFactory<T>.TryGetType(s);
			if ((object)type5 == null)
			{
				Debug.LogWarning("Cannot find " + typeof(T).Name + " type '" + s + "'");
			}
			return type5;
		}
	}
