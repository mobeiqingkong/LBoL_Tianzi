using System.Collections.Generic;
using JetBrains.Annotations;
using LBoL.Base;
using LBoL.Core;
using LBoL.Core.Battle;
using LBoL.Core.Battle.BattleActions;
using LBoL.Core.StatusEffects;
using LBoL.Core.Units;

namespace LBoL.EntityLib.StatusEffects.Basic;

[UsedImplicitly]
public sealed class TempElectric : StatusEffect
{
	public override string UnitEffectName => "ElectricLoop";

	protected override void OnAdded(Unit unit)
	{
		ReactOwnerEvent(base.Owner.DamageReceived, OnDamageReceived);
		ReactOwnerEvent(base.Owner.TurnStarted, OnOwnerTurnStarted);
	}

	private IEnumerable<BattleAction> OnDamageReceived(DamageEventArgs args)
	{
		if (args.Source != base.Owner && args.Source.IsAlive && args.DamageInfo.DamageType == DamageType.Attack && args.DamageInfo.Amount > 0f)
		{
			NotifyActivating();
			yield return new DamageAction(base.Owner, args.Source, DamageInfo.Reaction(base.Level), "电击");
		}
	}

	private IEnumerable<BattleAction> OnOwnerTurnStarted(UnitEventArgs args)
	{
		if (!base.Owner.IsExtraTurn)
		{
			yield return new RemoveStatusEffectAction(this);
		}
	}
}
