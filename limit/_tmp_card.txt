	internal sealed class CardFormatWrapper : GameEntityFormatWrapper
	{
		private readonly Card _card;

		public CardFormatWrapper(Card card)
			: base(card)
		{
			_card = card;
		}

		protected override string FormatArgument(object arg, string format)
		{
			if (!(arg is DamageInfo damageInfo))
			{
				if (!(arg is BlockInfo { Block: var block }))
				{
					if (!(arg is ShieldInfo { Shield: var shield }))
					{
						if (!(arg is ScryInfo { Count: var count }))
						{
							if (arg is FriendCostInfo friendCostInfo)
							{
								return "<indent=0><size=34><sprite=\"" + friendCostInfo.CostType.ToString() + "\" name=\"" + friendCostInfo.Cost + "\"></size><indent=80>";
							}
							return base.FormatArgument(arg, format);
						}
						BattleController battle = _card.Battle;
						if (battle == null)
						{
							return GameEntityFormatWrapper.WrappedFormatNumber(count, count, format);
						}
						int value = battle.CalculateScry(_card, count);
						return GameEntityFormatWrapper.WrappedFormatNumber(count, value, format);
					}
					BattleController battle2 = _card.Battle;
					if (battle2 == null)
					{
						return GameEntityFormatWrapper.WrappedFormatNumber(shield, shield, format);
					}
					int item = battle2.CalculateBlockShield(_card, 0f, shield).shield;
					return GameEntityFormatWrapper.WrappedFormatNumber(shield, item, format);
				}
				BattleController battle3 = _card.Battle;
				if (battle3 == null)
				{
					return GameEntityFormatWrapper.WrappedFormatNumber(block, block, format);
				}
				int item2 = battle3.CalculateBlockShield(_card, block, 0f).block;
				return GameEntityFormatWrapper.WrappedFormatNumber(block, item2, format);
			}
			int num = (int)damageInfo.Damage;
			BattleController battle4 = _card.Battle;
			if (battle4 == null)
			{
				return GameEntityFormatWrapper.WrappedFormatNumber(num, num, format);
			}
			int value2 = battle4.CalculateDamage(_card, battle4.Player, _card.PendingTarget, damageInfo);
			return GameEntityFormatWrapper.WrappedFormatNumber(num, value2, format);
		}
	}

	public ManaGroup Cost
	{
		get
		{
			if (IsXCost)
			{
				return BaseCost;
			}
			if (FreeCost || Summoned)
			{
				return ManaGroup.Empty;
			}
			if (IsForceCost)
			{
				return ForceCost;
			}
			ManaGroup corrected = (TurnCost + AdditionalCost + AuraCost).Corrected;
			corrected = (IsPurified ? corrected.Purified() : corrected);
			BattleController battle = Battle;
			if (battle != null && battle.ManaFreezeLevel > 0 && Zone == CardZone.Hand)
			{
				return corrected + ManaGroup.Anys(Battle.ManaFreezeLevel);
			}
			return corrected;
		}
	}

	public virtual bool Triggered
	{
		get
		{
			if (IsDebut && DebutActive)
			{
				return true;
			}
			if (IsInstinct && InstinctActive)
			{
				return true;
			}
			if (IsOverdrive && Overdrive(Value2))
			{
				return true;
			}
			if (CardType == CardType.Friend && Config.UltimateCost.HasValue && Loyalty >= -UltimateCost)
			{
				return true;
			}
			return false;
		}
	}

	public virtual IEnumerable<BattleAction> AfterUseAction()
	{
		if (RemoveFromBattleAfterPlay)
		{
			yield return EchoCloneAction();
			yield return new RemoveCardAction(this);
			yield break;
		}
		switch (CardType)
		{
		case CardType.Tool:
		{
			if (!DeckCounter.HasValue)
			{
				Debug.LogError(DebugName + " is tool card but has no DeckCounter");
				break;
			}
			Card deckCardByInstanceId = Battle.GameRun.GetDeckCardByInstanceId(InstanceId);
			if (--DeckCounter == 0)
			{
				if (deckCardByInstanceId != null)
				{
					Battle.GameRun.RemoveDeckCard(deckCardByInstanceId, triggerVisual: false);
				}
				yield return new RemoveCardAction(this);
				yield break;
			}
			if (deckCardByInstanceId != null)
			{
				deckCardByInstanceId.DeckCounter = DeckCounter;
			}
			break;
		}
		case CardType.Ability:
			yield return EchoCloneAction();
			yield return new RemoveCardAction(this);
			yield break;
		case CardType.Friend:
			if (Summoning)
			{
				IsEthereal = false;
				Battle.PlayerSummonAFriendThisTurn = true;
				if (Zone == CardZone.PlayArea)
				{
					if (Battle.HandZone.Count < Battle.MaxHand)
					{
						yield return new MoveCardAction(this, CardZone.Hand);
					}
					else
					{
						yield return new MoveCardAction(this, CardZone.Discard);
					}
				}
			}
			else if (UltimateUsed || Loyalty == 0)
			{
				yield return EchoCloneAction();
				yield return new RemoveCardAction(this);
				yield break;
			}
			break;
		}
		yield return EchoCloneAction();
		if (IsExile || IsTempExile)
		{
			IsTempExile = false;
			yield return new ExileCardAction(this);
		}
		if (Zone == CardZone.PlayArea)
		{
			yield return new MoveCardAction(this, CardZone.Discard);
		}
		int playCount = PlayCount + 1;
		PlayCount = playCount;
		if (IsDebut)
		{
			DebutCardPlayedOnce = true;
		}
	}

	public virtual void Upgrade()
	{
		if (!Config.IsUpgradable)
		{
			throw new InvalidOperationException(DebugName + " is not upgradable");
		}
		if (IsUpgraded)
		{
			throw new InvalidOperationException(DebugName + " already upgraded");
		}
		IsUpgraded = true;
		ProcessKeywordUpgrade();
		CostChangeInUpgrading();
		if (Config.Loyalty.HasValue && Config.UpgradedLoyalty.HasValue)
		{
			Loyalty += Config.UpgradedLoyalty.Value - Config.Loyalty.Value;
		}
		NotifyChanged();
	}

	protected void ProcessKeywordUpgrade()
	{
		Keyword keyword = Keywords & ~Config.Keywords;
		Keywords = Config.UpgradedKeywords | keyword;
	}

	protected void CostChangeInUpgrading()
	{
		if (IsXCost || !Config.UpgradedCost.HasValue || Config.UpgradedCost == Config.Cost)
		{
			return;
		}
		ManaGroup cost = Config.Cost;
		ManaGroup value = Config.UpgradedCost.Value;
		ManaGroup cost2 = cost - value;
		DecreaseBaseCost(cost2);
		if (cost2.Hybrid < 0)
		{
			List<ManaColor> getHybridColors = cost2.GetHybridColors;
			foreach (ManaColor item in getHybridColors)
			{
				if (BaseCost.GetValue(item) < 0 && BaseCost.Hybrid > 0)
				{
					int num = Math.Min(-BaseCost.GetValue(item), BaseCost.Hybrid);
					IncreaseBaseCost(ManaGroup.FromColor(item, num));
					DecreaseBaseCost(ManaGroup.Hybrids(num, cost2.HybridColor));
				}
			}
			foreach (ManaColor item2 in getHybridColors)
			{
				if (TurnCostDelta.GetValue(item2) < 0 && (TurnCostDelta.Hybrid > 0 || TurnCost.Hybrid > 0))
				{
					int num2 = Math.Min(-TurnCostDelta.GetValue(item2), BaseCost.Hybrid);
					IncreaseTurnCost(ManaGroup.FromColor(item2, num2));
					DecreaseTurnCost(ManaGroup.Hybrids(num2, cost2.HybridColor));
				}
			}
		}
		if (cost2.Any >= 0 && !BaseCost.IsInvalid && !TurnCost.IsInvalid)
		{
			return;
		}
		foreach (ManaColor item3 in ManaColors.SingleColorsWithHybrid)
		{
			if (BaseCost.GetValue(item3) < 0 && BaseCost.Any > 0)
			{
				int num3 = Math.Min(-BaseCost.GetValue(item3), BaseCost.Any);
				IncreaseBaseCost(ManaGroup.FromColor(item3, num3));
				DecreaseBaseCost(ManaGroup.Anys(num3));
			}
		}
		foreach (ManaColor item4 in ManaColors.SingleColorsWithHybrid)
		{
			if (TurnCostDelta.GetValue(item4) < 0 && (TurnCostDelta.Any > 0 || TurnCost.Any > 0))
			{
				int num4 = Math.Min(-TurnCostDelta.GetValue(item4), BaseCost.Any);
				IncreaseTurnCost(ManaGroup.FromColor(item4, num4));
				DecreaseTurnCost(ManaGroup.Anys(num4));
			}
		}
	}

	internal void EnterBattle(BattleController battle)
	{
		if (Battle != null)
		{
			throw new InvalidOperationException("Cannot enter battle while already in battle");
		}
		Battle = battle;
		if (CardType == CardType.Friend && base.GameRun.ExtraLoyalty > 0)
		{
			Loyalty += base.GameRun.ExtraLoyalty;
		}
		if (IsForbidden && base.GameRun.RemoveBadCardForbidden > 0)
		{
			CardType cardType = CardType;
			if (cardType == CardType.Misfortune || cardType == CardType.Status)
			{
				IsForbidden = false;
				IsExile = true;
			}
		}
		OnEnterBattle(battle);
	}

	{
		if (Battle == null)
		{

		}

		{
			AuraCost = ManaGroup.Empty;
		}
		OnLeaveBattle();
		_handlerHolder.ClearEventHandlers();
		Battle = null;
	}

	public virtual Interaction Precondition()
	{
		if (CardType == CardType.Friend && Summoned && Loyalty >= MinLoyaltyToChooseSkill)
		{
			List<FriendToken> list = new List<FriendToken>();
			if (Config.ActiveCost.HasValue && Loyalty >= -ActiveCost)
			{
				list.Add(FriendToken.Active);
			}
			if (Config.ActiveCost2.HasValue && Loyalty >= -ActiveCost2)
			{
				list.Add(FriendToken.Active2);
			}
			if (Config.UltimateCost.HasValue && Loyalty >= -UltimateCost)
			{
				list.Add(FriendToken.Ultimate);
			}
			List<Card> list2 = Clone(list.Count).ToList();
			foreach (var (index, friendToken) in list.WithIndices())
			{
				list2[index].FriendToken = friendToken;
			}
			return new MiniSelectCardInteraction(list2);
		}
		return null;
	}

	protected virtual IEnumerable<BattleAction> PlayCardPerformAction()
	{
		bool overrideAnima = false;
		if (Config.Perform.Length != 0)
		{
			if (Config.AutoPerform)
			{
				for (int i = 0; i < Config.Perform.Length; i++)
				{
					yield return CardPerformAction(i, PerformTargetType.Player);
				}
			}
			string[][] perform = Config.Perform;
			for (int j = 0; j < perform.Length; j++)
			{
				if (int.Parse(perform[j][0]) == 3)
				{
					overrideAnima = true;
				}
			}
		}
		if (!overrideAnima)
		{

	internal IEnumerable<BattleAction> GetActions(UnitSelector selector, ManaGroup consumingMana, Interaction precondition, bool kicker, bool summoning, IList<DamageAction> damageActions)
	{
		foreach (BattleAction item4 in PlayCardPerformAction())
		{
			yield return item4;
		}
		PendingManaUsage = consumingMana;
		if (selector.Type == TargetType.SingleEnemy)
		{
			PendingTarget = selector.SelectedEnemy;
		}
		if (kicker)
		{
			KickerPlaying = true;
			foreach (BattleAction item5 in KickerActions(selector, consumingMana, precondition))
			{
				if (item5 is DamageAction item)
				{
					damageActions.Add(item);
				}
				yield return item5;
			}
		}
		else
		{
			KickerPlaying = false;
			if (summoning)
			{
				yield return SkillAnime;
				foreach (BattleAction item6 in SummonActions(selector, consumingMana, precondition))
				{
					if (item6 is DamageAction item2)
					{
						damageActions.Add(item2);
					}
					yield return item6;
				}
			}
			else
			{
				foreach (BattleAction item7 in Actions(selector, consumingMana, precondition))
				{
					if (item7 is DamageAction item3)
					{
						damageActions.Add(item3);
					}
					yield return item7;
				}
			}
		}
		PendingManaUsage = null;
		PendingTarget = null;
	}

	{
		switch (CardType)
		{
		case CardType.Attack:
			SetGuns();
			if (CardGuns == null)
			{
				yield return AttackAction(selector);
				break;
			}
			foreach (GunPair gunPair in CardGuns.GunPairs)
			{
				yield return AttackAction(selector, gunPair);
			}
			break;
		case CardType.Defense:
			yield return DefenseAction();
			break;
		}
	}

	protected BattleAction AttackAction(UnitSelector selector, DamageInfo damage, GunPair gunPair = null)
	{
		if (Battle.BattleShouldEnd)
		{
			return new EndShootAction(Battle.Player);
		}
		string gunName;
		GunType gunType;
		if (gunPair != null)
		{
			gunName = gunPair.GunName;
			gunType = gunPair.GunType;
		}
		else
		{
			gunName = GunName;
			gunType = GunType.Single;
		}
		switch (Config.TargetType)
		{
		case TargetType.SingleEnemy:
		{
			EnemyUnit enemy2 = selector.GetEnemy(Battle);
			if (enemy2 != null && enemy2.IsAlive)
			{
				return new DamageAction(Battle.Player, enemy2, damage, gunName, gunType);
			}
			return new EndShootAction(Battle.Player);
		}
		case TargetType.RandomEnemy:
		{
			EnemyUnit enemy = selector.GetEnemy(Battle);
			if (enemy != null && enemy.IsAlive)
			{
				return new DamageAction(Battle.Player, enemy, damage, gunName, gunType);
			}
			return new EndShootAction(Battle.Player);
		}
		case TargetType.AllEnemies:
		{
			EnemyUnit[] enemies = selector.GetEnemies(Battle);
			if (enemies.Length != 0)
			{
				return new DamageAction(Battle.Player, enemies, damage, gunName, gunType);
			}
			return new EndShootAction(Battle.Player);
		}
		default:
			throw new ArgumentException($"Invalid TargetType for {DebugName}: {Config.TargetType}");
		}
	}

	protected bool Overdrive(int consume = 1)
	{
		if (consume <= 0)
		{
			throw new InvalidOperationException("Card: " + Name + " is trying to overdrive with consume blow zero.");
		}
		if (Battle.Player.GetStatusEffect<Concentration>() != null)
		{
			return true;
		}
		Charging statusEffect = Battle.Player.GetStatusEffect<Charging>();
		if (statusEffect == null)
		{
			return false;
		}
		if (statusEffect.Level < consume)
		{
			return false;
		}
		return true;
	}

	protected BattleAction OverdriveAction(int consume = 1)
	{
		if (consume <= 0)
		{
			throw new InvalidOperationException("Card: " + Name + " is trying to overdrive action with consume blow zero.");
		}
		Concentration statusEffect = Battle.Player.GetStatusEffect<Concentration>();
		if (statusEffect != null)
		{
			if (statusEffect.Level == 1)
			{
				return new RemoveStatusEffectAction(statusEffect);
			}
			int level = statusEffect.Level - 1;
			statusEffect.Level = level;
			return null;
		}
		Charging statusEffect2 = Battle.Player.GetStatusEffect<Charging>();
		if (statusEffect2 == null)
		{
			throw new InvalidOperationException("Card: " + Name + " is trying overdrive action but there is no charging se.");
		}
		if (statusEffect2.Level < consume)
		{
			throw new InvalidOperationException("Card: " + Name + " is trying overdrive action but charging se level can't afford.");
		}
		if (statusEffect2.Level == consume)
		{
			return new RemoveStatusEffectAction(statusEffect2);
		}
		statusEffect2.Level -= consume;
		return null;
	}

	private int BasicCardModify(int original, bool isDamage = false)
	{
		int num = original;
		if (IsBasic && base.GameRun != null)
		{
			if (isDamage && CardType == CardType.Attack)
			{
				num += ((Config.Cost.Amount > 1) ? base.GameRun.BasicAttackCardExtraDamage2 : base.GameRun.BasicAttackCardExtraDamage1);
			}
			if (base.GameRun.WanbaochuiFlag > 0 && PendingManaUsage.HasValue)
			{
				int colorless = PendingManaUsage.Value.Colorless;
				if (colorless > 0)
				{
					num *= colorless + 1;
				}
			}
			if (base.GameRun.BasicCardIncrease > 0)
			{
				num += base.GameRun.BasicCardIncrease;
			}
		}
		return num;
	}

	protected PerformAction CardPerformAction(int id, PerformTargetType? performType = PerformTargetType.Player, [CanBeNull] Unit target = null)
	{
		string[][] perform = Config.Perform;
		if (id > perform.Length || id < 0)
		{
			Debug.Log(this?.ToString() + "的CardPerformAction不存在第" + id + "项");
			return null;
		}
		string[] array = perform[id];
		float num = 0f;
		if (array.Length > 2)
		{
			num = float.Parse(array[2]);
		}
		Unit unit = Battle.Player;
		switch (performType)
		{
		case PerformTargetType.Player:
			if (target == null)
			{
				target = Battle.Player;
			}
			break;
		case PerformTargetType.TargetSelf:
			unit = target;
			break;
		}
		bool flag;
		switch (int.Parse(array[0]))
		{
		case 1:
			if (performType.HasValue)
			{
				PerformTargetType valueOrDefault = performType.GetValueOrDefault();
				if ((uint)(valueOrDefault - 3) <= 1u)
				{
					flag = true;
					goto IL_00f5;
				}
			}
			flag = false;
			goto IL_00f5;
		case 2:
			if (performType.HasValue)
			{
				PerformTargetType valueOrDefault = performType.GetValueOrDefault();
				if ((uint)(valueOrDefault - 3) <= 1u)
				{
					flag = true;
					goto IL_0191;
				}
			}
			flag = false;
			goto IL_0191;
		case 3:
			return PerformAction.Animation(Battle.Player, array[1], (array.Length > 2) ? num : 0.2f);
		case 4:
			return PerformAction.Sfx(array[1]);
		default:
			{
				Debug.Log(this?.ToString() + "的CardPerformAction的类型不符合预期");
				return null;
			}
			IL_00f5:
			if (flag)
			{
				using IEnumerator<EnemyUnit> enumerator = Battle.EnemyGroup.Alives.GetEnumerator();
				if (enumerator.MoveNext())
				{
					EnemyUnit current = enumerator.Current;
					return PerformAction.Gun((performType == PerformTargetType.EachEnemySelf) ? current : unit, current, array[1], num);
				}
			}
			return PerformAction.Gun(unit, target, array[1], num);
			IL_0191:
			if (flag)
			{
				string text = "All enemy names: ";
				using IEnumerator<(int, EnemyUnit)> enumerator2 = Battle.EnemyGroup.Alives.WithIndices().GetEnumerator();
				if (enumerator2.MoveNext())
				{
					var (num2, enemyUnit) = enumerator2.Current;
					if (num2 == Battle.EnemyGroup.Alives.Count() - 1)
					{
						text += enemyUnit.Name;
					}
					else
					{
						text = text + enemyUnit.Name + ", ";
					}
					return PerformAction.Effect(enemyUnit, array[1]);
				}
			}
			return PerformAction.Effect(target, array[1]);
		}
	}

	protected override string GetBaseDescription()
	{
		if (Config.EmptyDescription)
		{
			return string.Empty;
		}
		string text = string.Empty;
		if (CardType == CardType.Friend)
		{
			switch (FriendToken)
			{
			case FriendToken.None:
				if (!Config.EmptyDescription)
				{
					text += (IsUpgraded ? (UpgradedBaseDescription ?? base.BaseDescription) : base.BaseDescription);
				}
				if (HasExtraDescription1)
				{
					text += "\n";
					text += RawExtraDescription1;
				}
				if (HasExtraDescription2)
				{
					text += "\n";
					text += RawExtraDescription2;
				}
				if (HasExtraDescription3)
				{
					text += "\n";
					text += RawExtraDescription3;
				}
				break;
			case FriendToken.Active:
				text += RawExtraDescription1;
				break;
			case FriendToken.Active2:
				text += RawExtraDescription2;
				break;
			case FriendToken.Ultimate:
				text += RawExtraDescription3;
				break;
			default:
				throw new ArgumentOutOfRangeException();
			}
		}
		else
		{
			switch (ChoiceCardIndicator)
			{
			case 1:
				text = RawExtraDescription1;
				break;
			case 2:
				text = RawExtraDescription2;
				break;
			default:
			{
				if (Battle != null)
				{
					text = (IsUpgraded ? (UpgradedBaseDescription ?? base.BaseDescription) : base.BaseDescription);
					break;
				}
				object obj;
				if (!IsUpgraded)
				{
					obj = NonbattleBaseDescription ?? base.BaseDescription;
				}
				else
				{
					obj = UpgradedNonbattleBaseDescription ?? UpgradedBaseDescription;
					if (obj == null)
					{
						obj = NonbattleBaseDescription ?? base.BaseDescription;
					}
				}
				text = (string)obj;
				break;
			}
			}
		}
		return FollowByDetailIcon(text);
	}

	public IEnumerable<IDisplayWord> EnumerateDisplayWords(bool verbose)
	{
		CardConfig config = Config;
		IReadOnlyList<string> initEffects = (IsUpgraded ? config.UpgradedRelativeEffects : config.RelativeEffects);
		Keyword keyword = (IsUpgraded ? config.UpgradedRelativeKeyword : config.RelativeKeyword);
		switch (CardType)
		{
		case CardType.Ability:
			keyword |= Keyword.Ability;
			break;
		case CardType.Friend:
			keyword |= (Summoned ? Keyword.FriendSummoned : Keyword.Friend);
			keyword |= Keyword.Loyalty;
			break;
		case CardType.Tool:
			keyword |= Keyword.Tool;
			break;
		case CardType.Misfortune:
			keyword |= Keyword.Misfortune;
			break;
		}
		if (IsXCost)
		{
			keyword |= Keyword.XCost;
		}
		if (HasKicker)
		{
			keyword |= Keyword.Kicker;
		}
		if (IsUpgraded)
		{
			if (config.UpgradedCost.HasValue || config.Cost.Hybrid <= 0)
			{
				ManaGroup? upgradedCost = config.UpgradedCost;
				if (!upgradedCost.HasValue || upgradedCost.GetValueOrDefault().Hybrid <= 0)
				{
					goto IL_0171;
				}
			}
			keyword |= Keyword.HybridCost;
		}
		else if (config.Cost.Hybrid > 0)
		{
			keyword |= Keyword.HybridCost;
		}
		goto IL_0171;
		IL_0171:
		foreach (IDisplayWord item in Library.InternalEnumerateDisplayWords(base.GameRun, keyword, initEffects, verbose, Keywords))
		{
			yield return item;
		}
	}

	public virtual IEnumerable<Card> EnumerateRelativeCards()
	{
		CardConfig config = Config;
		IReadOnlyList<string> readOnlyList = (IsUpgraded ? config.UpgradedRelativeCards : (config.RelativeCards ?? Array.Empty<string>()));
		foreach (string item in readOnlyList)
		{
			Card card;
			if (item.Last() == '+')
			{
				string text = item;
				card = Library.CreateCard(text.Substring(0, text.Length - 1));
				card.Upgrade();
			}
			else
			{
				card = Library.CreateCard(item);
			}
			card.GameRun = base.GameRun;
			yield return card;
		}
	}

	private IEnumerable<string> EnumerateAutoAppendKeywordNames()
	{
		foreach (Keyword item in LBoL.Core.Keywords.EnumerateComponents(Keywords))
		{
			KeywordDisplayWord displayWord = LBoL.Core.Keywords.GetDisplayWord(item);
			if (displayWord != null)
			{
				if (displayWord.IsAutoAppend)
				{
					yield return displayWord.Name;
				}
			}
			else
			{
				Debug.LogWarning($"Cannot find display-word for {item}, maybe not localized");
				yield return "&lt;" + item.ToString() + "&gt;";
			}
		}
	}

	public override void Initialize()
	{
		base.Initialize();
		Config = CardConfig.FromId(base.Id);
		if (Config == null)
		{
			throw new InvalidDataException("Cannot find card config for <" + base.Id + ">");
		}
		Keywords = Config.Keywords;
		if (!Config.IsXCost)
		{
			BaseCost = ConfigCost;
			TurnCostDelta = ManaGroup.Empty;
		}
		if (CardType == CardType.Tool)
		{
			DeckCounter = ToolPlayableTimes;
		}
		if (CardType == CardType.Friend)
		{
			Loyalty = ConfigLoyalty;
		}
		if (PlentifulMana.HasValue)
		{
			IsPlentiful = true;
		}
	}

	protected void HandleBattleCardsChangedEvents(GameEventHandler<GameEventArgs> handler, bool hand = false, bool draw = false, bool discard = false, bool exile = false)
	{
		HandleBattleEvent(Battle.CardExiled, handler);
		HandleBattleEvent(Battle.CardRemoved, handler);
		HandleBattleEvent(Battle.CardMoved, handler);
		HandleBattleEvent(Battle.CardMovedToDrawZone, handler);
		if (hand)
		{
			HandleBattleEvent(Battle.CardsAddedToHand, handler);
		}
		if (draw)
		{
			HandleBattleEvent(Battle.CardsAddedToDrawZone, handler);
		}
		if (discard)
		{
			HandleBattleEvent(Battle.CardsAddedToDiscard, handler);
		}
		if (hand)
		{
			HandleBattleEvent(Battle.CardUsed, handler);
		}
		if (hand && draw)
		{
			HandleBattleEvent(Battle.CardDrawn, handler);
		}
		if (hand && draw && discard)
		{
			HandleBattleEvent(Battle.CardDiscarded, handler);
		}
		if (draw && discard)
		{
			HandleBattleEvent(Battle.Reshuffled, handler);
		}
	}

	public static int CompareCard(Card a, Card b)
	{
		int num = a.Config.Order.CompareTo(b.Config.Order);
		if (num != 0)
		{
			return num;
		}
		return string.Compare(a.Id, b.Id, StringComparison.Ordinal);
	}

	public static int CompareMuseumCardColors(IReadOnlyList<ManaColor> colorsA, IReadOnlyList<ManaColor> colorsB)
	{
		int count = colorsA.Count;
		int count2 = colorsB.Count;
		if (count == 0 && count2 != 0)
		{
			return 1;
		}
		if (count2 == 0 && count != 0)
		{
			return -1;
		}
		if (count == 0 && count2 == 0)
		{
			return 0;
		}
		int num = count - count2;
		if (num <= 0)
		{
			if (num < 0)
			{
				return -1;
			}
			for (int i = 0; i < count; i++)
			{
				int num2 = colorsA[i].CompareTo(colorsB[i]);
				if (num2 != 0)
				{
					return num2;
				}
			}
			return 0;
		}
		return 1;
	}

	protected int SynergyAmount(ManaGroup cost, ManaColor synergyColor, int synergyColorAmount = 1)
	{
		if (synergyColorAmount < 1)
		{
			throw new InvalidOperationException("can not get synergy with mana amount blew zero.");
		}
		int num = 0;
		return synergyColor switch
		{
			ManaColor.Philosophy => cost.Philosophy, 
			ManaColor.Any => cost.Amount, 
			_ => cost.Philosophy + cost.GetValue(synergyColor), 
		} / synergyColorAmount + base.GameRun.SynergyAdditionalCount;
	}

	protected int SynergyAmountComplexMana(ManaGroup cost, ManaGroup synergyMana)
	{
		int num = 0;
		ManaGroup empty = ManaGroup.Empty;
		while (cost.CanAfford(empty + synergyMana))
		{
			empty += synergyMana;
			num++;
		}
		return num + base.GameRun.SynergyAdditionalCount;
	}

	public void Verify()
	{
		if (CardType == CardType.Defense && !HasBlock && !HasShield)
		{
			Debug.LogError("<" + base.Id + "> must set block or shield in config");
		}
		if (CardType != CardType.Attack && CardType != CardType.Tool && IsAccuracy)
		{
			Debug.LogError("<" + base.Id + "> should not be accuracy when it is not an attack card.");
		}
		if (!Config.UpgradedCost.HasValue)
		{
			return;
		}
		ManaGroup cost = Config.Cost;
		ManaGroup value = Config.UpgradedCost.Value;
		if (cost.Amount < value.Amount)
		{
			throw new InvalidOperationException(DebugName + ": upgraded cost is illegal, Amount increase.");
		}
		foreach (ManaColor item in ManaColors.WUBRGC)
		{
			if (cost.GetValue(item) < value.GetValue(item))
			{
				throw new InvalidOperationException(DebugName + ": upgraded cost is illegal, There is some single color's value increase.");
			}
		}
	}

	public (Card card, Card upgradedCard) GetDetailInfoCard()
	{
		Card card = Library.CreateCard(base.Id);
		card.GameRun = base.GameRun;
		if (card.CanUpgrade)
		{
			Card card2 = Library.CreateCard(base.Id, upgraded: true, UpgradeCounter);
			card2.GameRun = base.GameRun;
			return (card: card, upgradedCard: card2);
		}
		return (card: card, upgradedCard: null);
	}

			EnchantmentStickerId = enchantmentId;
			EnchantmentChanged = true;
			NotifyChanged();
		}
	}

	{
		if (EnchantmentStickerId != null)
		{
			EnchantmentStickerId = null;
			EnchantmentChanged = true;
			NotifyChanged();
		}
	}
