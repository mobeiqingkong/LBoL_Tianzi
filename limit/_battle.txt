	private IEnumerator<object> ResolveDebugActions()
	{
		(BattleAction, string) result;
		while (_debugActionQueue.TryDequeue(out result))
		{
			var (root, recordName) = result;
			yield return _resolver.Resolve(root, recordName);
		}
	}

	private IEnumerator<object> ResolveAction(BattleAction battleAction, [MaybeNull] string recordName = null)
	{
		yield return ResolveDebugActions();
		yield return _resolver.Resolve(battleAction, recordName);
	}

	internal BattleController(GameRunController gameRun, EnemyGroup enemyGroup, IEnumerable<Card> deck)
	{
		GameRun = gameRun;
		_resolver = new ActionResolver(this);
		Player = gameRun.Player;
		EnemyGroup = enemyGroup;
		BattleMana = default(ManaGroup);
		DrawCardCount = GameRun.DrawCardCount;
		List<Card> list = new List<Card>();
		List<Card> list2 = new List<Card>();
		List<Card> list3 = new List<Card>();
		foreach (Card item in deck)
		{
			Card card = item.Clone();
			card.InstanceId = item.InstanceId;
			if (item.IsInitial)
			{
				list.Add(card);
			}
			else if (item.ShuffleToBottom)
			{
				list3.Add(card);
			}
			else
			{
				list2.Add(card);
			}
		}
		list.Shuffle(GameRun.ShuffleRng);
		list2.Shuffle(GameRun.ShuffleRng);
		list3.Shuffle(GameRun.ShuffleRng);
		_drawZone.AddRange(list);
		_drawZone.AddRange(list2);
		_drawZone.AddRange(list3);
		foreach (Card item2 in _drawZone)
		{
			item2.GameRun = GameRun;
			item2.Zone = CardZone.Draw;
			item2.EnterBattle(this);
		}
		Player.EnterBattle(this);
		foreach (EnemyUnit item3 in EnemyGroup)
		{
			item3.EnterBattle(this);
		}
	}

	internal void Leave()
	{
		foreach (Card item in EnumerateAllCards())
		{
			item.LeaveBattle();
			item.GameRun = null;
		}
		foreach (EnemyUnit item2 in EnemyGroup)
		{
			item2.LeaveBattle();
		}
		Player.LeaveBattle();
		Stats.TotalRounds = RoundCounter;
		Stats.RemainCardCount = _drawZone.Count + _discardZone.Count + _handZone.Count + _playArea.Count;
	}

	internal void NotifyMessage(BattleMessage message)
	{
		this.Notification?.Invoke(message);
	}

	private void ShuffleDrawPile()
	{
		if (_drawZone.Any((Card card) => card.ShuffleToBottom))
		{
			List<Card> list = _drawZone.Where((Card card) => !card.ShuffleToBottom).ToList();
			List<Card> list2 = _drawZone.Where((Card card) => card.ShuffleToBottom).ToList();
			list.Shuffle(GameRun.ShuffleRng);
			list2.Shuffle(GameRun.ShuffleRng);
			_drawZone.Clear();
			_drawZone.AddRange(list);
			_drawZone.AddRange(list2);
		}
		else
		{
			_drawZone.Shuffle(GameRun.ShuffleRng);
		}
	}

	internal void StartBattle()
	{
		if (Player.HasUs)
		{
			Player.Us.TurnAvailable = true;
			Player.Us.BattleAvailable = true;
		}
		else
		{
			Player.Us.TurnAvailable = false;
			Player.Us.BattleAvailable = false;
		}
	}

	internal Card DrawCard()
	{
		if (_handZone.Count >= MaxHand)
		{
			NotifyMessage(BattleMessage.HandFull);
			return null;
		}
		if (_drawZone.Empty())
		{
			NotifyMessage(BattleMessage.EmptyDraw);
			return null;
		}
		Card card = _drawZone[0];
		_drawZone.RemoveAt(0);
		_handZone.Add(card);
		card.Zone = CardZone.Hand;
		_turnDrawHistory.Add(card);
		return card;
	}

	internal Card DrawSelectedCard(Card card)
	{
		if (_handZone.Count >= MaxHand)
		{
			NotifyMessage(BattleMessage.HandFull);
			return null;
		}
		if (!_drawZone.Contains(card))
		{
			throw new InvalidOperationException("Drawing " + card.Name + " not in hand");
		}
		_drawZone.Remove(card);
		_handZone.Add(card);
		card.Zone = CardZone.Hand;
		_turnDrawHistory.Add(card);
		return card;
	}

	internal void Reshuffle()
	{
		foreach (Card item in _discardZone)
		{
			item.Zone = CardZone.Draw;
		}
		_drawZone.AddRange(_discardZone);
		_discardZone.Clear();
		ShuffleDrawPile();
	}

	internal void MoveCardToPlayArea(Card card)
	{
		if (!_playArea.Empty())
		{
			Debug.LogWarning("Moving <" + card.Name + "> to play-area while using other card: {" + ", ".Join(_playArea.Select((Card c) => c.Name)) + "}");
		}
		if (!_handZone.Contains(card))
		{
			throw new InvalidOperationException("Playing card " + card.Name + " not in hand zone: {" + ", ".Join(_handZone.Select((Card c) => c.Name)) + "}");
		}
		card.HandIndexWhenPlaying = _handZone.IndexOf(card);
		_handZone.Remove(card);
		_playArea.Add(card);
		card.Zone = CardZone.PlayArea;
	}

	internal void MoveCardToFollowArea(Card card)
	{
		if (_drawZone.Contains(card))
		{
			_drawZone.Remove(card);
		}
		else if (_discardZone.Contains(card))
		{
			_discardZone.Remove(card);
		}
		else if (_exileZone.Contains(card))
		{
			_exileZone.Remove(card);
		}
		else
		{
			if (!_handZone.Contains(card))
			{
				throw new InvalidOperationException("Follow-playing card " + card.Name + " not in correct zones.");
			}
			_handZone.Remove(card);
		}
		card.HandIndexWhenPlaying = 0;
		_followArea.Add(card);
		card.Zone = CardZone.FollowArea;
	}

	internal CancelCause Discard(Card card)
	{
		if (_discardZone.Count >= 9999)
		{
			NotifyMessage(BattleMessage.ZoneFull);
			return CancelCause.ZoneFull;
		}
		CardZone zone = card.Zone;
		(zone switch
		{
			CardZone.None => throw new InvalidOperationException("Discarding card '" + card.DebugName + "' is not in battle"), 
			CardZone.Draw => _drawZone, 
			CardZone.Hand => _handZone, 
			CardZone.PlayArea => _playArea, 
			CardZone.FollowArea => _followArea, 
			CardZone.Discard => throw new InvalidOperationException("Dicarding card '" + card.DebugName + "' already in discard zone"), 
			CardZone.Exile => throw new InvalidOperationException("Cannot discard '" + card.DebugName + "' from exile zone"), 
			_ => throw new ArgumentOutOfRangeException(), 
		}).Remove(card);
		_discardZone.Add(card);
		if (zone == CardZone.Hand || zone == CardZone.PlayArea)
		{
			card.OnLeaveHand();
		}
		card.Zone = CardZone.Discard;
		_turnDiscardHistory.Add(card);
		return CancelCause.None;
	}

	internal CancelCause ExileCard(Card card)
	{
		if (_exileZone.Count >= 9999)
		{
			NotifyMessage(BattleMessage.ZoneFull);
			return CancelCause.ZoneFull;
		}
		CardZone zone = card.Zone;
		(zone switch
		{
			CardZone.None => throw new InvalidOperationException("Exiling card '" + card.DebugName + "' is not in battle"), 
			CardZone.Draw => _drawZone, 
			CardZone.Hand => _handZone, 
			CardZone.PlayArea => _playArea, 
			CardZone.FollowArea => _followArea, 
			CardZone.Discard => _discardZone, 
			CardZone.Exile => throw new InvalidOperationException("Exiling card '" + card.DebugName + "' already in exile zone"), 
			_ => throw new ArgumentOutOfRangeException(), 
		}).Remove(card);
		_exileZone.Add(card);
		if (zone == CardZone.Hand || zone == CardZone.PlayArea)
		{
			card.OnLeaveHand();
		}
		card.Zone = CardZone.Exile;
		_turnExileHistory.Add(card);
		return CancelCause.None;
	}

	internal CancelCause TransformCard(Card sourceCard, Card destinationCard)
	{
		if (destinationCard.Zone != 0)
		{
			throw new InvalidOperationException($"Cannot Transform to {destinationCard.DebugName} while it's already in:{destinationCard.Zone}");
		}
		destinationCard.Zone = sourceCard.Zone;
		object obj = sourceCard.Zone switch
		{
			CardZone.None => throw new InvalidOperationException("Cannot transform card " + sourceCard.DebugName + " while it's not in any zone of battle."), 
			CardZone.Draw => _drawZone, 
			CardZone.Hand => _handZone, 
			CardZone.PlayArea => _playArea, 
			CardZone.FollowArea => _followArea, 
			CardZone.Discard => _discardZone, 
			CardZone.Exile => _exileZone, 
			_ => throw new ArgumentOutOfRangeException(), 
		};
		int index = ((List<Card>)obj).IndexOf(sourceCard);
		RemoveCard(sourceCard);
		((List<Card>)obj).Insert(index, destinationCard);
		destinationCard.GameRun = GameRun;
		destinationCard.InstanceId = ++_cardInstanceId;
		destinationCard.EnterBattle(this);
		GameRun.RevealCard(destinationCard);
		return CancelCause.None;
	}

	internal static void MoveCardCheck(Card card, CardZone dstZone)
	{
		switch (dstZone)
		{
		case CardZone.None:
			throw new InvalidOperationException("Cannot move card '" + card.DebugName + "' to None zone");
		case CardZone.Draw:
			throw new InvalidOperationException("Cannot move card '" + card.DebugName + "' to draw-zone");
		}
		if (dstZone == card.Zone)
		{
			throw new InvalidOperationException($"Cannot move card '{card.DebugName}' (in {card.Zone}) to same zone");
		}
	}

	internal CancelCause MoveCard(Card card, CardZone dstZone)
	{
		MoveCardCheck(card, dstZone);
		CardZone zone = card.Zone;
		List<Card> list = card.Zone switch
		{
			CardZone.None => throw new InvalidOperationException($"Cannot move new card '{card.DebugName}' to {dstZone}"), 
			CardZone.Draw => _drawZone, 
			CardZone.Hand => _handZone, 
			CardZone.PlayArea => _playArea, 
			CardZone.FollowArea => _followArea, 
			CardZone.Discard => _discardZone, 
			CardZone.Exile => _exileZone, 
			_ => throw new ArgumentOutOfRangeException(), 
		};
		if (!list.Contains(card))
		{
			throw new InvalidOperationException($"Fatal: {card.Zone} does not contain card '{card.DebugName}'.");
		}
		if (dstZone == CardZone.Hand && _handZone.Count >= MaxHand)
		{
			NotifyMessage(BattleMessage.HandFull);
			return CancelCause.HandFull;
		}
		List<Card> list2 = dstZone switch
		{
			CardZone.Hand => _handZone, 
			CardZone.Discard => _discardZone, 
			CardZone.Exile => _exileZone, 
			_ => throw new ArgumentOutOfRangeException("dstZone", dstZone, null), 
		};
		if (list2.Count >= 9999)
		{
			NotifyMessage(BattleMessage.ZoneFull);
			return CancelCause.ZoneFull;
		}
		list.Remove(card);
		list2.Add(card);
		if ((zone == CardZone.Hand || zone == CardZone.PlayArea) && (dstZone == CardZone.Exile || dstZone == CardZone.Discard))
		{
			card.OnLeaveHand();
		}
		card.Zone = dstZone;
		return CancelCause.None;
	}

	internal CancelCause MoveCardToDrawZone(Card card, DrawZoneTarget target)
	{
		CardZone zone = card.Zone;
		if (_drawZone.Count >= 9999)
		{
			NotifyMessage(BattleMessage.ZoneFull);
			return CancelCause.ZoneFull;
		}
		(zone switch
		{
			CardZone.None => throw new InvalidOperationException("Cannot move new card " + card.DebugName + " to draw zone, use AddCardsToDrawZoneAction instead"), 
			CardZone.Draw => _drawZone, 
			CardZone.Hand => _handZone, 
			CardZone.PlayArea => _playArea, 
			CardZone.FollowArea => _followArea, 
			CardZone.Discard => _discardZone, 
			CardZone.Exile => _exileZone,

	internal void TurnStartDecreaseDuration(Unit target)
	{
		foreach (StatusEffect statusEffect in target.StatusEffects)
		{
			if (!statusEffect.HasDuration)
			{
				continue;
			}
			if (target.IsExtraTurn)
			{
				if (!statusEffect.Config.DurationDecreaseTiming.HasFlag(DurationDecreaseTiming.ExtraTurnStart))
				{
					continue;
				}
			}
			else if (!statusEffect.Config.DurationDecreaseTiming.HasFlag(DurationDecreaseTiming.NormalTurnStart))
			{
				continue;
			}
			if (!statusEffect.IsAutoDecreasing)
			{
				statusEffect.IsAutoDecreasing = true;
				continue;
			}
			if (statusEffect.Duration <= 0)
			{
				Debug.LogError("Turn start: found <" + statusEffect.DebugName + ">.Duration = " + statusEffect.Duration);
				continue;
			}
			int duration = statusEffect.Duration - 1;
			statusEffect.Duration = duration;
			statusEffect.NotifyChanged();
			if (statusEffect.Duration == 0)
			{
				React(new RemoveStatusEffectAction(statusEffect), null, ActionCause.TurnStart);
			}
		}
	}

	internal void TurnEndDecreaseDuration(Unit target)
	{
		foreach (StatusEffect statusEffect in target.StatusEffects)
		{
			if (!statusEffect.HasDuration)
			{
				continue;
			}
			if (target.HasStatusEffect<ExtraTurn>() || (target.HasStatusEffect<SuperExtraTurn>() && !target.IsExtraTurn))
			{
				if (!statusEffect.Config.DurationDecreaseTiming.HasFlag(DurationDecreaseTiming.EndTurnForExtra))
				{
					continue;
				}
			}
			else if (!statusEffect.Config.DurationDecreaseTiming.HasFlag(DurationDecreaseTiming.EndTurnForRound))
			{
				continue;
			}
			if (!statusEffect.IsAutoDecreasing)
			{
				statusEffect.IsAutoDecreasing = true;
				continue;
			}
			if (statusEffect.Duration <= 0)
			{
				Debug.LogError("Turn end: found <" + statusEffect.DebugName + ">.Duration = " + statusEffect.Duration);
				continue;
			}
			int duration = statusEffect.Duration - 1;
			statusEffect.Duration = duration;
			statusEffect.NotifyChanged();
			if (statusEffect.Duration == 0)
			{
				React(new RemoveStatusEffectAction(statusEffect), null, ActionCause.TurnStart);
			}
		}
	}

	internal ManaGroup LockTurnMana(ManaGroup group)
	{
		ManaGroup a = LockedTurnMana + group;

	}

	internal bool ConvertMana(ManaGroup input, ManaGroup output, bool allowPartial, out ManaGroup resultInput, out ManaGroup resultOutput)
	{
		ManaGroup battleMana = BattleMana;
		ManaGroup manaGroup = battleMana - input;
		if (manaGroup.IsInvalid)
		{
			if (!allowPartial)
			{
				Debug.LogError($"[Battle] Cannot convert mana ({input} => {output}): available = {battleMana}");
				resultInput = ManaGroup.Empty;
				resultOutput = ManaGroup.Empty;
				return false;
			}
			manaGroup = manaGroup.Corrected;
		}
		resultInput = battleMana - manaGroup;
		BattleMana = (manaGroup + output).ClampComponentMax(99);
		resultOutput = BattleMana - manaGroup;
		return true;
	}

	internal void ConsumeMana(ManaGroup group)
	{
		if (!BattleMana.CanAfford(group))
		{
			throw new InvalidOperationException($"Pool ({BattleMana}) cannot afford cost {group}");
		}
		if (group.IsInvalid)
		{
			throw new InvalidOperationException($"Consuming invalid mana: {group}");
		}
		BattleMana -= group;
	}

	public int CalculateDamage(GameEntity actionSource, Unit source, Unit target, DamageInfo damageInfo)
	{
		DamageDealingEventArgs damageDealingEventArgs = new DamageDealingEventArgs();
		damageDealingEventArgs.Source = source;
		damageDealingEventArgs.Targets = ((target == null) ? null : new Unit[1] { target });
		damageDealingEventArgs.DamageInfo = damageInfo;
		damageDealingEventArgs.ActionSource = actionSource;
		damageDealingEventArgs.Cause = ActionCause.OnlyCalculate;
		DamageDealingEventArgs damageDealingEventArgs2 = damageDealingEventArgs;
		source.DamageDealing.Execute(damageDealingEventArgs2);
		damageInfo = damageDealingEventArgs2.DamageInfo;
		if (target != null)
		{
			DamageEventArgs damageEventArgs = new DamageEventArgs
			{
				Source = source,
				Target = target,
				DamageInfo = damageInfo,
				Cause = ActionCause.OnlyCalculate
			};
			target.DamageReceiving.Execute(damageEventArgs);
			damageInfo = damageEventArgs.DamageInfo;
		}
		return Math.Max(0, damageInfo.Damage.RoundToInt(MidpointRounding.AwayFromZero));
	}

			return result;
		}
		Debug.LogError("Adding " + effect.GetType().Name + " to " + target.Name + " failed.");
		return null;
	}

	internal bool RemoveStatusEffect(Unit target, StatusEffect effect)
	{
		if (target.TryRemoveStatusEffect(effect))
		{
			TriggerGlobalStatusChanged();
			return true;
		}
		Debug.LogError("Removing " + effect.GetType().Name + " from " + target.Name + " failed.");
		return false;
	}

	internal (int power, int bluePoint, int money) GenerateEnemyPoints(EnemyUnit enemy)
	{
		RandomGen gameRunEventRng = GameRun.GameRunEventRng;
		MinMax powerLoot = enemy.Config.PowerLoot;
		int num = gameRunEventRng.NextInt(powerLoot.Min, powerLoot.Max) + gameRunEventRng.NextInt(GameRun.ExtraPowerLowerbound, GameRun.ExtraPowerUpperbound);
		if (enemy.Config.Type == EnemyType.Boss && GameRun.Puzzles.HasFlag(PuzzleFlag.LowStageRegen))
		{
			num = ((float)num * 0.5f).RoundToInt();
		}
		MinMax bluePointLoot = enemy.Config.BluePointLoot;
		int num2 = gameRunEventRng.NextInt(bluePointLoot.Min, bluePointLoot.Max);
		if (enemy.HasStatusEffect<MirrorImage>())
		{
			num = ((float)num * 0.25f).RoundToInt();
			num2 = ((float)num2 * 0.25f).RoundToInt();
		}
		return (power: num, bluePoint: num2, money: 0);
	}

	internal void Die(Unit unit, bool isSuicide, int power, int bluePoint, int money)
	{
		unit.Die();
		if (unit is PlayerUnit)
		{
			if (isSuicide)
			{
				Stats.PlayerSuicide = true;
			}
		}
		else if (unit is EnemyUnit)
		{
			if (power > 0)
			{
				GameRun.InternalGainPower(power);
			}
			if (bluePoint > 0)
			{
				Stats.BluePoint += bluePoint;
			}
			if (money > 0)
			{
				GameRun.InternalGainMoney(money);
			}
		}
	}

	private IEnumerator<object> PlayerTurnFlow()
	{
		bool isExtra = false;
		int continuousTurnCount = 0;
		while (!BattleShouldEnd)
		{
			PlayerTurnShouldEnd = false;
			PlayerUnit player = Player;
			int turnCounter = player.TurnCounter + 1;
			player.TurnCounter = turnCounter;
			turnCounter = continuousTurnCount + 1;
			continuousTurnCount = turnCounter;
			Player.IsExtraTurn = isExtra;
			SuperExtraTurn statusEffect = Player.GetStatusEffect<SuperExtraTurn>();
			if (isExtra)
			{
				if (statusEffect == null)
				{
					ExtraTurn statusEffect2 = Player.GetStatusEffect<ExtraTurn>();
					if (statusEffect2.Level == 1)
					{
						yield return ResolveAction(new RemoveStatusEffectAction(statusEffect2));
					}
					else
					{
						turnCounter = statusEffect2.Level - 1;
						statusEffect2.Level = turnCounter;
					}
				}
				else if (statusEffect.Limit == 1)
				{
					statusEffect.Status = TurnStatus.ExtraTurnByThis;
					statusEffect.Limit = 0;
				}
				else
				{
					statusEffect.Status = TurnStatus.ExtraTurn;
					ExtraTurn statusEffect3 = Player.GetStatusEffect<ExtraTurn>();
					if (statusEffect3.Level == 1)
					{
						yield return ResolveAction(new RemoveStatusEffectAction(statusEffect3));
					}
					else
					{
						turnCounter = statusEffect3.Level - 1;
						statusEffect3.Level = turnCounter;
					}
				}
			}
			else if (statusEffect != null)
			{
				statusEffect.Status = TurnStatus.NaturalTurn;
				statusEffect.Limit = 1;
			}
			foreach (ICustomCounter value in _customCounterTable.Values)
			{
				if (value.AutoResetTiming.HasFlag(CustomCounterResetTiming.PlayerTurnStart))
				{
					value.Reset(this);
				}
			}
			yield return ResolveAction(new StartPlayerTurnAction(Player, isExtra));
			while (!PlayerTurnShouldEnd && !BattleShouldEnd)
			{
				if (_playerInputAction != null)
				{
					Debug.LogWarning("[BattleController] Player input action is not null");
					_playerInputAction = null;
					_playerInputActionRecordName = null;
				}
				IsWaitingPlayerInput = true;
				yield return new WaitUntil(() => !IsWaitingPlayerInput || _debugActionQueue.Count > 0);
				if (_playerInputAction == null)
				{
					yield return ResolveDebugActions();
					continue;
				}
				BattleAction playerInputAction = _playerInputAction;
				string playerInputActionRecordName = _playerInputActionRecordName;
				_playerInputAction = null;
				_playerInputActionRecordName = null;
				foreach (ICustomCounter value2 in _customCounterTable.Values)
				{
					if (value2.AutoResetTiming.HasFlag(CustomCounterResetTiming.PlayerActionStart))
					{
						value2.Reset(this);
					}
				}
				yield return ResolveAction(playerInputAction, playerInputActionRecordName);
				foreach (ICustomCounter value3 in _customCounterTable.Values)
				{
					if (value3.AutoResetTiming.HasFlag(CustomCounterResetTiming.PlayerActionEnd))
					{
						value3.Reset(this);
					}
				}
			}
			if (IsWaitingPlayerInput)
			{
				IsWaitingPlayerInput = false;
				Debug.LogWarning("IsWaitingPlayerInput is still true");
			}
			if (_playerInputAction != null)
			{
				Debug.LogWarning("_playerInputAction is still not null: " + _playerInputAction.GetType().Name);
				_playerInputAction = null;
				_playerInputActionRecordName = null;
			}
			foreach (ICustomCounter value4 in _customCounterTable.Values)
			{
				if (value4.AutoResetTiming.HasFlag(CustomCounterResetTiming.PlayerTurnEnd))
				{
					value4.Reset(this);
				}
			}
			yield return ResolveAction(new EndPlayerTurnAction(Player));
			statusEffect = Player.GetStatusEffect<SuperExtraTurn>();
			if (statusEffect == null)
			{
				if (!Player.HasStatusEffect<ExtraTurn>())
				{
					break;
				}
			}
			else if (!Player.HasStatusEffect<ExtraTurn>() && statusEffect.Limit == 0)
			{
				statusEffect.Status = TurnStatus.OutTurn;
				break;
			}
			isExtra = true;
		}
		if (continuousTurnCount > Stats.ContinuousTurnCount)
		{
			Stats.ContinuousTurnCount = continuousTurnCount;
		}
	}

	private IEnumerator<object> EnemyTurnFlow()
	{
		if (!BattleShouldEnd)
		{
			yield return ResolveAction(new StartAllEnemyTurnAction());
		}
		EnemyUnit[] array = (from enemy in EnemyGroup.Alives
			orderby enemy.MovePriority, enemy.RootIndex
			select enemy).ToArray();
		EnemyUnit[] array2 = array;
		foreach (EnemyUnit enemy2 in array2)
		{
			bool flag = false;
			while (!BattleShouldEnd && !enemy2.IsDead)
			{
				int turnCounter = enemy2.TurnCounter + 1;
				enemy2.TurnCounter = turnCounter;
				enemy2.IsExtraTurn = flag;
				if (flag)
				{
					ExtraTurn statusEffect = enemy2.GetStatusEffect<ExtraTurn>();
					if (statusEffect.Level == 1)
					{
						yield return ResolveAction(new RemoveStatusEffectAction(statusEffect));
					}
					else
					{
						turnCounter = statusEffect.Level - 1;
						statusEffect.Level = turnCounter;
					}
					enemy2.UpdateTurnMoves();
				}
				yield return ResolveAction(new EnemyTurnAction(enemy2), "EnemyTurn: " + enemy2.Name);
				if (!enemy2.HasStatusEffect<ExtraTurn>())
				{
					break;
				}
				flag = true;
			}
		}
		if (!BattleShouldEnd)
		{
			yield return ResolveAction(new EndAllEnemyTurnAction());
		}
	}

	public IEnumerator<object> Flow()
	{
		yield return ResolveAction(new StartBattleAction());
		while (!BattleShouldEnd)
		{
			int roundCounter = RoundCounter + 1;
			RoundCounter = roundCounter;
			yield return ResolveAction(new StartRoundAction());
			if (!BattleShouldEnd)
			{
				yield return PlayerTurnFlow();
			}
			if (!BattleShouldEnd)
			{
				yield return EnemyTurnFlow();
			}
			if (!BattleShouldEnd)
			{
				yield return ResolveAction(new EndRoundAction());
			}
		}
		if (!_forceWin && !_escape && Player.IsAlive)
		{
			List<Unit> list = new List<Unit>();
			foreach (EnemyUnit alife in EnemyGroup.Alives)
			{
				if (!alife.IsServant)
				{
					Debug.LogError(alife.DebugName + " is not servant but still alive while battle end");
				}
				alife.Status = UnitStatus.Dying;
				list.Add(alife);
			}
			if (!list.Empty())
			{
				yield return ResolveAction(new DieAction(list.Zip(Enumerable.Repeat(DieCause.ServantDie, list.Count)), Player, null));
			}
		}
		yield return ResolveAction(new EndBattleAction());
	}

	public bool DoesPlayerStatusEffectForbidUse(Card card, out StatusEffect effect)
	{

		{

				effect = statusEffect;
				return true;
			}
		}
		effect = null;
		return false;
	}

	public bool DoesHandCardPreventUse(Card targetCard, out Card card)
	{
		foreach (Card item in HandZone)
		{
			if (item != targetCard && item.ShouldPreventOtherCardUsage(targetCard))
			{
				card = item;
				return true;
			}
		}
		card = null;
		return false;
	}

	public void RequestUseCard(Card card, UnitSelector selector, ManaGroup mana, bool kicker)
	{
		if (!IsWaitingPlayerInput)
		{
			throw new InvalidOperationException("Cannot play card: not waiting for player input");
		}
		if (DoesHandCardPreventUse(card, out var card2))
		{
			throw new InvalidOperationException("Cannot play card: prevented by " + card2.DebugName);
		}
		if (DoesPlayerStatusEffectForbidUse(card, out var effect))
		{
			throw new InvalidOperationException("Cannot play card: prevented by " + effect.DebugName);
		}
		if (card.IsForbidden)
		{
			throw new InvalidOperationException("Cannot play forbidden card " + card.Name);
		}
		if (!card.CanUse)
		{
			throw new InvalidOperationException("Cannot play card " + card.Name + " (CanUse == false)");
		}
		if (mana.IsInvalid)
		{
			throw new InvalidOperationException(string.Format("UseCard with invalid '{0}': {1}", "mana", mana));
		}
		int? moneyCost = card.Config.MoneyCost;
		if (moneyCost.HasValue)
		{
			int valueOrDefault = moneyCost.GetValueOrDefault();
			if (GameRun.Money < valueOrDefault)
			{
				throw new InvalidOperationException($"Cannot afford card money cost (consuming: {valueOrDefault}, available: {GameRun.Money})");
			}
		}
		if (!BattleMana.CanAfford(mana))
		{
			throw new InvalidOperationException($"Cannot afford card mana cost (consuming: {mana}, available: {BattleMana})");
		}
		if (card.IsXCost && !mana.CanAfford(card.XCostRequiredMana))
		{
			throw new InvalidOperationException($"Cannot afford X cost required mana {card.XCostRequiredMana} with payment {mana}");
		}
		_playerInputAction = new UseCardAction(card, selector, mana, kicker).SetCause(ActionCause.Player);
		_playerInputActionRecordName = "UseCard: " + card.Name;
		_isWaitingPlayerInput = false;
	}

	public void RequestUseUs(UnitSelector selector)
	{
		if (!IsWaitingPlayerInput)
		{
			throw new InvalidOperationException("Cannot use US: not waiting for player input");
		}
		if (!Player.HasUs)
		{
			throw new InvalidOperationException("Player doesn't has US");
		}
		UltimateSkill us = Player.Us;
		if (!us.Available)
		{
			throw new InvalidOperationException("US is not available");
		}
		if (Player.Power < us.PowerCost)
		{
			throw new InvalidOperationException($"Cannot afford US power {us.PowerCost}, current: {Player.Power}");
		}
		_playerInputAction = new UseUsAction(us, selector, us.PowerCost).SetCause(ActionCause.Player);
		_playerInputActionRecordName = "UseUs: " + us.DebugName;
		_isWaitingPlayerInput = false;
	}

	public void CheckManaFreeze()
	{
		ManaFreezeLevel = 0;
		List<ManaFreezer> list = new List<ManaFreezer>();
		foreach (Card item2 in HandZone)
		{
			if (item2 is ManaFreezer item)
			{
				list.Add(item);
			}
		}
		if (list.Count <= 0)
		{
			return;
		}
		foreach (ManaFreezer item3 in list)
		{
			ManaFreezeLevel = Math.Max(item3.FreezeLevel, ManaFreezeLevel);
		}
	}
