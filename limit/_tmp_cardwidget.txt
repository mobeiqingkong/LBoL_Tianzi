	private static string GetSimpleIllustrator(string id)
	{
		if (id.EndsWith("(Sketch)"))
		{
			return id.Substring(0, id.Length - 8);
		}
		if (id.EndsWith("(Community)"))
		{
			return id.Substring(0, id.Length - 11);
		}
		return id;
	}

	private static async UniTask<Texture2D> LoadTextureAsync(string pathClone)
	{
		ResourceRequest request = Resources.LoadAsync<Texture2D>(pathClone);
		await request.ToUniTask();
		await UniTask.NextFrame();
		return request.asset as Texture2D;
	}

	public void RefreshBgTextureForLoopOrder()
	{
		Card card = _card;
		if (card != null && card.CardType != CardType.Tool && card.CardType != CardType.Misfortune)
		{
			IReadOnlyList<ManaColor> readOnlyList;
			if (!card.IsPurified)
			{
				readOnlyList = card.Config.Colors;
			}
			else
			{
				IReadOnlyList<ManaColor> readOnlyList2 = new List<ManaColor>();
				readOnlyList = readOnlyList2;
			}
			if (readOnlyList.Count == 2)
			{
				SetBgTexture(_card);
				costWidget.SetCost(_card, _card.Zone == CardZone.Hand);
			}
		}
	}

	public static (Texture2D main, Texture2D sub) GetBgTexture(Card card)
	{
		if (card.CardType == CardType.Tool)
		{
			if (_toolBgTextureTable.TryGetValue((card.Config.Rarity, card.UseTransparentTexture), out var value))
			{
				return (main: value, sub: null);
			}
			Debug.LogError($"Cannot find BG sprite for '{card.Config.Rarity}'-Tool");
			return (main: null, sub: null);
		}
		if (card.CardType == CardType.Misfortune)
		{
			if (_misfortuneBgTextureTable.TryGetValue((card.Config.Rarity, card.UseTransparentTexture), out var value2))
			{
				return (main: value2, sub: null);
			}
			Debug.LogError($"Cannot find BG sprite for '{card.Config.Rarity}'-Tool");
			return (main: null, sub: null);
		}
		Rarity rarity = card.Config.Rarity;
		IReadOnlyList<ManaColor> readOnlyList;
		if (!card.IsPurified)
		{
			readOnlyList = card.Config.Colors;
		}
		else
		{
			IReadOnlyList<ManaColor> readOnlyList2 = new List<ManaColor>();
			readOnlyList = readOnlyList2;
		}
		IReadOnlyList<ManaColor> readOnlyList3 = readOnlyList;
		if (readOnlyList3.Count >= 3)
		{
			if (_rainbowBgTextureTable.TryGetValue((card.Config.Rarity, card.UseTransparentTexture), out var value3))
			{
				return (main: value3, sub: null);
			}
			Debug.LogError($"Cannot find BG texture for '{card.Config.Rarity}'-Rainbow");
			return (main: null, sub: null);
		}
		if (readOnlyList3.Count == 2)
		{
			ManaColor firstOut;
			ManaColor secondOut;
			if (GameMaster.IsLoopOrder)
			{
				ManaColors.GetLoopOrder(readOnlyList3[0], readOnlyList3[1], out firstOut, out secondOut);
			}
			else
			{
				firstOut = readOnlyList3[0];
				secondOut = readOnlyList3[1];
			}
			if (!_bgTextureTable.TryGetValue((firstOut, rarity, card.UseTransparentTexture), out var value4))
			{
				Debug.LogError($"Cannot find BG texture for '{firstOut}-{rarity}'");
			}
			if (!_bgTextureTable.TryGetValue((secondOut, rarity, card.UseTransparentTexture), out var value5))
			{
				Debug.LogError($"Cannot find BG texture for '{secondOut}-{rarity}'");
			}
			return (main: value4, sub: value5);
		}
		ManaColor item = ((readOnlyList3.Count > 0) ? readOnlyList3[0] : ManaColor.Any);
		if (!_bgTextureTable.TryGetValue((item, rarity, card.UseTransparentTexture), out var value6))
		{
			Debug.LogError($"Cannot find BG texture for '{readOnlyList3[0]}-{rarity}'");
		}
		return (main: value6, sub: null);
	}

	private static Sprite GetOwnerSprite(Card card)
	{
		string owner = card.Config.Owner;
		if (string.IsNullOrWhiteSpace(owner))
		{
			return null;
		}
		if (_ownerSpriteTable.TryGetValue(owner, out var value))
		{
			return value;
		}
		Debug.LogError("Cannot find owner sprite for '" + owner + "'");
		return null;
	}

	private static Sprite GetTypeSprite(Card card)
	{
		if (_typeSpriteTable.TryGetValue(card.Config.Type, out var value))
		{
			return value;
		}
		Debug.LogError($"Cannot find type sprite for '{card.Config.Type}'");
		return null;
	}

	private void OnEnable()
	{
		L10nManager.LocaleChanged += OnLocaleChanged;
		GameMaster.SettingsChanged += OnSettingsChanged;
		if (_card != null)
		{
			AddHandlers(_card);
		}
	}

	private void OnDisable()
	{
		L10nManager.LocaleChanged -= OnLocaleChanged;
		GameMaster.SettingsChanged -= OnSettingsChanged;
		if (_card != null)
		{
			RemoveHandlers(_card);
		}
		HideTooltip();
		DOTween.Kill(this);
	}

	private void OnDestroy()
	{
		if (_card != null)
		{
			RemoveHandlers(_card);

		}
	}

	{
		card.PropertyChanged -= OnPropertyChanged;

	}

	private void OnSettingsChanged(GameSettingsSaveData settings)
	{
		if (!this)
		{
			Debug.LogError("OnSettingsChanged invoked on destroyed: " + _rawPath);
		}
		else if (_card != null)
		{
			if ((bool)illustrator.gameObject)
			{
				illustrator.gameObject.SetActive(settings.ShowIllustrator);
			}
			RefreshCardImage();
		}
		else if ((bool)illustrator.gameObject)
		{
			illustrator.gameObject.SetActive(value: false);
		}
	}

	private void LazySetCard()
	{
		if (_card != null)
		{
			cardIdText.text = _card.GetType().Name;
			cardIdText.gameObject.SetActive(GameMaster.ShowCardId);
			otherInfo.text = ((_card.Config.DebugLevel > 0) ? "Debug" : null);
			otherInfo.gameObject.SetActive(_card.Config.DebugLevel > 0);
			RefreshCardImage();
			RefreshPackSticker();
			RefreshEnchantmentSticker();
			illustrator.gameObject.SetActive(GameMaster.ShowIllustrator);
			_dontShowMana = _card.IsForbidden;
			costWidget.gameObject.SetActive(!_dontShowMana);
			SetBgTexture(_card);
			Sprite ownerSprite = GetOwnerSprite(_card);
			cardOwnerIcon.sprite = ownerSprite;
			cardOwnerIcon.gameObject.SetActive(ownerSprite);
			Sprite typeSprite = GetTypeSprite(_card);
			cardTypeIcon.sprite = typeSprite;
			cardTypeIcon.gameObject.SetActive(typeSprite);
			baseLoyaltyObj.SetActive(_card.CardType == CardType.Friend);
			MarginAsFriend = _card.CardType == CardType.Friend && !NotReveal;
			_shouldChangeImageWhenUpgrade = !_card.Config.UpgradeImageId.IsNullOrEmpty();
			if (_shouldChangeImageWhenUpgrade)
			{
				_changedImageWhenUpgrade = _card.IsUpgraded;
			}
		}
	}

	private void RefreshCardImage()
	{
		string text = GameMaster.GetPreferredCardIllustrator(_card);
		if (_tempIllustratorId != null)
		{
			text = _tempIllustratorId;
		}
		if (text != null && text != "")
		{
			CurrentIllustrator = text;
			unfinished.gameObject.SetActive(value: false);
		}
		else if (_card.Config.Illustrator.IsNullOrEmpty())
		{
			CurrentIllustrator = "";
			unfinished.gameObject.SetActive(value: false);
		}
		else
		{
			CurrentIllustrator = ((!_card.Config.Illustrator.IsNullOrEmpty()) ? _card.Config.Illustrator : "");
			unfinished.gameObject.SetActive(value: false);
		}
		string text2 = ((_card.Config.UpgradeImageId.IsNullOrEmpty() || !_card.IsUpgraded) ? (_card.Config.ImageId.IsNullOrEmpty() ? _card.Id : _card.Config.ImageId) : _card.Config.UpgradeImageId);
		RefreshCardImageTexture(text2 + text);
	}

	private void RefreshCardImageTexture(string imageKey)
	{
		_cardImageLoadVersion++;
		int cardImageLoadVersion = _cardImageLoadVersion;
		Texture texture = ResourcesHelper.TryGetCardImage(imageKey);
		if ((bool)texture)
		{
			cardImage.texture = texture;
			CardImageLoadComplete();
			return;
		}
		cardImage.texture = defaultCardImage;
		if (GameEntry.PostJobEnd)
		{
			CardImageLoadComplete();
			return;
		}
		ShowCardImageLoading();
		WaitForCardImageAsync(imageKey, cardImageLoadVersion).Forget();
	}

	private async UniTask WaitForCardImageAsync(string imageKey, int version)
	{
		while ((bool)this && version == _cardImageLoadVersion)
		{
			await UniTask.Delay(TimeSpan.FromSeconds(UnityEngine.Random.Range(0.5f, 2f)));
			if (!this || version != _cardImageLoadVersion)
			{
				break;
			}
			Texture texture = ResourcesHelper.TryGetCardImage(imageKey);
			if ((bool)texture)
			{
				cardImage.texture = texture;
				CardImageLoadComplete();
				break;
			}
			if (GameEntry.PostJobEnd)
			{
				CardImageLoadComplete();
				break;
			}
		}
	}

	private void ShowCardImageLoading()
	{
		if ((bool)cardLoading)
		{
			cardLoading.DOKill();
			Color color = cardLoading.color;
			color.a = 1f;
			cardLoading.color = color;
			cardLoading.gameObject.SetActive(value: true);
		}
	}

	public void OnActivating()
	{
		if (!this)
		{
			Debug.Log("Activating " + _card.DebugName + " error: Widget " + _rawPath + " already destroyed.");
		}
		else
		{
			PlayActivatingEdge();
			DOTween.Sequence().Append(base.transform.DOScale(1.2f, 0.2f)).AppendInterval(0.3f)
				.Append(base.transform.DOScale(1f, 0.2f))
				.SetUpdate(isIndependentUpdate: true)
				.SetLink(base.gameObject);
		}
	}

	private void SetProperties()
	{
		if (_card == null)
		{
			return;
		}
		TextMeshProUGUI textMeshProUGUI = nameText;
		object text;
		if (!_card.IsUpgraded)
		{
			text = _card.Name;
		}
		else
		{
			int? upgradeCounter = _card.UpgradeCounter;
			text = ((upgradeCounter.HasValue && upgradeCounter.GetValueOrDefault() > 0) ? (_card.Name + "+" + _card.UpgradeCounter.Value) : (_card.Name + "+"));
		}
		textMeshProUGUI.text = (string)text;
		typeText.text = ("CardType." + _card.Config.Type).Localize();
		descriptionText.text = _card.Description;
		nameText.color = (_card.IsUpgraded ? GlobalConfig.UpgradedGreen : Color.white);
		if (_card.EnchantmentChanged)
		{
			RefreshEnchantmentSticker();
		}
		costWidget.SetCost(_card, _card.Zone == CardZone.Hand);
		if (_card.CardType == CardType.Friend)
		{
			baseLoyaltyObj.SetActive(!_card.Summoned);
			if (!_card.Summoned)
			{
				baseLoyalty.text = _card.Loyalty.ToString();
			}
		}
		if (NotReadyInMuseum)
		{
			descriptionText.text = "Museum.CardNotReady".Localize();
		}
		else if (NotReveal)
		{
			descriptionText.text = "???";
			nameText.text = "Museum.UnRevealCard".Localize();
		}
		SetBgTexture(_card);
		if (_shouldChangeImageWhenUpgrade && _card.IsUpgraded && !_changedImageWhenUpgrade)
		{
			RefreshCardImage();
			_changedImageWhenUpgrade = true;
		}
		MarginAsFriend = _card.CardType == CardType.Friend && !NotReveal;
		descriptionText.ForceMeshUpdate();
		int lineCount = descriptionText.textInfo.lineCount;
		if (NotReveal)
		{
			descriptionText.alignment = TextAlignmentOptions.Center;
		}
		else
		{
			descriptionText.alignment = ((lineCount > 1 || _card.CardType == CardType.Friend) ? TextAlignmentOptions.Left : TextAlignmentOptions.Center);
		}
	}

	public void SetCardEdge(EdgeStatus status)
	{
		Edge = status;
		switch (status)
		{
		case EdgeStatus.None:
			SetEdgeAfford(b: false);
			SetEdgeAffordKicker(b: false);
			SetEdgeHigh(b: false);
			SetEdgeHighKicker(b: false);
			break;
		case EdgeStatus.Afford:
			SetEdgeAfford(b: true);
			SetEdgeAffordKicker(b: false);
			SetEdgeHigh(b: false);
			SetEdgeHighKicker(b: false);
			break;
		case EdgeStatus.AffordKicker:
			SetEdgeAfford(b: false);
			SetEdgeAffordKicker(b: true);
			SetEdgeHigh(b: false);
			SetEdgeHighKicker(b: false);
			break;
		case EdgeStatus.High:
			SetEdgeAfford(b: false);
			SetEdgeAffordKicker(b: false);
			SetEdgeHigh(b: true);
			SetEdgeHighKicker(b: false);
			break;
		case EdgeStatus.HighKicker:
			SetEdgeAfford(b: false);
			SetEdgeAffordKicker(b: false);
			SetEdgeHigh(b: false);
			SetEdgeHighKicker(b: true);
			break;
		default:
			throw new ArgumentOutOfRangeException("status", status, null);
		}
	}

	private void SetEdgeAfford(bool b)

	{

	}

	private void ShowPackSticker(string packId, bool on = true)
	{
		stickerPack.gameObject.SetActive(value: true);
		stickerPack.sprite = ResourcesHelper.TryGetSticker(packId + "Sticker", on);
	}

	public void RefreshPackSticker()
	{
		if (_card.Config.Pack.IsNullOrEmpty())
		{
			HidePackSticker();
			return;
		}
		List<string> list = null;
		if (Singleton<GameMaster>.Instance.CurrentGameRun != null)
		{
			list = Singleton<GameMaster>.Instance.CurrentGameRun.Packs;
		}
		else if (Singleton<GameMaster>.Instance.CurrentProfile != null)
		{
			list = Singleton<GameMaster>.Instance.CurrentProfile.Settings.Packs;
		}
		bool on = list?.Contains(_card.Config.Pack) ?? false;
		ShowPackSticker(_card.Config.Pack, on);
	}

	private void HideEnchantmentSticker()
	{
		stickerEnchantment.gameObject.SetActive(value: false);
	}

	private void ShowEnchantmentSticker(string enchantmentId)
	{
		stickerEnchantment.gameObject.SetActive(value: true);
		stickerEnchantment.sprite = ResourcesHelper.TryGetSticker(enchantmentId + "Sticker");
	}

	public void RefreshEnchantmentSticker()
	{
		if (_card.EnchantmentStickerId.IsNullOrEmpty())
		{
			HideEnchantmentSticker();
		}
		else
		{
			ShowEnchantmentSticker(_card.EnchantmentStickerId);
		}
		_card.EnchantmentChanged = false;
	}

	public void ShowDeckIndex(int index, bool top = false, bool bottom = false)
	{
		string text = index.ToString();
		if (top)
		{
			text = ((!bottom) ? (text + "Game.TopSuffix".Localize()) : (text + "Game.TopBottomSuffix".Localize()));
		}
		else if (bottom)
		{
			text += "Game.BottomSuffix".Localize();
		}
		deckIndex.text = text;
		deckIndex.gameObject.SetActive(value: true);
	}

	public void HideDeckIndex()
	{
		deckIndex.gameObject.SetActive(value: false);
	}

	private void CardImageLoadComplete()
	{
		if (!cardLoading || !cardLoading.gameObject.activeSelf)
		{
			return;
		}
		cardLoading.DOKill();
		cardLoading.DOFade(0f, 0.3f).OnComplete(delegate
		{
			if ((bool)cardLoading)
			{
				cardLoading.gameObject.SetActive(value: false);
			}
		});
	}
