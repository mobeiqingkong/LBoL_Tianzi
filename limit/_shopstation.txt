using System;
using System.Collections;
using System.Collections.Generic;
using System.Linq;
using LBoL.Base;
using LBoL.Core.Cards;
using LBoL.Core.SaveData;
using UnityEngine;

namespace LBoL.Core.Stations;

public sealed class ShopStation : Station
{
	public int DiscountCardNo;

	public override StationType Type => StationType.Shop;

	public List<ShopItem<Card>> ShopCards { get; private set; }

	private List<string> ShopCardIds => ShopCards.Select((ShopItem<Card> shopCard) => shopCard.Content.Id).ToList();

	public List<ShopItem<Exhibit>> ShopExhibits { get; private set; }

	public bool CanUseCardService { get; internal set; } = true;


	public int UpgradeDeckCardPrice => base.GameRun.UpgradeDeckCardPrice;

	public int RemoveDeckCardPrice => base.GameRun.RemoveDeckCardPrice;

	protected internal override void OnEnter()
	{
		List<ShopItem<Card>> list = new List<ShopItem<Card>>();
		DiscountCardNo = base.GameRun.ShopRng.NextInt(0, 7);
		Card[] shopNormalCards = base.Stage.GetShopNormalCards();
		foreach (Card card in shopNormalCards)
		{
			list.Add(new ShopItem<Card>(base.GameRun, card, GetPrice(card, list.Count == DiscountCardNo)));
		}
		list[DiscountCardNo].IsDiscounted = true;
		shopNormalCards = base.Stage.GetShopToolCards(2);
		foreach (Card card2 in shopNormalCards)
		{
			list.Add(new ShopItem<Card>(base.GameRun, card2, GetPrice(card2, isDiscounted: false)));
		}
		ShopCards = list;
		List<ShopItem<Exhibit>> list2 = new List<ShopItem<Exhibit>>();
		for (int j = 0; j < 3; j++)
		{
			Exhibit shopExhibit = base.Stage.GetShopExhibit(j == 2);
			list2.Add(new ShopItem<Exhibit>(base.GameRun, shopExhibit, GetPrice(shopExhibit)));
		}
		ShopExhibits = list2;
	}

	private int GetPrice(Card card, bool isDiscounted)
	{
		float num = ((card.CardType != CardType.Tool) ? ((float)(card.Config.Rarity switch
		{
			Rarity.Common => GlobalConfig.CardPrices[0], 
			Rarity.Uncommon => GlobalConfig.CardPrices[1], 
			Rarity.Rare => GlobalConfig.CardPrices[2], 
			_ => throw new ArgumentOutOfRangeException(), 
		})) : ((float)(card.Config.Rarity switch
		{
			Rarity.Common => GlobalConfig.ToolPrices[0], 
			Rarity.Uncommon => GlobalConfig.ToolPrices[1], 
			Rarity.Rare => GlobalConfig.ToolPrices[2], 
			_ => throw new ArgumentOutOfRangeException(), 
		})));
		float num2 = base.GameRun.ShopRng.NextFloat(-0.08f, 0f) + 1f;
		num *= num2;
		if (isDiscounted)
		{
			num /= 2f;
		}
		return Mathf.RoundToInt(num);
	}

	private int GetPrice(Exhibit exhibit)
	{
		float num = exhibit.Config.Rarity switch
		{
			Rarity.Common => GlobalConfig.ExhibitPrices[0], 
			Rarity.Uncommon => GlobalConfig.ExhibitPrices[1], 
			Rarity.Rare => GlobalConfig.ExhibitPrices[2], 
			_ => throw new ArgumentOutOfRangeException(), 
		};
		float num2 = base.GameRun.ShopRng.NextFloat(-0.08f, 0f) + 1f;
		return Mathf.RoundToInt(num * num2);
	}

	public void BuyCard(ShopItem<Card> cardItem)
	{
		int num = ShopCards.IndexOf(cardItem);
		if (num < 0)
		{
			throw new ArgumentException(cardItem.Content.Name + " not in shop.");
		}
		if (cardItem.IsSoldOut)
		{
			throw new ArgumentException(cardItem.Content.Name + " is already sold out.");
		}
		base.GameRun.BuyCard(cardItem);
		if (base.GameRun.ShopResupplyFlag > 0)
		{
			Card card = base.Stage.SupplyShopCard(cardItem.Content, ShopCardIds);
			ShopCards[num] = new ShopItem<Card>(base.GameRun, card, GetPrice(card, isDiscounted: false));
		}
		RefreshAfterBought();
	}

	public IEnumerator BuyExhibitRunner(ShopItem<Exhibit> exhibitItem)
	{
		int index = ShopExhibits.IndexOf(exhibitItem);
		if (index < 0)
		{
			throw new ArgumentException(exhibitItem.Content.Name + " not in shop.");
		}
		if (exhibitItem.IsSoldOut)
		{
			throw new ArgumentException(exhibitItem.Content.Name + " is already sold out.");
		}
		yield return base.GameRun.BuyExhibitRunner(exhibitItem, new VisualSourceData
		{
			SourceType = VisualSourceType.Shop,
			Index = index
		});
		foreach (ShopItem<Card> item in ShopCards.Where((ShopItem<Card> shopCard) => !shopCard.IsSoldOut))
		{
			base.GameRun.UpgradeNewDeckCardOnFlags(item.Content);
		}
		if (base.GameRun.ShopResupplyFlag > 0)
		{
			Exhibit shopExhibit = base.Stage.GetShopExhibit(shopOnly: false);
			ShopExhibits[index] = new ShopItem<Exhibit>(base.GameRun, shopExhibit, GetPrice(shopExhibit));
		}
		RefreshAfterBought();
	}

	public void UpgradeDeckCard(Card card)
	{
		GameRunController gameRun = base.GameRun;
		if (gameRun.Money < UpgradeDeckCardPrice)
		{
			throw new InvalidOperationException("Insufficient fund");
		}
		gameRun.ConsumeMoney(UpgradeDeckCardPrice);
		gameRun.Stats.ShopConsumed += UpgradeDeckCardPrice;
		gameRun.UpgradeDeckCard(card, triggerVisual: true);
		CanUseCardService = false;
		RefreshAfterBought();
	}

	public void RemoveDeckCard(Card card)
	{
		GameRunController gameRun = base.GameRun;
		if (gameRun.Money < RemoveDeckCardPrice)
		{
			throw new InvalidOperationException("Insufficient fund");
		}
		gameRun.ConsumeMoney(RemoveDeckCardPrice);
		gameRun.Stats.ShopConsumed += RemoveDeckCardPrice;
		gameRun.RemoveDeckCard(card);
		CanUseCardService = false;
		int shopRemoveCardCounter = gameRun.ShopRemoveCardCounter + 1;
		gameRun.ShopRemoveCardCounter = shopRemoveCardCounter;
		RefreshAfterBought();
	}

	public void RefreshAfterBought()
	{
		foreach (ShopItem<Card> shopCard in ShopCards)
		{
			shopCard.Content.NotifyChanged();
		}
		foreach (ShopItem<Exhibit> shopExhibit in ShopExhibits)
		{
			shopExhibit.Content.NotifyChanged();
		}
	}

	internal override StationRecord GenerateRecord()
	{
		return new StationRecord
		{
			Type = StationType.Shop
		};
	}
}
