using System.Collections.Generic;
using LBoL.Base;

namespace LBoL.EntityLib.Cards.Character.Marisa;

public sealed class Potion : Card
{
	private sealed class PotionAchievementCounter : ICustomCounter
	{
		private int _counter;

		public CustomCounterResetTiming AutoResetTiming => CustomCounterResetTiming.PlayerActionStart;

		public void Increase(BattleController battle)
		{
			_counter++;
			if (_counter >= 8 && battle.GameRun.IsAutoSeed && battle.GameRun.JadeBoxes.Empty())
			{
				battle.GameRun.AchievementHandler.UnlockAchievement(AchievementKey.Potion);
			}
		}

		public void Reset(BattleController battle)
		{
			_counter = 0;
		}
	}

	protected override int AdditionalDamage
	{
		get
		{
			if (base.Battle == null)
			{
				return 0;
			}
			return (base.Battle.Player?.GetStatusEffect<PotionBaseDamageSe>())?.Level ?? 0;
		}
	}

	public override IEnumerable<BattleAction> OnDraw()
	{
		return EnterHandReactor();
	}

	public override IEnumerable<BattleAction> OnMove(CardZone srcZone, CardZone dstZone)
	{
		if (dstZone != CardZone.Hand)
		{
			return null;
		}
		return EnterHandReactor();
	}

	protected override void OnEnterBattle(BattleController battle)
	{
		if (base.Zone == CardZone.Hand)
		{
			React((LazySequencedReactor)AddToHandReactor);
		}
	}

	private IEnumerable<BattleAction> AddToHandReactor()
	{
		NotifyActivating();
		if (base.Battle.UpgradeTempCardsLevel > 0 && base.CanUpgradeAndPositive)
		{
			Upgrade();
		}
		List<DamageAction> list = new List<DamageAction>();
		foreach (BattleAction action in EnterHandReactor())
		{
			yield return action;
			if (action is DamageAction item)
			{
				list.Add(item);
			}
		}
		if (list.NotEmpty())
		{
			yield return new StatisticalTotalDamageAction(list);
		}
		base.Battle.IncreaseCounter<PotionAchievementCounter>();
	}

	protected override IEnumerable<BattleAction> Actions(UnitSelector selector, ManaGroup consumingMana, Interaction precondition)
	{
		return EnterHandReactor(ensureInHand: false);
	}

	private IEnumerable<BattleAction> EnterHandReactor(bool ensureInHand = true)
	{
		if (base.Battle.BattleShouldEnd)
		{
			yield break;
		}
		if (ensureInHand && base.Zone != CardZone.Hand)
		{
			Debug.LogWarning(Name + " is not in hand.");
			yield break;
		}
		yield return new GainManaAction(base.Mana);
		int num = 1;
		SplitPotionSe statusEffect = base.Battle.Player.GetStatusEffect<SplitPotionSe>();
		if (statusEffect != null)
		{
			num += statusEffect.Level;
		}
		EnemyUnit[] array = base.Battle.EnemyGroup.Alives.SampleManyOrAll(num, base.GameRun.BattleRng);
		EnemyUnit[] array2 = array;
		foreach (EnemyUnit target in array2)
		{
			if (target != null && target.IsAlive)
			{
				yield return AttackAction(target);
				PoisonPotionSe statusEffect2 = base.Battle.Player.GetStatusEffect<PoisonPotionSe>();
				if (statusEffect2 != null && target.IsAlive)
				{
					statusEffect2.NotifyActivating();
					yield return DebuffAction<Poison>(target, statusEffect2.Level, 0, 0, 0, startAutoDecreasing: true, 0.1f);
				}
				ColdPotionSe statusEffect3 = base.Battle.Player.GetStatusEffect<ColdPotionSe>();
				if (statusEffect3 != null && target.IsAlive)
				{
					statusEffect3.NotifyActivating();
					yield return DebuffAction<Cold>(target, 0, 0, 0, 0, startAutoDecreasing: true, 0.1f);
				}
			}
		}
		yield return new ExileCardAction(this);
		ChargingPotionSe statusEffect4 = base.Battle.Player.GetStatusEffect<ChargingPotionSe>();
		if (statusEffect4 != null)
		{
			statusEffect4.NotifyActivating();
			yield return BuffAction<Charging>(statusEffect4.Level);
		}
		PotionDefenseSe statusEffect5 = base.Battle.Player.GetStatusEffect<PotionDefenseSe>();
		if (statusEffect5 != null)
		{
			statusEffect5.NotifyActivating();
			yield return DefenseAction(0, statusEffect5.Level, BlockShieldType.Direct, cast: false);
		}
		base.Battle.IncreaseCounter<PotionAchievementCounter>();
		yield return new SpecialResolveAction(this);
	}
}

======== LBoL.Core.Battle.BattleActions.DrawManyCardAction ========
using System.Collections.Generic;
using LBoL.Core.Cards;

namespace LBoL.Core.Battle.BattleActions;

public class DrawManyCardAction : SimpleAction
{
	private readonly int _count;

	private readonly List<Card> _cards = new List<Card>();

	public IReadOnlyList<Card> DrawnCards => _cards.AsReadOnly();

	public override bool IsCanceled => false;

	public DrawManyCardAction(int count)
	{
		_count = count;
	}

	protected override void ResolvePhase()
	{
		React(new Reactor(ResolvePhaseEnumerator()));
	}

	private IEnumerable<BattleAction> ResolvePhaseEnumerator()
	{
		int i = 0;
		while (i < _count)
		{
			DrawCardAction draw = new DrawCardAction();
			yield return draw;
			if (draw.Args.Card != null)
			{
				_cards.Add(draw.Args.Card);
			}
			int num = i + 1;
			i = num;
		}
	}
}

======== LBoL.Core.Battle.BattleActions.MoveCardAction ========
using System.Collections.Generic;
using System.Linq;
using LBoL.Base.Extensions;
using LBoL.Core.Cards;

namespace LBoL.Core.Battle.BattleActions;

public class MoveCardAction : EventBattleAction<CardMovingEventArgs>
{
	public CardTransitionType TransitionType { get; private set; }

	public DreamCardsAction DreamCardsAction { get; internal set; }

	public MoveCardAction(Card card, CardZone dstZone)
	{
		BattleController.MoveCardCheck(card, dstZone);
		base.Args = new CardMovingEventArgs
		{
			Card = card,
			SourceZone = card.Zone,
			DestinationZone = dstZone
		};
	}

	internal override IEnumerable<Phase> GetPhases()
	{
		if (base.Args.DestinationZone == CardZone.Hand && base.Battle.HandZone.Count == base.Battle.MaxHand)
		{
			while (base.Battle.HandZone.Count == base.Battle.MaxHand && base.Battle.HandZone.Any((Card card) => card.IsAutoExile))
			{
				Card firstAutoExile = base.Battle.HandZone.FirstOrDefault((Card card) => card.IsAutoExile);
				if (firstAutoExile != null)
				{
					yield return CreatePhase("AutoExile", delegate
					{
						base.Battle.React(new ExileCardAction(firstAutoExile), base.Args.Card, ActionCause.AutoExile);
					});
				}
			}
		}
		yield return CreateEventPhase("Moving", base.Args, base.Battle.CardMoving);
		CardZone srcZone = base.Args.Card.Zone;
		IEnumerable<BattleAction> reactor = null;
		yield return CreatePhase("Main", delegate
		{
			if (!base.Args.IsCanceled)
			{
				CancelCause cancelCause = base.Battle.MoveCard(base.Args.Card, base.Args.DestinationZone);
				if (cancelCause != 0)
				{
					base.Args.ForceCancelBecause(cancelCause);
				}
				else
				{
					reactor = base.Args.Card.OnMove(srcZone, base.Args.DestinationZone);
					if (reactor != null && base.Args.Card.OnMoveVisual)
					{
						TransitionType = CardTransitionType.SpecialBegin;
					}
				}
			}
		}, hasViewer: true);
		if (reactor != null)
		{
			List<DamageAction> damageActions = new List<DamageAction>();
			yield return CreatePhase("SpecialMove", delegate
			{
				React(new Reactor(StatisticalTotalDamageAction.WrapReactorWithStats(reactor, damageActions)), base.Args.Card, ActionCause.Card);
			});
			if (damageActions.NotEmpty())
			{
				yield return CreatePhase("Statistics", delegate
				{
					base.Battle.React(new StatisticalTotalDamageAction(damageActions), base.Args.Card, ActionCause.Card);
				});
			}
			if (base.Args.Card.OnMoveVisual)
			{
				TransitionType = CardTransitionType.SpecialEnd;
				yield return CreatePhase("SpecialMoved", delegate
				{
				}, hasViewer: true);
			}
		}
		yield return CreateEventPhase("Moved", base.Args, base.Battle.CardMoved);
	}
}
