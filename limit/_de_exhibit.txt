using System;
using System.Collections;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using JetBrains.Annotations;
using LBoL.Base;
using LBoL.Base.Extensions;
using LBoL.ConfigData;
using LBoL.Core.Attributes;
using LBoL.Core.Battle;
using LBoL.Core.Cards;
using LBoL.Core.Units;
using UnityEngine;

namespace LBoL.Core;

[Localizable]
public abstract class Exhibit : GameEntity, INotifyActivating
{
	private readonly WeakReference<PlayerUnit> _owner = new WeakReference<PlayerUnit>(null);

	private bool _active;

	private bool _blackout;

	private int _counter;

	private readonly GameEventHandlerHolder _gameRunHandlerHolder = new GameEventHandlerHolder();

	private readonly GameEventHandlerHolder _battleHandlerHolder = new GameEventHandlerHolder();

	public ExhibitConfig Config { get; private set; }

	[UsedImplicitly]
	public override EntityName SelfName => GetName().WithColor();

	public override string Description
	{
		get
		{
			try
			{
				string baseDescription = GetBaseDescription();
				return (baseDescription == null) ? "" : baseDescription.RuntimeFormat(base.FormatWrapper);
			}
			catch (Exception exception)
			{
				Debug.LogException(exception);
				return "<Error>";
			}
		}
	}

	public string FlavorText => LocalizeProperty("FlavorText", decorated: false, required: false);

	protected string ExtraDescription => LocalizeProperty("ExtraDescription", decorated: true);

	public string OwnerId { get; private set; }

	public virtual string IconName => base.Id;

	public virtual string OverrideIconName => null;

	internal override GameEventPriority DefaultEventPriority => (GameEventPriority)Config.Order;

	public PlayerUnit Owner
	{
		get
		{
			if (!_owner.TryGetTarget(out var target))
			{
				return null;
			}
			return target;
		}
		internal set
		{
			_owner.SetTarget(value);
		}
	}

	public int Value1 => Config.Value1 ?? throw new InvalidDataException(DebugName + " has no 'Value1' in config");

	public int Value2 => Config.Value2 ?? throw new InvalidDataException(DebugName + " has no 'Value2' in config");

	public int Value3 => Config.Value3 ?? throw new InvalidDataException(DebugName + " has no 'Value3' in config");

	[UsedImplicitly]
	public ManaGroup Mana => Config.Mana ?? throw new InvalidDataException("<" + base.Id + "> has empty Mana config");

	[UsedImplicitly]
	public BaseManaGroup BaseMana
	{
		get
		{
			ManaColor? baseManaColor = Config.BaseManaColor;
			if (baseManaColor.HasValue)
			{
				ManaColor valueOrDefault = baseManaColor.GetValueOrDefault();
				return new BaseManaGroup(ManaGroup.Single(valueOrDefault) * Config.BaseManaAmount);
			}
			return new BaseManaGroup(ManaGroup.Empty);
		}
	}

	public bool HasCounter => Config.HasCounter;

	[UsedImplicitly]
	public int InitialCounter => Config.InitialCounter.GetValueOrDefault();

	public virtual bool ShowCounter => true;

	public bool Active
	{
		get
		{
			return _active;
		}
		set
		{
			if (_active != value)
			{
				_active = value;
				NotifyChanged();
			}
		}
	}

	public bool Blackout
	{
		get
		{
			return _blackout;
		}
		set
		{
			if (_blackout != value)
			{
				_blackout = value;
				NotifyChanged();
			}
		}
	}

	public int Counter
	{
		get
		{
			return _counter;
		}
		set
		{
			int num = Math.Min(value, 999);
			if (_counter != num)
			{
				_counter = num;
				NotifyChanged();
			}
		}
	}

	public ExhibitLosableType LosableType => Config.LosableType;

	public virtual bool CanLoseInBattle => true;

	public bool IsSentinel => Config.IsSentinel;

	public int? CardInstanceId { get; protected internal set; }

	public BattleController Battle => base.GameRun?.Battle;

	public event Action Activating;

	protected override string LocalizeProperty(string key, bool decorated = false, bool required = true)
	{
		return TypeFactory<Exhibit>.LocalizeProperty(base.Id, key, decorated, required);
	}

	protected virtual IReadOnlyList<string> LocalizeListProperty(string key, bool required = true)
	{
		return TypeFactory<Exhibit>.LocalizeListProperty(base.Id, key, required);
	}

	protected override string GetBaseDescription()
	{
		return LocalizeProperty("Description", decorated: true, required: false);
	}

	public EntityName GetName()
	{
		return new EntityName("Exhibit." + base.Id, Name);
	}

	public override void Initialize()
	{
		base.Initialize();
		Config = ExhibitConfig.FromId(base.Id);
		if (Config == null)
		{
			throw new InvalidDataException("Cannot find exhibit config for <" + base.Id + ">");
		}
		OwnerId = (Config.Owner.IsNullOrWhiteSpace() ? null : Config.Owner);
		if (Config.HasCounter)
		{
			int? initialCounter = Config.InitialCounter;
			if (initialCounter.HasValue)
			{
				int valueOrDefault = initialCounter.GetValueOrDefault();
				Counter = valueOrDefault;
			}
		}
	}

	internal void TriggerAdding(PlayerUnit unit)
	{
		OnAdding(unit);
	}

	internal void TriggerAdded(PlayerUnit unit)
	{
		OnAdded(unit);
	}

	internal void TriggerRemoving(PlayerUnit unit)
	{
		OnRemoving(unit);
		_gameRunHandlerHolder.ClearEventHandlers();
	}

	internal void TriggerRemoved(PlayerUnit unit)
	{
		OnRemoved(unit);
	}

	internal IEnumerator TriggerGain(PlayerUnit unit)
	{
		IEnumerator enumerator = SpecialGain(unit);
		if (enumerator != null)
		{
			yield return enumerator;
		}
		else
		{
			OnGain(unit);
		}
	}

	internal void TriggerGainInstantly(PlayerUnit unit)
	{
		if (SpecialGain(unit) != null)
		{
			Debug.LogError("Cannot gain " + DebugName + " instantly (init-exhibit or debug-add) because it has special gain interaction, triggering OnGain() instead.");
		}
		OnGain(unit);
	}

	internal void TriggerLose(PlayerUnit unit)
	{
		if (Battle != null)
		{
			OnLeaveBattle();
			_battleHandlerHolder.ClearEventHandlers();
		}
		OnLose(unit);
	}

	internal void EnterBattle()
	{
		OnEnterBattle();
	}

	internal void LeaveBattle()
	{
		OnLeaveBattle();
		_battleHandlerHolder.ClearEventHandlers();
	}

	protected virtual void OnAdding(PlayerUnit player)
	{
	}

	protected virtual void OnAdded(PlayerUnit player)
	{
	}

	protected virtual void OnRemoving(PlayerUnit player)
	{
	}

	protected virtual void OnRemoved(PlayerUnit player)
	{
	}

	protected virtual IEnumerator SpecialGain(PlayerUnit unit)
	{
		return null;
	}

	protected virtual void OnGain(PlayerUnit player)
	{
	}

	protected virtual void OnLose(PlayerUnit player)
	{
	}

	protected virtual void OnEnterBattle()
	{
	}

	protected virtual void OnLeaveBattle()
	{
	}

	public void NotifyActivating()
	{
		this.Activating?.Invoke();
	}

	protected override void React(Reactor reactor)
	{
		if (Battle == null)
		{
			Debug.LogError("[Exhibit: " + DebugName + "] Cannot react outside battle");
		}
		else
		{
			Battle.React(reactor, this, ActionCause.Exhibit);
		}
	}

	public virtual IEnumerable<IDisplayWord> EnumerateDisplayWords(bool verbose)
	{
		return Library.InternalEnumerateDisplayWords(base.GameRun, Config.Keywords, Config.RelativeEffects, verbose);
	}

	public virtual IEnumerable<Card> EnumerateRelativeCards()
	{
		IReadOnlyList<string> readOnlyList = Config.RelativeCards ?? Array.Empty<string>();
		foreach (string item in readOnlyList)
		{
			Card card;
			if (item.Last() == '+')
			{
				string text = item;
				card = Library.CreateCard(text.Substring(0, text.Length - 1));
				card.Upgrade();
			}
			else
			{
				card = Library.CreateCard(item);
			}
			card.GameRun = base.GameRun;
			yield return card;
		}
	}

	protected void HandleGameRunEvent<TEventArgs>(GameEvent<TEventArgs> @event, GameEventHandler<TEventArgs> handler, GameEventPriority priority) where TEventArgs : GameEventArgs
	{
		_gameRunHandlerHolder.HandleEvent(@event, handler, priority);
	}

	protected void HandleGameRunEvent<TEventArgs>(GameEvent<TEventArgs> @event, GameEventHandler<TEventArgs> handler) where TEventArgs : GameEventArgs
	{
		HandleGameRunEvent(@event, handler, DefaultEventPriority);
	}

	protected void HandleBattleEvent<TEventArgs>(GameEvent<TEventArgs> @event, GameEventHandler<TEventArgs> handler, GameEventPriority priority) where TEventArgs : GameEventArgs
	{
		_battleHandlerHolder.HandleEvent(@event, handler, priority);
	}

	protected void HandleBattleEvent<TEventArgs>(GameEvent<TEventArgs> @event, GameEventHandler<TEventArgs> handler) where TEventArgs : GameEventArgs
	{
		HandleBattleEvent(@event, handler, DefaultEventPriority);
	}

	protected void ReactBattleEvent<TEventArgs>(GameEvent<TEventArgs> @event, EventSequencedReactor<TEventArgs> reactor, GameEventPriority priority) where TEventArgs : GameEventArgs
	{
		HandleBattleEvent(@event, delegate(TEventArgs args)
		{
			React(reactor(args));
		}, priority);
	}

	protected void ReactBattleEvent<TEventArgs>(GameEvent<TEventArgs> @event, EventSequencedReactor<TEventArgs> reactor) where TEventArgs : GameEventArgs
	{
		ReactBattleEvent(@event, reactor, DefaultEventPriority);
	}
}
