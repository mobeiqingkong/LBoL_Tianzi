		{
			if (displayWord is IOnTooltipDisplay onTooltipDisplay)
			{

			}
			return displayWord;
		}

		private static IEnumerable<string> AppendCustomKeywords(IReadOnlyList<string> keywords, bool displayVerbose, Card card)
		{
			return keywords.Concat(from kw in card.AllCustomKeywords()
				where !kw.isVerbose || displayVerbose
				select kw.kwSEid);
		}

	[HarmonyPatch(typeof(Card), "EnumerateAutoAppendKeywordNames")]
	private class Card_Description_Patch
	{
		private static void Postfix(Card __instance, ref IEnumerable<string> __result)
		{
			IEnumerable<IGrouping<KwDescPos, string>> source = (from kw in __instance.AllCustomKeywords()
				where kw.descPos != KwDescPos.DoNotDisplay
				select kw).GroupBy((CardKeyword kw) => kw.descPos, delegate(CardKeyword kw)
			{
				if (kw.hasExtendedKeywordName)
				{
					StatusEffect statusEffect = LBoL.Core.TypeFactory<StatusEffect>.CreateInstance(kw.kwSEid);
					if (statusEffect is IExtendedKeywordName extendedKeywordName)
					{
						return extendedKeywordName.ExtendedKeywordName(__instance);
					}
					ManualLogSource obj = Log.LogDev();
					if (obj != null)
					{
						obj.LogWarning((object)("Keyword kw expects extended keyword name but " + kw.kwSEid + " does not implement IExtendedKeywordName"));
					}
				}
				return LBoL.Core.TypeFactory<StatusEffect>.LocalizeProperty(kw.kwSEid, "Name", decorated: true, required: false);
			});
			__result = source.SelectMany(delegate(IGrouping<KwDescPos, string> g)
			{
				object result2;
				if (g.Key != KwDescPos.First)
				{
					result2 = Enumerable.Empty<string>();
				}
				else
				{
					result2 = g;
				}
				return (IEnumerable<string>)result2;
			}).Concat(__result).Concat(source.SelectMany(delegate(IGrouping<KwDescPos, string> g)
			{
				object result;
				if (g.Key != KwDescPos.Last)
				{
					result = Enumerable.Empty<string>();
				}
				else
				{
					result = g;
				}
				return (IEnumerable<string>)result;
			}));
		}
	}

		{
			return keywords != Keyword.None || card.AllCustomKeywords().Count() > 0;

	private static ConditionalWeakTable<Card, Dictionary<string, CardKeyword>> cwt_keywords = new ConditionalWeakTable<Card, Dictionary<string, CardKeyword>>();

	private static Dictionary<string, CardKeyword> GetKeywords(Card card)
	{
		return cwt_keywords.GetValue(card, (Card _) => new Dictionary<string, CardKeyword>());
	}

	public static IEnumerable<CardKeyword> AllCustomKeywords(this Card card)
	{
		if (cwt_keywords.TryGetValue(card, out var value))
		{
			return value.Values;
		}
		return new HashSet<CardKeyword>();
	}

	public static bool HasCustomKeyword(this Card card, string kwId)
	{
		return GetKeywords(card).ContainsKey(kwId);
	}

	public static void AddCustomKeyword(this Card card, CardKeyword keyword)
	{
		if (GetKeywords(card).TryGetValue(keyword.kwSEid, out var value))
		{
			value.Merge(keyword);
		}
		else
		{
			GetKeywords(card).Add(keyword.kwSEid, keyword);
		}
	}

	public static void AddCustomKeyword<T>(this Card card, T keyword) where T : CardKeyword
	{
		card.AddCustomKeyword((CardKeyword)keyword);
	}

	public static bool RemoveCustomKeyword(this Card card, CardKeyword keyword)
	{
		return GetKeywords(card).Remove(keyword.kwSEid);
	}

	public static bool TryGetCustomKeyword(this Card card, string kwId, out CardKeyword rezKeyword)
	{
		return GetKeywords(card).TryGetValue(kwId, out rezKeyword);
	}

	public static bool TryGetCustomKeyword<T>(this Card card, string kwId, out T rezKeyword) where T : CardKeyword
	{
		CardKeyword rezKeyword2;
		bool flag = card.TryGetCustomKeyword(kwId, out rezKeyword2);
		rezKeyword = null;
		if (flag)
		{
			rezKeyword = (T)rezKeyword2;
		}
		return flag;
	}

	public static CardKeyword GetCustomKeyword(this Card card, string kwId)
	{
		card.TryGetCustomKeyword(kwId, out var rezKeyword);
		return rezKeyword;
	}

	public static T GetCustomKeyword<T>(this Card card, string kwId) where T : CardKeyword
	{
		return card.GetCustomKeyword(kwId) as T;
	}
