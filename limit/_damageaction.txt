using System;
using System.Collections.Generic;
using System.Linq;
using LBoL.Base;
using LBoL.Base.Extensions;
using LBoL.Core.Cards;
using LBoL.Core.Units;
using UnityEngine;

namespace LBoL.Core.Battle.BattleActions;

public sealed class DamageAction : BattleAction
{
	private bool _dealingPassed;

	public DamageDealingEventArgs DealingArgs { get; }

	public DamageEventArgs[] DamageArgs { get; }

	public string GunName { get; }

	public GunType GunType { get; }

	public override bool IsModified
	{
		get
		{
			if (!_dealingPassed)
			{
				return DealingArgs.IsModified;
			}
			return DamageArgs.Any((DamageEventArgs arg) => arg.IsModified);
		}
	}

	public override string[] Modifiers
	{
		get
		{
			if (!_dealingPassed)
			{
				return DealingArgs.Modifiers;
			}
			return DamageArgs.SelectMany((DamageEventArgs arg) => arg.Modifiers).Distinct().ToArray();
		}
	}

	public override bool IsCanceled
	{
		get
		{
			if (!_dealingPassed)
			{
				return DealingArgs.IsCanceled;
			}
			return DamageArgs.All((DamageEventArgs arg) => arg.IsCanceled);
		}
	}

	public override CancelCause CancelCause
	{
		get
		{
			if (!_dealingPassed)
			{
				return DealingArgs.CancelCause;
			}
			return DamageArgs.Aggregate(CancelCause.None, (CancelCause current, DamageEventArgs args) => current | args.CancelCause);
		}
	}

	public DamageAction(Unit source, IEnumerable<Unit> targets, DamageInfo damageInfo, string gunName = "Instant", GunType gunType = GunType.Single)
	{
		Unit[] array = targets.ToArray();
		if (array.Any((Unit t) => t == null))
		{
			throw new ArgumentException("Creating DamageAction with null target");
		}
		DealingArgs = new DamageDealingEventArgs
		{
			Source = source,
			Targets = array,
			GunName = gunName,
			DamageInfo = damageInfo
		};
		DamageArgs = array.Select((Unit t) => new DamageEventArgs
		{
			Source = source,
			Target = t,
			GunName = gunName
		}).ToArray();
		GunName = gunName;
		GunType = gunType;
	}

	public DamageAction(Unit source, Unit unit, DamageInfo damageInfo, string gunName = "Instant", GunType gunType = GunType.Single)
		: this(source, new Unit[1] { unit }, damageInfo, gunName, gunType)
	{
	}

	public static DamageAction LoseLife(Unit target, int life, string gunName = "Instant")
	{
		return new DamageAction(target, target, DamageInfo.HpLose(life), gunName);
	}

	public static DamageAction Reaction(Unit target, int damage, string gunName = "Instant")
	{
		return new DamageAction(target, target, DamageInfo.Reaction(damage), gunName);
	}

	public override BattleAction SetCause(ActionCause cause)
	{
		base.SetCause(cause);
		DealingArgs.Cause = cause;
		DamageEventArgs[] damageArgs = DamageArgs;
		for (int i = 0; i < damageArgs.Length; i++)
		{
			damageArgs[i].Cause = cause;
		}
		return this;
	}

	public override BattleAction SetSource(GameEntity source)
	{
		base.SetSource(source);
		DealingArgs.ActionSource = source;
		DamageEventArgs[] damageArgs = DamageArgs;
		for (int i = 0; i < damageArgs.Length; i++)
		{
			damageArgs[i].ActionSource = source;
		}
		return this;
	}

	public override void ClearModifiers()
	{
		DealingArgs.ClearModifiers();
		DamageEventArgs[] damageArgs = DamageArgs;
		for (int i = 0; i < damageArgs.Length; i++)
		{
			damageArgs[i].ClearModifiers();
		}
	}

	internal override IEnumerable<Phase> GetPhases()
	{
		if (DealingArgs.Targets.All((Unit t) => t.IsDead))
		{
			Debug.LogWarning("[DamageAction] Targets [" + string.Join(", ", DealingArgs.Targets.Select((Unit t) => t.Name)) + "] all dead, action is canceled");
			yield break;
		}
		Unit source = DealingArgs.Source;
		if (source != null)
		{
			yield return CreateEventPhase("DamageDealing", DealingArgs, source.DamageDealing);
		}
		_dealingPassed = true;
		DamageInfo damageInfo = DealingArgs.DamageInfo;
		foreach (var item4 in DealingArgs.Targets.Zip(DamageArgs))
		{
			Unit item = item4.Item1;
			DamageEventArgs item2 = item4.Item2;
			item2.DamageInfo = damageInfo;
			if (item2.Target.IsInvalidTarget)
			{
				item2.ForceCancelBecause(CancelCause.InvalidTarget);
			}
			else
			{
				yield return CreateEventPhase("DamageReceiving", item2, item.DamageReceiving);
			}
		}
		yield return CreatePhase("MeasureDamage", delegate
		{
			DamageEventArgs[] damageArgs4 = DamageArgs;
			foreach (DamageEventArgs damageEventArgs7 in damageArgs4)
			{
				if (damageEventArgs7.IsCanceled)
				{
					break;
				}
				damageEventArgs7.DamageInfo = damageEventArgs7.Target.MeasureDamage(damageEventArgs7.DamageInfo);
				damageEventArgs7.IsModified = true;
			}
		});
		DamageEventArgs[] damageArgs = DamageArgs;
		foreach (DamageEventArgs damageEventArgs in damageArgs)
		{
			Unit source2 = damageEventArgs.Source;
			if (source2 != null && !damageEventArgs.IsCanceled)
			{
				yield return CreateEventPhase("DamageGiving", damageEventArgs, source2.DamageGiving);
			}
		}
		damageArgs = DamageArgs;
		foreach (DamageEventArgs damageEventArgs2 in damageArgs)
		{
			if (!damageEventArgs2.IsCanceled)
			{
				yield return CreateEventPhase("DamageTaking", damageEventArgs2, damageEventArgs2.Target.DamageTaking);
			}
		}
		yield return CreatePhase("Main", delegate
		{
			DamageEventArgs[] damageArgs3 = DamageArgs;
			foreach (DamageEventArgs damageEventArgs6 in damageArgs3)
			{
				if (damageEventArgs6.IsCanceled)
				{
					break;
				}
				damageEventArgs6.DamageInfo = base.Battle.Damage(damageEventArgs6.Source, damageEventArgs6.Target, damageEventArgs6.DamageInfo, base.Source);
				damageEventArgs6.IsModified = true;
			}
		}, hasViewer: true);
		List<(Unit, DieCause)> deadUnits = new List<(Unit, DieCause)>();
		DamageEventArgs[] damageArgs2 = DamageArgs;
		foreach (DamageEventArgs damageEventArgs3 in damageArgs2)
		{
			Unit target = damageEventArgs3.Target;
			if (target.Hp == 0 && target.Status == UnitStatus.Alive)
			{
				target.Status = UnitStatus.Dying;
				DieCause item3 = damageEventArgs3.DamageInfo.DamageType switch
				{
					DamageType.HpLose => DieCause.LoseHp, 
					DamageType.Reaction => DieCause.Reaction, 
					DamageType.Attack => DieCause.Attack, 
					_ => throw new ArgumentOutOfRangeException(), 
				};
				deadUnits.Add((target, item3));
				damageEventArgs3.Kill = true;
			}
		}
		if (deadUnits.Count > 0)
		{
			yield return CreatePhase("Dying", delegate
			{
				React(new DieAction(deadUnits, DealingArgs.Source, DealingArgs.ActionSource), base.Source);
			});
		}
		damageArgs = DamageArgs;
		foreach (DamageEventArgs damageEventArgs4 in damageArgs)
		{
			if (!damageEventArgs4.IsCanceled)
			{
				Unit source3 = damageEventArgs4.Source;
				if (source3 != null)
				{
					yield return CreateEventPhase("DamageDealt", damageEventArgs4, source3.DamageDealt);
				}
			}
		}
		damageArgs = DamageArgs;
		foreach (DamageEventArgs damageEventArgs5 in damageArgs)
		{
			if (!damageEventArgs5.IsCanceled)
			{
				yield return CreateEventPhase("DamageReceived", damageEventArgs5, damageEventArgs5.Target.DamageReceived);
			}
		}
	}

	public override string ExportDebugDetails()
	{
		if (!_dealingPassed)
		{
			return DealingArgs.ExportDebugDetails();
		}
		return string.Join(", ", DamageArgs.Select((DamageEventArgs arg) => arg.ExportDebugDetails()));
	}
}
