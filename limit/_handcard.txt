using System;
using LBoL.Base.Extensions;
using LBoL.Core;
using LBoL.Core.Cards;
using LBoL.Presentation.InputSystemExtend;
using LBoL.Presentation.UI.Panels;
using LBoL.Presentation.UI.Widgets;
using TMPro;
using UnityEngine;

namespace LBoL.Presentation.UI.ExtraWidgets;

public class HandCard : MonoBehaviour
{
	public Transform cardRoot;

	[SerializeField]
	private Transform placeHolder;

	[SerializeField]
	private RectTransform rectTransform;

	[SerializeField]
	private TextMeshProUGUI shortcutText;

	private BattleManaPanel _battleManaPanel;

	private float _tickTime;

	private float _accumulatedTime;

	private bool _isHovered;

	private bool _isActiveHand;

	public const float NormalScale = 0.7f;

	public const float HoverScale = 1.3f;

	public const float ActiveHandScale = 1f;

	public const float SpecialReactingScale = 1.2f;

	private bool _showShortcut;

	private bool _specialReacting;

	private bool _pendingUse;

	private bool isDisappering;

	private const float ApproachingRatePerTick = 0.12f;

	private const float TiltApproachingRatePerTick = 0.05f;

	private const int TickPerSecond = 200;

	private const float TickDuration = 0.005f;

	private const float MaxAngle = 15f;

	private const float RiseRate = 0.001f;

	public CanvasGroup CanvasGroup => CardWidget.CanvasGroup;

	public bool IsActiveHand
	{
		get
		{
			return _isActiveHand;
		}
		set
		{
			_isActiveHand = value;
			if (value)
			{
				AudioManager.Card(1);
				CanvasGroup.blocksRaycasts = false;
				MoveToParent(ActiveHandParent);
			}
			else
			{
				AudioManager.Card(2);
				CanvasGroup.blocksRaycasts = true;
				MoveToParent(_isHovered ? HoveredParent : NormalParent);
				CostMoreLeft = ShouldShowLeftCost;
			}
		}
	}

	public Transform NormalParent { get; set; }

	public Transform HoveredParent { get; set; }

	public Transform ActiveHandParent { get; set; }

	public Vector3 NormalPosition { get; set; }

	public Quaternion NormalRotation { get; set; }

	public Vector3 HoveredPosition { get; set; }

	public Quaternion HoveredRotation { get; set; }

	public Vector3 ActiveHandPosition { get; set; }

	public Quaternion ActiveHandRotation { get; set; }

	public Vector3 SpecialReactingPosition { get; set; }

	public Quaternion SpecialReactingRotation { get; set; }

	public int HandIndex { get; set; }

	public bool ShowShortcut
	{
		get
		{
			return _showShortcut;
		}
		set
		{
			_showShortcut = value;
			shortcutText.gameObject.SetActive(value);
			shortcutText.text = UiManager.GetHandShortcutDisplayString(HandIndex);
		}
	}

	public bool IsMostRightHand => CardWidget.Card?.IsMostRightHand ?? false;

	public bool ShouldShowLeftCost
	{
		get
		{
			if (GameMaster.CostMoreLeft && !IsMostRightHand)
			{
				return !IsActiveHand;
			}
			return false;
		}
	}

	public bool CostMoreLeft
	{
		get
		{
			return CardWidget.CostMoreLeft;
		}
		set
		{
			CardWidget.CostMoreLeft = value;
		}
	}

	public Vector3 PendingUsePosition { get; set; }

	public Vector3 TargetWorldPosition
	{
		get
		{
			Transform parent = base.transform.parent;
			Vector3 position = (_pendingUse ? PendingUsePosition : ((!_specialReacting) ? (_isActiveHand ? ActiveHandPosition : (_isHovered ? HoveredPosition : NormalPosition)) : SpecialReactingPosition));
			return parent.TransformPoint(position);
		}
	}

	public bool SpecialReacting
	{
		get
		{
			return _specialReacting;
		}
		set
		{
			_specialReacting = value;
		}
	}

	public bool PendingUse
	{
		get
		{
			return _pendingUse;
		}
		set
		{
			_pendingUse = value;
			CardWidget.ShowManaHand = !_pendingUse;
		}
	}

	public bool IsDisappearing
	{
		get
		{
			return isDisappering;
		}
		set
		{
			if (value)
			{
				GamepadCardController componentInChildren = GetComponentInChildren<GamepadCardController>();
				componentInChildren.SetNavigationOrigin(value: false);
				componentInChildren.IsHandCard = false;
			}
			isDisappering = value;
		}
	}

	public Card Card => CardWidget.Card;

	public CardWidget CardWidget { get; set; }

	private Vector2 TiltVector2 { get; set; }

	private Quaternion StandardRotation { get; set; }

	public bool CanUse { get; private set; }

	public bool CanKickerUse { get; private set; }

	public void SetToInactiveWithoutAudio()
	{
		if (_isActiveHand)
		{
			_isActiveHand = false;
			CanvasGroup.blocksRaycasts = true;
			MoveToParent(_isHovered ? HoveredParent : NormalParent);
		}
	}

	private void Awake()
	{
		if (Singleton<InputDeviceManager>.Instance.CurrentInputDevice == InputDeviceType.Gamepad)
		{
			ShowShortcut = false;
		}
		else
		{
			ShowShortcut = GameMaster.ShowShortcut;
		}
	}

	private void UpdateGamepadTipVisibility()
	{
		if (!(CardWidget == null))
		{
			bool active = Singleton<InputDeviceManager>.Instance.CurrentInputDevice == InputDeviceType.Gamepad;
			CardWidget.GamepadTipBattle.SetActive(active);
			CardWidget.GamepadTipNormal.SetActive(value: false);
		}
	}

	private void OnEnable()
	{
		Singleton<InputDeviceManager>.Instance.onInputDeviceChanged.AddListener(OnInputDeviceChanged);
	}

	public void RefreshStatus()
	{
		CardWidget.RefreshStatus();
		if (ShowShortcut)
		{
			shortcutText.text = UiManager.GetHandShortcutDisplayString(HandIndex);
		}
		CardWidget.CostMoreLeft = ShouldShowLeftCost;
	}

	private void Update()
	{
		if (!IsDisappearing)
		{
			_accumulatedTime += Time.unscaledDeltaTime;
			if (_specialReacting)
			{
				Approach(SpecialReactingPosition, SpecialReactingRotation, 1.2f);
			}
			else if (_pendingUse)
			{
				Approach(PendingUsePosition, ActiveHandRotation, 1f);
			}
			else if (_isActiveHand)
			{
				Approach(ActiveHandPosition, ActiveHandRotation, 1f, tilting: true);
			}
			else if (_isHovered)
			{
				Approach(HoveredPosition, HoveredRotation, 1.3f);
			}
			else
			{
				Approach(NormalPosition, NormalRotation, 0.7f);
			}
		}
	}

	private void OnDestroy()
	{
		if (placeHolder != null)
		{
			UnityEngine.Object.Destroy(placeHolder.gameObject);
		}
	}

	private void MoveToParent(Transform parent, bool worldPositionStays = true)
	{
		if (!parent)
		{
			throw new ArgumentNullException("parent");
		}
		if (!(rectTransform.parent == parent))
		{
			if (parent == NormalParent)
			{
				rectTransform.SetParent(parent, worldPositionStays);
				rectTransform.SetSiblingIndex(placeHolder.GetSiblingIndex());
				placeHolder.SetParent(rectTransform);
			}
			else if (rectTransform.parent == NormalParent)
			{
				placeHolder.SetParent(NormalParent);
				placeHolder.SetSiblingIndex(rectTransform.GetSiblingIndex());
				rectTransform.SetParent(parent, worldPositionStays);
			}
			else
			{
				rectTransform.SetParent(parent, worldPositionStays);
			}
		}
	}

	public void MoveToParentWhenReordering(Transform parent)
	{
		if (parent == null)
		{
			throw new ArgumentNullException("parent");
		}
		if (placeHolder.parent == rectTransform)
		{
			base.transform.SetParent(parent);
		}
		else
		{
			placeHolder.SetParent(parent);
		}
	}

	private void ApproachTick(Vector3 position, Quaternion rotation, float scale, bool tilting)
	{
		if (rectTransform.localPosition != position)
		{
			rectTransform.localPosition = 0.12f.Lerp(rectTransform.localPosition, position);
		}
		Vector3 vector = new Vector3(scale, scale, 1f);
		if (rectTransform.localScale != vector)
		{
			rectTransform.localScale = 0.12f.Lerp(rectTransform.localScale, vector);
		}
		if (StandardRotation != rotation)
		{
			StandardRotation = 0.12f.Lerp(StandardRotation, rotation);
		}
		if (tilting)
		{
			Vector2 vector2 = (position - rectTransform.localPosition) * 0.001f;
			TiltVector2 += vector2;
		}
		if (TiltVector2.magnitude > 0.001f)
		{
			if (TiltVector2.magnitude > 1f)
			{
				TiltVector2 = TiltVector2.normalized;
			}
			TiltVector2 = 0.05f.Lerp(TiltVector2, Vector2.zero);
			Quaternion quaternion = Quaternion.AngleAxis(15f * TiltVector2.magnitude, new Vector3(TiltVector2.y, 0f - TiltVector2.x));
			rectTransform.localRotation = quaternion * StandardRotation;
		}
		else
		{
			rectTransform.localRotation = StandardRotation;
		}
	}

	private void Approach(Vector3 position, Quaternion rotation, float scale, bool tilting = false)
	{
		while (_tickTime < _accumulatedTime)
		{
			_tickTime += 0.005f;
			ApproachTick(position, rotation, scale, tilting);
		}
	}

	public void SetUsableStatus(bool canUse, bool canKickerUse)
	{
		CanUse = canUse;
		CanKickerUse = canKickerUse;
		UpdateEdge();
	}

	private void UpdateEdge()
	{
		CardWidget.SetCardEdge(CanUse ? (Card.Triggered ? CardWidget.EdgeStatus.High : ((!CanKickerUse) ? CardWidget.EdgeStatus.Afford : CardWidget.EdgeStatus.AffordKicker)) : CardWidget.EdgeStatus.None);
	}

	public void StartHover()
	{
		if (CardWidget.Card == null)
		{
			throw new InvalidOperationException("PointerEnter for hand without <Card> instance.");
		}
		if (!_isActiveHand && !_isHovered)
		{
			MoveToParent(HoveredParent);
		}
		_isHovered = true;
		CostMoreLeft = false;
		CardWidget.ShowTooltip();
		AudioManager.Card();
	}

	public void EndHover()
	{
		_isHovered = false;
		if (ShouldShowLeftCost)
		{
			CostMoreLeft = true;
		}
		if (!_isActiveHand)
		{
			MoveToParent(NormalParent);
		}
		CardWidget.HideTooltip();
	}

	public void CancelUse()
	{
		try
		{
			IsActiveHand = false;
		}
		catch (Exception ex)
		{
			Debug.LogError("[HandCard] CancelUse IsActiveHand=false failed for '" + Card?.DebugName + "': " + ex.Message);
			_isActiveHand = false;
			if ((bool)CanvasGroup)
			{
				CanvasGroup.blocksRaycasts = true;
			}
			if ((bool)NormalParent)
			{
				MoveToParent(NormalParent);
			}
		}
		try
		{
			UpdateEdge();
			EndHover();
			if ((bool)CardWidget)
			{
				GamepadCardController component = CardWidget.GetComponent<GamepadCardController>();
				if ((bool)component)
				{
					component.IsHandCard = true;
					component.EnableNavigation = true;
				}
			}
			UpdateGamepadTipVisibility();
		}
		catch (Exception ex2)
		{
			Debug.LogError("[HandCard] CancelUse for '" + Card?.DebugName + "': " + ex2.Message + "\n" + ex2.StackTrace);
		}
	}

	public void ReturnToHand()
	{
		SetToInactiveWithoutAudio();
		UpdateEdge();
		EndHover();
		GamepadCardController component = CardWidget.GetComponent<GamepadCardController>();
		component.IsHandCard = true;
		component.EnableNavigation = true;
		UpdateGamepadTipVisibility();
	}

	public void Exile()
	{
		ShowShortcut = false;
		CardWidget.Exile().RootObject = base.gameObject;
	}

	public void Remove()
	{
		ShowShortcut = false;
		CardWidget.Remove().RootObject = base.gameObject;
	}

	public void TransformEffect(TransformCardType presentationType)
	{
		CardWidget.PlayTransformEffect(presentationType);
	}

	public void Summon()
	{
		CardWidget.Summon();
	}

	public void OnInputDeviceChanged(InputDeviceType type)
	{
		if (type == InputDeviceType.Gamepad)
		{
			ShowShortcut = false;
		}
		else
		{
			ShowShortcut = GameMaster.ShowShortcut;
		}
		UpdateGamepadTipVisibility();
	}
}
