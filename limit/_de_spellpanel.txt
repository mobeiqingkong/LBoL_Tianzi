using System;
using System.Collections;
using System.Collections.Generic;
using Cysharp.Threading.Tasks;
using DG.Tweening;
using JetBrains.Annotations;
using LBoL.Core;
using LBoL.Presentation.UI.Widgets;
using TMPro;
using UnityEngine;
using UnityEngine.UI;

namespace LBoL.Presentation.UI.Panels;

public class SpellPanel : UiPanel
{
	private sealed class Entry
	{
		[UsedImplicitly]
		public string Title { get; set; }

		[UsedImplicitly]
		public string Name { get; set; }
	}

	[SerializeField]
	private Transform root;

	[SerializeField]
	private Image portrait;

	[SerializeField]
	private Image portraitShadow;

	[SerializeField]
	private TextMeshProUGUI spellName;

	[SerializeField]
	private Image speedLine1;

	[SerializeField]
	private Image speedLine2;

	[SerializeField]
	private Image trail;

	[SerializeField]
	private Image background;

	[SerializeField]
	private ParticleSystem particleRoot;

	[SerializeField]
	private Transform spellRoot;

	private readonly Dictionary<string, SpellDeclareWidget> declareTemplates = new Dictionary<string, SpellDeclareWidget>();

	private string _currentKey;

	private IDisposable _tooltipSuppression;

	private IDictionary<string, Entry> _l10nTable;

	public override PanelLayer Layer => PanelLayer.Top;

	public void Awake()
	{
		root.gameObject.SetActive(value: false);
		spellName.gameObject.SetActive(value: false);
	}

	public override async UniTask CustomLocalizationAsync()
	{
		try
		{
			_l10nTable = await Localization.LoadFileAsync<Entry>("Spell");
		}
		catch (Exception ex)
		{
			Debug.LogError("[SpellPanel] localization failed: " + ex.Message);
			_l10nTable = new Dictionary<string, Entry>();
		}
	}

	public void SetSpellNameByKey(string configKey, string spellKey)
	{
		_currentKey = spellKey;
		if (_l10nTable == null || !_l10nTable.TryGetValue(configKey, out var value))
		{
			value = new Entry();
		}
		Entry entry = value;
		if (entry.Name == null)
		{
			string text2 = (entry.Name = "<Empty>");
		}
		if (value.Title != null)
		{
			spellName.text = "UI.SpellTitled".LocalizeFormat(value.Title, value.Name);
		}
		else
		{
			spellName.text = "UI.SpellUntitled".LocalizeFormat(value.Name);
		}
	}

	public void CallSpellDeclare(Sprite targetSprite, Color[] colors, float x = 0f, float y = 0f, float s = 1f)
	{
		portrait.sprite = targetSprite;
		portraitShadow.sprite = targetSprite;
		trail.color = colors[1];
		speedLine1.color = colors[0];
		speedLine2.color = colors[1];
		portraitShadow.color = colors[2];
		ParticleSystem[] componentsInChildren = particleRoot.GetComponentsInChildren<ParticleSystem>();
		for (int i = 0; i < componentsInChildren.Length; i++)
		{
			ParticleSystem.MainModule main = componentsInChildren[i].main;
			main.startColor = colors[3];
		}
		Transform obj = portrait.transform;
		obj.localPosition = new Vector3(x, y, 0f);
		obj.localScale = new Vector3(s, s, 1f);
		portraitShadow.transform.localScale = new Vector3(s, s, 1f);
		InternalSpellDeclare();
	}

	private void InternalSpellDeclare()
	{
		StopCoroutine("CoInternalSpellDeclare");
		ReleaseTooltipSuppression();
		_tooltipSuppression = TooltipsLayer.SuppressVisuals();
		StartCoroutine("CoInternalSpellDeclare");
	}

	private void ReleaseTooltipSuppression()
	{
		_tooltipSuppression?.Dispose();
		_tooltipSuppression = null;
	}

	private void OnDisable()
	{
		ReleaseTooltipSuppression();
	}

	private IEnumerator CoInternalSpellDeclare()
	{
		spellName.gameObject.SetActive(value: true);
		spellName.GetComponent<RectTransform>().DOAnchorPos(Vector2.zero, 0.6f).From(new Vector2(1000f, 0f));
		spellName.DOFade(1f, 0.2f).From(0f);
		root.gameObject.SetActive(value: true);
		root.DOKill();
		root.GetComponent<CanvasGroup>().DOFade(1f, 0f);
		portrait.DOKill();
		portraitShadow.DOKill();
		Vector2 vector = new Vector2(portrait.transform.localPosition.x, portrait.transform.localPosition.y);
		portrait.GetComponent<RectTransform>().DOAnchorPos(new Vector2(200f, -500f) + vector, 1.5f).From(new Vector2(1000f, -1000f))
			.SetEase(Ease.OutCubic);
		portraitShadow.GetComponent<RectTransform>().DOAnchorPos(new Vector2(0f, -300f) + vector, 1.5f).From(new Vector2(1000f, -1000f))
			.SetEase(Ease.OutCubic);
		background.GetComponent<RectTransform>().DOAnchorPos(new Vector2(1920f, 0f), 8f).From(new Vector2(-1920f, 0f))
			.SetLoops(-1, LoopType.Restart)
			.SetEase(Ease.Linear);
		speedLine1.GetComponent<RectTransform>().DOAnchorPos(new Vector2(5120f, 0f), 0.5f).From(Vector2.zero)
			.SetLoops(-1, LoopType.Restart)
			.SetEase(Ease.Linear);
		speedLine2.GetComponent<RectTransform>().DOAnchorPos(new Vector2(-4096f, 0f), 0.5f).From(Vector2.zero)
			.SetLoops(-1, LoopType.Restart)
			.SetEase(Ease.Linear);
		trail.GetComponent<RectTransform>().DOAnchorPos(new Vector2(0f, 5000f), 0.5f).From(new Vector2(0f, -1000f));
		root.GetComponent<CanvasGroup>().DOFade(0f, 0.4f).From(1f)
			.SetDelay(1.5f);
		particleRoot.Play();
		yield return new WaitForSeconds(2f);
		background.GetComponent<RectTransform>().DOKill();
		speedLine1.GetComponent<RectTransform>().DOKill();
		speedLine2.GetComponent<RectTransform>().DOKill();
		root.gameObject.SetActive(value: false);
		spellName.gameObject.SetActive(value: false);
		ReleaseTooltipSuppression();
	}
}
